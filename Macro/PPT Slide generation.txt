Private Sub SlideGen()

  On Error Resume Next

If MsgBox("Are you sure to generate the slide ?", vbYesNo + vbQuestion, "ELGI-IT Confirmation") = vbYes Then
    strPresPath = Application.ActiveWorkbook.Path & "\FillTmpNew.ppt"
    strNewPresPath = Application.ActiveWorkbook.Path & "\SalesPerf.ppt"
    
    Set oPPTApp = CreateObject("PowerPoint.Application")
    oPPTApp.Visible = msoTrue
    
    Set oPPTOrg = oPPTApp.Presentations.Open(strPresPath, False, False, False)
    
    If oPPTOrg Is Nothing Then
        MsgBox "Template File is missing.Please Check!!!", vbOKOnly + vbInformation, "ELGI IT"
        oPPTApp.Quit
        Set oPPTApp = Nothing
        Exit Sub
    End If
    
  '  Region = InputBox("Please Enter the Vertical Name :", "ELGI-IT")
    
    Region = UCase(ThisWorkbook.Worksheets(1).Cells(1, 1))
    SCur = Trim(ThisWorkbook.Worksheets(1).Cells(1, 12)) & Trim(ThisWorkbook.Worksheets(1).Cells(1, 14)) & "        " & Trim(ThisWorkbook.Worksheets(1).Cells(1, 6)) & Trim(ThisWorkbook.Worksheets(1).Cells(1, 10))
    
    Set oPPTFile = oPPTApp.Presentations.Add(msoTrue)
    
    Set OpptOld = oPPTOrg.Slides(3)
    OpptOld.Copy
    
    oPPTFile.PageSetup.SlideHeight = oPPTOrg.PageSetup.SlideHeight
    oPPTFile.PageSetup.SlideOrientation = oPPTOrg.PageSetup.SlideOrientation
    oPPTFile.PageSetup.SlideSize = oPPTOrg.PageSetup.SlideSize
    oPPTFile.PageSetup.SlideWidth = oPPTOrg.PageSetup.SlideWidth
    
    oPPTFile.Slides.Paste
    
    Application.CutCopyMode = False
    
    SlideNum = 1
    oPPTFile.Slides(1).Select
    
    oPPTFile.Slides(1).Design = OpptOld.Design
    oPPTFile.Slides(1).FollowMasterBackground = OpptOld.FollowMasterBackground
    oPPTFile.Slides(1).ColorScheme = OpptOld.ColorScheme
    
    Set oPPTSales = oPPTFile.Slides(1).Shapes("Head")
    With oPPTSales
         .TextFrame.TextRange.Text = Region
    End With
    
    Set oPPTSales = oPPTFile.Slides(1).Shapes("Shead")
    With oPPTSales
         .TextFrame.TextRange.Text = MonthName(Month(Date) - 1) & " - " & Str(Year(DateAdd("M", -1, Date)))
         
    End With
    
    Set oPPTSales = oPPTFile.Slides(1).Shapes("Tplace")
    With oPPTSales
         .TextFrame.TextRange.Text = SCur
    End With
    
    
    Call FillValues

    Set OpptOld = oPPTOrg.Slides(4)
    OpptOld.Copy
    
    oPPTFile.PageSetup.SlideHeight = oPPTOrg.PageSetup.SlideHeight
    oPPTFile.PageSetup.SlideOrientation = oPPTOrg.PageSetup.SlideOrientation
    oPPTFile.PageSetup.SlideSize = oPPTOrg.PageSetup.SlideSize
    oPPTFile.PageSetup.SlideWidth = oPPTOrg.PageSetup.SlideWidth
    
    oPPTFile.Slides.Paste
    
     
    Application.CutCopyMode = False
    
    SlideNum = oPPTFile.Slides.Count
    oPPTFile.Slides(SlideNum).Select
    
    oPPTFile.Slides(SlideNum).Design = OpptOld.Design
    oPPTFile.Slides(SlideNum).FollowMasterBackground = OpptOld.FollowMasterBackground
    oPPTFile.Slides(SlideNum).ColorScheme = OpptOld.ColorScheme
    
    FName = Mnam & "Sales"
     
    Kill Application.ActiveWorkbook.Path & "\FName.ppt"
    
   ' FName = "test"
  '  FName = InputBox("Enter the Presentation File Name:", "ELGI IT", "SalesPerf")
    
    strNewPresPath = Application.ActiveWorkbook.Path & "\" & FName
    
    oPPTFile.SaveAs strNewPresPath
    oPPTFile.Close
    oPPTOrg.Close
    
    oPPTApp.Quit
    
    Set oPPtHead = Nothing
    Set oPPtDefect = Nothing
    Set oPPTSales = Nothing
    Set oPPTFile = Nothing
    Set oPPTApp = Nothing
    Set oPPTOrg = Nothing
    Set OpptOld = Nothing
  
       
    MsgBox "Presentation Created in path " & strNewPresPath, vbOKOnly + vbInformation, "ELGI IT"
End If
End Sub

Private Sub FillDefect()

            If Trim(ThisWorkbook.Worksheets(2).Cells(DRw, 1)) = Trim(ThisWorkbook.Worksheets(3).Cells(3, 1)) Then

                ThisWorkbook.Worksheets(3).Cells(DRwAct, 3) = Trim(ThisWorkbook.Worksheets(2).Cells(DRw, 4))
                ThisWorkbook.Worksheets(3).Cells(DRwAct, 4) = Trim(ThisWorkbook.Worksheets(2).Cells(DRw, 5))
                ThisWorkbook.Worksheets(3).Cells(DRwAct, 10) = Trim(ThisWorkbook.Worksheets(2).Cells(DRw, 6))
                ThisWorkbook.Worksheets(3).Cells(DRwAct, 11) = Trim(ThisWorkbook.Worksheets(2).Cells(DRw, 7))
