insert into Inventory_cons (COMPANY,CWAR,ITEM,QSKT)
select a.COMPANY,a.T$CWAR,a.T$ITEM,a.T$QSKT
from 
(select distinct COMPANY,T$CWAR,T$ITEM,sum(T$QSKT) T$QSKT from TWHINA112ALL
where T$KOOR in (1,2,32,36)
group by COMPANY,T$CWAR,T$ITEM
)a



update Inventory_cons set stock_955=a.STOC
from 
(select comp,WAR,ITEM,sum(STOC) STOC from Inven_eval_955_to_LN 
group by comp,WAR,ITEM
)a
where Inventory_cons.COMPANY=a.COMP
and Inventory_cons.CWAR=a.WAR
and ltrim(rtrim(Inventory_cons.ITEM))=ltrim(rtrim(a.ITEM))


update Inventory_cons set stock_215=a.QHND
from 
(select COMPANY,CWAR,ITEM,sum(QHND) QHND 
from TWHWMD215ALL
group by COMPANY,CWAR,ITEM
having sum(QHND)>0
)a
where Inventory_cons.COMPANY=a.COMPANY
and Inventory_cons.CWAR=a.CWAR
and ltrim(rtrim(Inventory_cons.ITEM))=ltrim(rtrim(a.ITEM))

update Inventory_cons  set Remarks='LN'
where (qskt >= stock_955 and stock_215 <= stock_955)
and Remarks is null

update Inventory_cons  set Remarks='LN'
where qskt >= stock_215
and Remarks is null

update Inventory_cons  set Remarks='LN'
where isnull(stock_955,0)=0
and Remarks is null

update Inventory_cons  set Remarks='BAAN'
where Remarks is null

update Inventory set stock_955=a.stock_955,Cumulative_stk=a.Cum_Stock,Remarks=a.Remarks
from(
select a.COMPANY,a.CWAR,a.ITEM,trdt,b.stock_955,b.Remarks,
sum(a.QSKT) over (partition by a.COMPANY,a.CWAR,a.ITEM  order by TRDT desc
rows between UNBOUNDED PRECEDING and CURRENT ROW) as Cum_Stock
from Inventory a
inner join  (select * from Inventory_cons where remarks ='BAAN') b
on a.COMPANY=b.Company
and a.CWAR=b.CWAR
and ltrim(rtrim(a.ITEM))=ltrim(rtrim(b.ITEM))
)a
where Inventory.COMPANY=a.Company
and Inventory.CWAR=a.CWAR
and ltrim(rtrim(Inventory.ITEM))=ltrim(rtrim(a.ITEM))
and Inventory.TRDT=a.TRDT


update Inventory set TRDT = a.LRDT
from(
select comp,war,item,max(LRDT) LRDT
--count(distinct LRDT) LRDT 
from Inven_eval_955_to_LN
group by  comp,war,item
having max(LRDT)<>'9999-01-01 00:00:00.000'
)a
where Inventory.COMPANY=a.COMP
and Inventory.CWAR=a.WAR
and ltrim(rtrim(Inventory.ITEM))=ltrim(rtrim(a.ITEM))
and stock_955 > Cumulative_stk
and Inventory.Remarks='BAAN'


update Inventory set Ageing='LessThan30' where datediff(dd,TRDT,getdate()-1) <30
update Inventory set Ageing='30-60' where datediff(dd,TRDT,getdate()-1) between 30 and 60
update Inventory set Ageing='61-90' where datediff(dd,TRDT,getdate()-1) between 61 and 90
update Inventory set Ageing='91-120' where datediff(dd,TRDT,getdate()-1) between 91 and 120
update Inventory set Ageing='121-150' where datediff(dd,TRDT,getdate()-1) between 121 and 150
update Inventory set Ageing='151-180' where datediff(dd,TRDT,getdate()-1) between 151 and 180
update Inventory set Ageing='181-365' where datediff(dd,TRDT,getdate()-1) between 181 and 365
update Inventory set Ageing='GreaterThan_1Year' where datediff(dd,TRDT,getdate()-1) between 366 and 730
update Inventory set Ageing='GreaterThan_2Years' where datediff(dd,TRDT,getdate()-1) between 731 and 1095
update Inventory set Ageing='GreaterThan_3Years' where datediff(dd,TRDT,getdate()-1) between 1096 and 1460
update Inventory set Ageing='GreaterThan_4Years' where datediff(dd,TRDT,getdate()-1) > 1460




update Inventory set AgeingOrdr=1 where Ageing='LessThan30'
update Inventory set AgeingOrdr=2 where Ageing='30-60'
update Inventory set AgeingOrdr=3 where Ageing='61-90'
update Inventory set AgeingOrdr=4 where Ageing='91-120'
update Inventory set AgeingOrdr=5 where Ageing='121-150'
update Inventory set AgeingOrdr=6 where Ageing='151-180'
update Inventory set AgeingOrdr=7 where Ageing='181-365'
update Inventory set AgeingOrdr=8 where Ageing='GreaterThan_1Year'
update Inventory set AgeingOrdr=9 where Ageing='GreaterThan_2Years'
update Inventory set AgeingOrdr=10 where Ageing='GreaterThan_3Years'
update Inventory set AgeingOrdr=11 where Ageing='GreaterThan_4Years'
update Inventory set AgeingOrdr=100 where Ageing ='NA'


delete from Inventory where QSKT is null


end
