
select a.CUNO,nama as CustName,sum(BalanceAmt) from (

--AMNT,AMTI,

(select 
TTYP,NINV,LINE,TDOC,DOCN,LINO,CUNO,DOCD as Invdate,TREC,CCUR,RATE,RATF,SPOT,AMNT,AMTI,AMTH as invAmt ,VATC,CVAT,VATA,VATI,VATH,SVAM,SVAH,LPDT,

a.TTYP,a.NINV,a.LINE,TDOC,DOCN,LINO,a.CUNO,a.Invdate,TREC,CCUR,RATE,RATF,SPOT,
--AMNT,AMTI,
VATC,CVAT,VATA,VATI,VATH,SVAM,SVAH,LPDT,

VATP,CREP,LIQD,ADRS,LAMT,STEP,RCPT,DOUB,CAIN,LVAT,TYPA,DOCA,PCUN,TEXT,CREG
