update Tcodeprocess_from_BAAN set ddate=a.TB_date
from (
select distinct
convert(datetime,convert(varchar,YEAR)+'-'+right('00'+cast(prod as varchar(2)),2)+'-'+convert(varchar,day(EOMONTH(getdate()-1)))) as TB_date
from ttegld257_TCODE 
where prod =(select max(PROD) from ttegld257_TCODE where YEAR=(select max(YEAR) from ttegld257_TCODE))
and YEAR=(select max(YEAR) from ttegld257_TCODE)
)a

--Step 3: Updating lamt column in Tcodeprocess_from_BAAN table from the tables 
--such as ttegld257_tcode,TCODE_ICT,TCODE_CONS_ICT

update Tcodeprocess_from_BAAN set 
lamt=AB.TB_Amount-AB.ICT_Amnt-AB.CON_ICT_Amnt
from
(
select 
--main.*,
concat(ltrim(rtrim(main.comp)),ltrim(rtrim(main.code))) MainCombiCode,
A.TB_date,
isnull(A.Amount,0) TB_Amount,
isnull(B.Amount,0) ICT_Amnt,
isnull(C.Debit_Credit,0) CON_ICT_Amnt

from Tcodeprocess_from_BAAN main
left outer join 
(
select Company,CombiCode,TB_date,sum(Amount) as Amount from 
(
select OCMP as Company,CODE as CombiCode,
convert(datetime,convert(varchar,YEAR)+'-'+right('00'+cast(prod as varchar(2)),2)+'-'+convert(varchar,day(EOMONTH(getdate()-1)))) as TB_date,
CBAL as Amount from ttegld257_TCODE where
prod =(select max(PROD) from ttegld257_TCODE where YEAR=(select max(YEAR) from ttegld257_TCODE))
and YEAR=(select max(YEAR) from ttegld257_TCODE)
)a
group by Company,CombiCode,TB_date
) A
on concat(ltrim(rtrim(main.comp)),ltrim(rtrim(main.code)))=ltrim(rtrim(A.CombiCode))
left outer join TCODE_ICT B
on concat(ltrim(rtrim(main.comp)),ltrim(rtrim(main.code)))=ltrim(rtrim(B.TCode))
left outer join TCODE_CONS_ICT C
on concat(ltrim(rtrim(main.comp)),ltrim(rtrim(main.code)))=ltrim(rtrim(c.[Combi code]))
)AB
where 
concat(ltrim(rtrim(Tcodeprocess_from_BAAN.comp)),ltrim(rtrim(Tcodeprocess_from_BAAN.code)))=AB.MainCombiCode
--order by main.fcode,main.slno,main.subd,main.ordr,concat(ltrim(rtrim(main.comp)),ltrim(rtrim(main.code)))

update Tcodeprocess_from_BAAN set lamt=lamt*(-1)
where slno<=10

where slno in(22,23)
