sum(sales_amount) sales_INR,Currency,rate_usd,
sum(Sales_USD) Sales_USD,Fin_year,Fin_month,Monthname
FROM 	
ELGi_Sales_Details a 
WHERE	(a.Invoice_Date BETWEEN '04/01/2019' AND GETDATE()-1)

--select SaleOrderType from SaleType_Master where Sales_Category in (0,2,5,6)
--union all
--select '' as SaleOrderType from SaleType_Master
--)
and a.sales_category not in (1)
and Dpst='90231'
and areacode='634'
group by Organ,Invoice_Date,company,trantype,Invoice,dpst,dpstname,areacode,areaname,custcode,custname,Currency,rate_usd,Fin_year,Fin_month,Monthname
order by Invoice_Date
