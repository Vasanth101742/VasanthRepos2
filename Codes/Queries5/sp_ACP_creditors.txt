(select COMPANY,TTYP,NINV,DOCD,DOCN,IFBP,LINE,

else DUED end as DueDate from TTFACP200ALL
where DOCN=0
and concat(convert(varchar,year),RIGHT('00'+CAST(prod AS VARCHAR(2)),2))<=CONCAT(@Finyear,@Finmonth)
--and NINV=17000003
--and IFBP='SU1S02374'
--and COMPANY=411

)a

(select COMPANY,IFBP,TTYP,NINV,LINE,ccur as Currency,

SUM(convert(numeric,ISNULL(AMTH$1,0)+ISNULL(PADH$1,0)+ISNULL(DC1H$1,0)+ISNULL(CDAM$1,0))) as PaidAmnt

from TTFACP200ALL
where DOCN<>0
and concat(convert(varchar,year),RIGHT('00'+CAST(prod AS VARCHAR(2)),2))<=CONCAT(@Finyear,@Finmonth)
--and NINV=16000577
--and IFBP='SU1EH0004'
--and COMPANY=465
group by COMPANY,IFBP,TTYP,NINV,LINE,ccur
)b
on a.company=b.company
and a.NINV=b.NINV
and a.TTYP=b.TTYP
and a.IFBP=b.IFBP
and a.LINE=b.LINE
--and a.Currency=b.Currency

group by a.DOCD,a.COMPANY,a.TTYP,a.NINV,a.IFBP,a.DueDate,
FinYear,FinMonth,CalYear,CalMonth,a.CurRate,a.Currency,a.LINE
)a
--where suppcode='SU1E00483'
