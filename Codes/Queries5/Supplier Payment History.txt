[YEAR],PROD
from TTFACP200ALL
where DOCN<>0
----and concat(convert(varchar,year),RIGHT('00'+CAST(prod AS VARCHAR(2)),2))<=CONCAT(@Finyear,@Finmonth)
----and NINV=16000577
----and IFBP='SU1EH0004'
