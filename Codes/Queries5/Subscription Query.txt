update BI_ReportSubscriptions set ReportName=a.Name,
ReportCreatorID=a.CreatedByID,
CreationDate=a.CreationDate,
ReportModifierID=a.ModifiedByID,
ModifiedDate=a.ModifiedDate
from [Catalog] a
where BI_ReportSubscriptions.ReportID= a.ItemID

update BI_ReportSubscriptions set SubscriptionOwnerID=a.OwnerID,
SubscriptionModifierID=a.ModifiedByID,
--SubscriptionModifiedBy
SubscriptionDesc=a.[Description],
LastStatus=a.LastStatus,
LastRunTime=a.LastRunTime
from [Subscriptions] a

update BI_ReportSubscriptions set ScheduleName=a.Name,
StartDate=a.StartDate,
SchNextRunTime=a.NextRunTime,
ScheduleType=a.Type,
ScheduleState=a.State
from
(select c.ScheduleID,c.Name,c.StartDate,c.CreatedById,c.EventType,c.LastRunStatus,c.LastRunTime,
c.Type,c.State,c.NextRunTime,Path
from [dbo].[Schedule] c)a
where BI_ReportSubscriptions.ScheduleID=a.ScheduleID


update BI_ReportSubscriptions set ReportCreatedBy=a.UserName
from [dbo].[Users] a
where BI_ReportSubscriptions.ReportCreatorID=a.UserID

update BI_ReportSubscriptions set ReportModifiedBy=a.UserName
from [dbo].[Users] a
where BI_ReportSubscriptions.ReportModifierID=a.UserID

update BI_ReportSubscriptions set SubscriptionOwner=a.UserName
from [dbo].[Users] a
where BI_ReportSubscriptions.SubscriptionOwnerID=a.UserID

update BI_ReportSubscriptions set SubscriptionModifiedBy=a.UserName
from [dbo].[Users] a
where BI_ReportSubscriptions.SubscriptionModifierID=a.UserID

select * from BI_ReportSubscriptions
