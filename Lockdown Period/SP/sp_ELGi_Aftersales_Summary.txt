Declare @FromDate date = case when month(getdate()-1)<4 then convert(varchar,year(getdate()-1)-1)+'-04-01' else convert(varchar,year(getdate()-1))+'-04-01' end

year(Date) year,Month,Monthaname,

year(Date),Month,Monthaname,YTDFlag
