WareHouse,Price,qty,discount,sales,SalesRepCode,LOBCode,DeliveryTerms,

l.Company,l.Trantype,l.invnum,l.OrderNum,convert(date,l.OrderDate) ,l.PositionNum,l.CustCode,l.ItemCode,
l.DeliveryDate,l.WareHouse,l.Price,l.t_qoor,l.t_amld,
(l.t_oamt* (case when h.t_rats_1=0 or h.t_rats_1 is null then 1 else h.t_rats_1 end)) as sales,
h.SalesRep,h.LineOfBusiness,
h.DeliveryTerms,l.t_oamt as Foreign_currency_amount,
h.Currency,h.ForwardAgent,h.PaymentTerm,h.t_rats_1
from (select * From Sales_Header where company in (select Company from ELGI_Companies_Master where CompanyGroup='ELGI')) h
inner join Sales_Lines L
on h.Company =l.Company 
and ltrim(rtrim(h.OrderNum))=ltrim(rtrim(L.OrderNum))
where LTRIM(RTRIM(L.ItemCode)) not in ('FRT' ,'EXPORT-FREIGHT','M000037703')
and t_sqnb =0
and h.t_clyn<>1
and l.t_clyn<>1

l.Company,l.Trantype,l.invnum,l.OrderNum,convert(date,l.OrderDate) ,l.PositionNum,l.CustCode,l.ItemCode,
