as
begin


TRUNCATE TABLE production_plan_vs_actual
TRUNCATE TABLE ELGi_production_details
TRUNCATE TABLE LEPSAC_Items_Tmp1
TRUNCATE TABLE ELGi_Production_Backlogs
TRUNCATE TABLE PDNO_CANCELLED
TRUNCATE TABLE Production_Summary
Truncate table WIP_Ageing 
Truncate table Production_ALC
Truncate table Production_ALCdate_Temp
Truncate table Production_Summary_Temp
Truncate Table Production_ALC_Tmp
Truncate Table To_be_Completed_Ageing


--Planned
insert  into production_plan_vs_actual
--select * into production_plan_vs_actual from (
select distinct
convert(date,RDLD) as date1,
company,
PDNO as Prodorderno,
CWAR as Warehouse,
a.MITM as Item,
sum(case when RDLD='9999-01-01 00:00:00.000' then 0 else (case when osta>=1 then qrdr else 0 end) end) as Planned_qty,
0 as Released_qty,
0 as Actual_qty,
0 as WIP_qty,
0 as avg_bud_price,
0 as sls_price,
0 as Dispatch_Qty
from TTISFC001ALL a 
--where a.PDNO not like '%[a-z]%'
--a.PDNO not like '%p%'
--and a.PDNO not like '%sc%'
--where a.PDNO='290000207' --for testing purpose
group by convert(date,RDLD),company,PDNO,CWAR,MITM,OSTA




--Released
insert  into production_plan_vs_actual
select distinct
convert(date,ReleaseDt) as date1,
company,
PDNO as Prodorderno,
CWAR as Warehouse,
a.MITM as Item,
0 as Planned_qty,
sum(case when ADLD='9999-01-01 00:00:00.000' then 0 else (case when osta>=4 then qrdr else 0 end) end) as Released_qty,
0 as Actual_qty,
0 as WIP_qty,
0 as avg_bud_price,
0 as sls_price,
0 as Production_Qty
from (
select a.*,b.date as ReleaseDt from  TTISFC001ALL a 
left outer join
(
select * from (
select company,PDNO,OSTA,convert(date,date) as date,row_number() over (partition by company,pdno order by date) dateRank 
from TTICST151ALL
where OSTA in (4,6)
--and PDNO='290000207' --for testing Purpose
)b1
where dateRank=1
)b
on a.company=b.company
and a.PDNO=b.PDNO
--where a.PDNO not like '%[a-z]%'
--a.PDNO not like '%p%'
--and a.PDNO not like '%sc%'
--and a.PDNO='230009816' --for testing Purpose
) a

--where a.PDNO not like '%[a-z]%'
--a.PDNO not like '%p%'
--and a.PDNO not like '%sc%'
--and a.PDNO='230009816' --for testing purpose
group by ReleaseDt,company,PDNO,CWAR,MITM,OSTA



--WIP
insert  into production_plan_vs_actual
select distinct
convert(date,WIP_Dt) as date1,
company,
PDNO as Prodorderno,
CWAR as Warehouse,
a.MITM as Item,
0 as Planned_qty,
0 as Released_qty,
0 as Actual_qty,
sum(case when ADLD='9999-01-01 00:00:00.000' then 0 else (case when osta=6 then qrdr else 0 end) end) as WIP_qty,
0 as avg_bud_price,
0 as sls_price,
0 as Production_Qty
from (
select a.*,b.date as WIP_Dt from  TTISFC001ALL a 
left outer join
(
select * from (
select COMPANY,PDNO,OSTA,convert(date,date) as date,row_number() over (partition by COMPANY,pdno order by date) dateRank 
from TTICST151ALL
where OSTA in (6)
--and PDNO='290000207' --for testing Purpose
)b1
where dateRank=1
)b
on a.COMPANY=b.COMPANY
and a.PDNO=b.PDNO
--where a.pdno='290000207'
--where a.PDNO not like '%[a-z]%'
--a.PDNO not like '%p%'
--and a.PDNO not like '%sc%'
--and a.PDNO='230009816' --for testing Purpose
) a
--where a.PDNO not like '%[a-z]%'
--a.PDNO not like '%p%'
--and a.PDNO not like '%sc%'
--and a.PDNO='230009816' --for testing purpose
group by WIP_Dt,company,PDNO,CWAR,MITM,OSTA


--Actual

insert  into production_plan_vs_actual
select convert(date,TRDT) TRDT,a.COMPANY,
ORNO,a.CWAR,item,0,0,
sum(QSTK) as Actual_qty
,0,0,sum(isnull(sls_price,0)) as sls_price,
0
--,sum(convert(float,isnull(c.QDLV,0))) as QDLV
from 
(select distinct company,PDNO,MITM from TTISFC001ALL )a1
left outer join
(select * from TWHINR110ALL where KOST=3 and KOOR=1
and company in (select company from elgi_companies_master where companygroup='ELGI')
--and ORNO not like '%[a-z]%'
--ORNO not like '%p%'
--and ORNO not like '%sc%'
) a
on a1.company=a.company
and a1.PDNO=a.ORNO
and ltrim(rtrim(a1.MITM))=ltrim(rtrim(a.item))

left outer join
(
select a.Company,a.itemcode,max(Price) as sls_price  from Sales_Lines a
inner join
(select itemcode, max(deliverydate) as date  from Sales_Lines  group by itemcode) b
on a.itemcode=b.itemcode 
and a.deliverydate=b.date  
group by a.Company,a.itemcode
)b
on a.COMPANY=b.Company 
and	ltrim(rtrim(a.item))=ltrim(rtrim(b.itemcode))
group by convert(date,TRDT),a.COMPANY,ORNO,a.CWAR,item
order by convert(date,TRDT)



insert  into production_plan_vs_actual
select convert(date,CLDT) CLDT, --CLDT : Actual Closing Date
a.COMPANY,PDNO,a.CWAR,MITM,0,0,
sum(QRDR) as Actual_qty
,0,0,0,0
from TTISFC001ALL a 
where company in (select company from elgi_companies_master where companygroup='ATS')
AND OSTA=9

--where a.PDNO not like '%[a-z]%'
--a.PDNO not like '%p%'
--and a.PDNO not like '%sc%'
--and a.PDNO='261004462' --for testing purpose

group by convert(date,CLDT),company,PDNO,CWAR,MITM,OSTA



--Dispatch_qty


Insert into production_plan_vs_actual
SELECT convert(date,b.t_idat) t_idat,a.company,null as Prodorderno,c.WareHouse,t_item,
0 as Planned_qty,0 as Released_qty,0 as Actual_qty,0 as WIP_qty,
0 as avg_bud_price,0 as sls_price,sum(isnull(t_dqua,0)) t_dqua
--t_cofc,Trantype,a.OrderNum,a.PositionNum,a.invnum,a.CustCode,a.t_rate_1 ,a.t_rate_2 ,a.t_txai,
--sum(isnull(a.t_amth_1,0)) t_amth_1,sum(isnull(a.t_txah_1,0)) t_txah_1,sum(isnull(a.t_amti,0)) t_amti,sum(isnull(t_pric,0)),
--sum(isnull(a.t_amth_1,0)),saletype
FROM CI_Lines a
inner join CI_Header b
on a.Company=b.Company 
and ltrim(rtrim(a.Trantype)) =ltrim(rtrim(b.Trantyp ))
and ltrim(rtrim(a.invnum)) =ltrim(rtrim(b.invnum)) 
inner join (select distinct Company,OrderNum,--PositionNum,
WareHouse from Sales_Lines where ltrim(rtrim(WareHouse))<>'') c
on  a.company=c.Company
and a.OrderNum=c.OrderNum
--and a.PositionNum=c.PositionNum
--where ltrim(rtrim(a.t_item))= 'X990614'
--and convert(date,b.t_idat) in ('2018-09-03','2018-09-05','2018-09-11','2018-09-12','2018-09-13','2018-09-14')
group by convert(date,b.t_idat),a.company,c.WareHouse,t_item
--t_cofc,Trantype,a.OrderNum,a.PositionNum,a.invnum,a.CustCode,t_item ,a.t_rate_1 ,a.t_rate_2 ,a.t_txai,saletype



insert into ELGi_production_details(company,warehouse,item,itemdescr,Description,
Prodorderno,date1,sls_price,Planned_qty,Released_qty,
Actual_qty,WIP_qty,avg_bud_price,Dispatch_qty,CalMonth,CalYear,
MonthPlan,Item_DayRnk,Item_MnthRnk,SalesOrderNo,
CustCode,CustName)
select a.company,a.warehouse,a.item,null as itemdescr,null as Description,
a.Prodorderno,a.date1,sls_price,
sum(a.Planned_qty) Planned_qty,
sum(a.Released_qty) Released_qty,
sum(a.Actual_qty) Actual_qty,
sum(a.WIP_qty) WIP_qty,
sum(a.avg_bud_price) avg_bud_price,
sum(Dispatch_qty) Dispatch_qty,
--,row_number() over(partition by a.company,a.warehouse,a.item,date1 order by a.company,a.warehouse,a.item,date1) as Rnk
month(a.date1) as CalMonth,
year(a.date1) as CalYear,
null as MonthPlan,
row_number() over(partition by a.company,a.warehouse,a.item,a.date1 order by a.company,a.warehouse,a.item,a.date1) as Item_DayRnk,
row_number() over(partition by a.company,a.warehouse,a.item,year(a.date1),month(a.date1) 
order by a.company,a.warehouse,a.item,year(a.date1),month(a.date1)) as Item_MnthRnk,
null as SaleOrderno,null as CustCode,null as CustName
from production_plan_vs_actual a
where ltrim(rtrim(a.Warehouse))<>''
group by a.company,a.warehouse,a.item,a.Prodorderno,a.date1,sls_price

--Truncate table WIP_Ageing
Insert into WIP_Ageing 
--select * into WIP_Ageing from (
select company,warehouse,item,itemdescr,Description,Prodorderno,
date1,WIPPendingDays,NationalHolidays,Days_in_WIP,Esclvl1,
Esclvl2,Escalationlevel,
ROW_NUMBER() over (partition by Escalationlevel order by date1) as rownum,
null as Escl_Comments,null as FabNo,null as PlanDelDate,null as OTADate,
WIP_qty
from (
select *,Esclvl1+Esclvl2 as Escalationlevel from (
select *,WIPPendingDays-NationalHolidays as Days_in_WIP,
case when Description='Lepsac' and WIPPendingDays-NationalHolidays =3 then 1 
when Description='OFSAC' and WIPPendingDays-NationalHolidays =3 then 1
when Description='Dpsac' and WIPPendingDays-NationalHolidays =2 then 1
when Description='Encap' and WIPPendingDays-NationalHolidays =2 then 1
when Description='Sepsac' and WIPPendingDays-NationalHolidays =2 then 1
when Description like '%RCD%' and WIPPendingDays-NationalHolidays =1 then 1
when Description like '%ENGC%' and WIPPendingDays-NationalHolidays =1 then 1
when Description like 'PVD' and WIPPendingDays-NationalHolidays =1 then 1
when Description like 'Airend' and WIPPendingDays-NationalHolidays =1 then 1
when Description like 'TOPBLOCK' and WIPPendingDays-NationalHolidays =1 then 1
when Description like 'Motor Division' and WIPPendingDays-NationalHolidays =1 then 1
when Description like 'Others' and WIPPendingDays-NationalHolidays =1 then 1
else 0 end as Esclvl1,
case when Description='Lepsac' and WIPPendingDays-NationalHolidays >=4 then 2 
when Description='OFSAC' and WIPPendingDays-NationalHolidays >=4 then 2
when Description='Dpsac' and WIPPendingDays-NationalHolidays >=3 then 2
when Description='Encap' and WIPPendingDays-NationalHolidays >2 then 2
when Description='Sepsac' and WIPPendingDays-NationalHolidays >2 then 2
when Description like '%RCD%' and WIPPendingDays-NationalHolidays >1 then 2
when Description like '%ENGC%' and WIPPendingDays-NationalHolidays >1 then 2
when Description like 'PVD' and WIPPendingDays-NationalHolidays >1 then 2
when Description like 'Airend' and WIPPendingDays-NationalHolidays >1 then 2
when Description like 'TOPBLOCK' and WIPPendingDays-NationalHolidays >1 then 2
when Description like 'Motor Division' and WIPPendingDays-NationalHolidays >1 then 2
when Description like 'Others' and WIPPendingDays-NationalHolidays >1 then 2
else 0 end as Esclvl2


--case when WIPPendingDays-NationalHolidays >=6  then 1 else 0   end as Esclvl3
from (
select company,warehouse,item,itemdescr,Description,Prodorderno,date1,
(DATEDIFF(dd, date1, getdate()-1) + 1)
-(DATEDIFF(wk, date1, getdate()-1) * 2)
