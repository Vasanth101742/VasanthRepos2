select * , 
(select count(distinct DATE) from CompanyCalendar where 
(convert(date,date) between convert(date,Targetdate_otp) and (dateadd(dd,-1,convert(date,case when ( delivery_date='1900-01-01 00:00:00.000' or delivery_date='1970-01-01 00:00:00.000')then getdate()
 else delivery_date end)))and CCAL ='company' and DFCA<>4))AS OTP_GAPDAYS
	
	,0 as score_0,0 as score_100,0 as score_blank,
 0 as Finalscore_0,0 as Finalscore_100,0 as Finalscore_blank,
	0 as final_score_max_blank,0 as final_score_max_0,0 as final_score_max_100
 
	from(

select *,dateadd(dd,case when TargetLeadTime=0 then 0 else (TargetLeadTime-workdays) end,Target_date_old)as Target_date From (
select *,(select count(distinct DATE) from CompanyCalendar where 
(convert(date,date) between convert(date,Clearence_Date) and convert(date,dateadd(dd,TargetLeadTime,Clearence_Date))
 and CCAL ='company' and DFCA<>4))as workdays,
	(select count(distinct DATE) from CompanyCalendar where 
(convert(date,date) between convert(date,Clearence_Date) and (dateadd(dd,-1,convert(date,case when (Completed_Date='1900-01-01 00:00:00.000' or Completed_Date='1970-01-01 00:00:00.000')then getdate()
 else completed_date end)))and CCAL ='company' and DFCA<>4)) as LeadtimeOTA,'                     ' as Remarks,
	(select count(distinct DATE) from CompanyCalendar where 
(convert(date,date) between convert(date,Clearence_Date) and (dateadd(dd,-1,convert(date,case when (Prod_Gen_Date ='1900-01-01 00:00:00.000' or Prod_Gen_Date='1970-01-01 00:00:00.000')then getdate()
 else Prod_Gen_Date end)))and CCAL ='company' and DFCA<>4))AS OTG_GAPDAYS,
	dateadd(dd,otp,Clearence_Date)as Targetdate_otp From UTN_NEW)a
	)a
