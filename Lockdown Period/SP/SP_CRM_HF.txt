CREATE PROC [dbo].[SP_CRM_HF] @fromdate date,@todate date
AS
BEGIN

truncate table  HF_VISITS
truncate table HF_ENQ
truncate table HF_WON
truncate table Productive_Visits_through_visits
truncate table Productive_Visits_Direct_Enquiries
truncate table CRM_HF_FINAL
truncate table CRM_HF_FINAL_ver1


INSERT INTO HF_VISITS 

select *  From (
select a.*,b.type,'Dealer' as new,row_number()over(partition by account_id order by account_id) as distcust
  From(
select 
CustomerName,Account_ID,CityName,StateName,Customer_Cluster,Customer_Classification,Industry,
Dealer,Dealer_Engineer,EnquiryID,CustomerVisitID,MonthstartDate,VisitDate,VisitStatus,
EnquiryStage,EnquiryStatus,Probability,EstimatedCloseDate,ActualCloseDate,Createdon,
OwnerName,SalesOffice,ItemCode,Product,priceperunit,quantity,baseamount,ProductCategory,
opportunityid,ownerid,Region,elgi_businessverticalname,Quotedate,
Negotiationdate,elgi_losttoname,elgi_reasonname,elgi_remarks,elgi_numbers,elgi_pricewelosttheorderat,
elgi_model,pincode,businessunitidname,elgi_customertypesalesname,elgi_cityname,elgi_statename,
elgi_locationofcustomer,elgi_lineofbusinessname,Enq_Desc
  From crm_dealer where convert(date,visitdate) between @fromdate and @todate 
and (elgi_businessverticalname='Industrial' or elgi_businessverticalname='Blank'))a
left join (select distinct  * From productive_Master)b
on a.VisitStatus=b.[CRM Visit Status]

union all

select  a.*,b.type,'Eng' as new,row_number()over(partition by account_id order by account_id) as distcust
  From(
select 
CustomerName,Account_ID,CityName,StateName,Customer_Cluster,Customer_Classification,Industry,
Dealer,Dealer_Engineer,EnquiryID,CustomerVisitID,MonthstartDate,VisitDate,VisitStatus,
EnquiryStage,EnquiryStatus,Probability,EstimatedCloseDate,ActualCloseDate,Createdon,
OwnerName,SalesOffice,ItemCode,Product,priceperunit,quantity,baseamount,ProductCategory,
opportunityid,ownerid,Region,elgi_businessverticalname,Quotedate,
Negotiationdate,elgi_losttoname,elgi_reasonname,elgi_remarks,elgi_numbers,elgi_pricewelosttheorderat,
elgi_model,pincode,businessunitidname,elgi_customertypesalesname,elgi_cityname,elgi_statename,
elgi_locationofcustomer,elgi_lineofbusinessname,Enq_Desc
  From CRM_ENGINEER where convert(date,visitdate) between @fromdate and @todate 
and (elgi_businessverticalname='Industrial' or elgi_businessverticalname='Blank'))a
left join (select distinct  * From productive_Master)b	
on a.VisitStatus=b.[CRM Visit Status])a


delete from HF_VISITS where Account_ID is null


INSERT INTO  HF_ENQ

select * from (
select 
CustomerName,Account_ID,CityName,StateName,Customer_Cluster,Customer_Classification,Industry,
Dealer,Dealer_Engineer,EnquiryID,CustomerVisitID,MonthstartDate,VisitDate,VisitStatus,
EnquiryStage,EnquiryStatus,Probability,EstimatedCloseDate,ActualCloseDate,Createdon,
OwnerName,SalesOffice,ItemCode,Product,priceperunit,quantity,baseamount,ProductCategory,
opportunityid,ownerid,Region,elgi_businessverticalname,Quotedate,
Negotiationdate,elgi_losttoname,elgi_reasonname,elgi_remarks,elgi_numbers,elgi_pricewelosttheorderat,
elgi_model,pincode,businessunitidname,elgi_customertypesalesname,elgi_cityname,elgi_statename,
elgi_locationofcustomer,elgi_lineofbusinessname,Enq_Desc
,'' as type,'Dealer' as new,
row_number()over(partition by account_id order by account_id) as distcust
  From crm_dealer where convert(date,createdon) between @fromdate and @todate
and (elgi_businessverticalname='Industrial' or elgi_businessverticalname='Blank')
and ProductCategory in ('s0','s1','s2')

union all

select 
CustomerName,Account_ID,CityName,StateName,Customer_Cluster,Customer_Classification,Industry,
Dealer,Dealer_Engineer,EnquiryID,CustomerVisitID,MonthstartDate,VisitDate,VisitStatus,
EnquiryStage,EnquiryStatus,Probability,EstimatedCloseDate,ActualCloseDate,Createdon,
OwnerName,SalesOffice,ItemCode,Product,priceperunit,quantity,baseamount,ProductCategory,
opportunityid,ownerid,Region,elgi_businessverticalname,Quotedate,
Negotiationdate,elgi_losttoname,elgi_reasonname,elgi_remarks,elgi_numbers,elgi_pricewelosttheorderat,
elgi_model,pincode,businessunitidname,elgi_customertypesalesname,elgi_cityname,elgi_statename,
elgi_locationofcustomer,elgi_lineofbusinessname,Enq_Desc
,'' as type,'Eng' as new ,
row_number()over(partition by account_id order by account_id) as distcust
 From crm_engineer where convert(date,createdon) between @fromdate and @todate
and (elgi_businessverticalname='Industrial' or elgi_businessverticalname='Blank')
and ProductCategory in ('s0','s1','s2'))a

delete from  HF_ENQ where Account_ID is null


INSERT INTO  HF_WON
select * from (
select 
CustomerName,Account_ID,CityName,StateName,Customer_Cluster,Customer_Classification,Industry,
Dealer,Dealer_Engineer,EnquiryID,CustomerVisitID,MonthstartDate,VisitDate,VisitStatus,
EnquiryStage,EnquiryStatus,Probability,EstimatedCloseDate,ActualCloseDate,Createdon,
OwnerName,SalesOffice,ItemCode,Product,priceperunit,quantity,baseamount,ProductCategory,
opportunityid,ownerid,Region,elgi_businessverticalname,Quotedate,
Negotiationdate,elgi_losttoname,elgi_reasonname,elgi_remarks,elgi_numbers,elgi_pricewelosttheorderat,
elgi_model,pincode,businessunitidname,elgi_customertypesalesname,elgi_cityname,elgi_statename,
elgi_locationofcustomer,elgi_lineofbusinessname,Enq_Desc
,'' as type,'Dealer' as New,row_number()over(partition by account_id order by account_id) as distcust
  From crm_dealer where convert(date,ActualCloseDate) between @fromdate and @todate
and (elgi_businessverticalname='Industrial' or elgi_businessverticalname='Blank')
and ProductCategory in ('s0','s1','s2')
and EnquiryStatus='Won'

union all

select 
CustomerName,Account_ID,CityName,StateName,Customer_Cluster,Customer_Classification,Industry,
Dealer,Dealer_Engineer,EnquiryID,CustomerVisitID,MonthstartDate,VisitDate,VisitStatus,
EnquiryStage,EnquiryStatus,Probability,EstimatedCloseDate,ActualCloseDate,Createdon,
OwnerName,SalesOffice,ItemCode,Product,priceperunit,quantity,baseamount,ProductCategory,
opportunityid,ownerid,Region,elgi_businessverticalname,Quotedate,
Negotiationdate,elgi_losttoname,elgi_reasonname,elgi_remarks,elgi_numbers,elgi_pricewelosttheorderat,
elgi_model,pincode,businessunitidname,elgi_customertypesalesname,elgi_cityname,elgi_statename,
elgi_locationofcustomer,elgi_lineofbusinessname,Enq_Desc
,'' as type,'Eng'as New,
row_number()over(partition by account_id order by account_id) as distcust
  From CRM_ENGINEER where convert(date,ActualCloseDate) between @fromdate and @todate
and (elgi_businessverticalname='Industrial' or elgi_businessverticalname='Blank')
and ProductCategory in ('s0','s1','s2')
and EnquiryStatus='Won')a

delete from  HF_WON where Account_ID is null

update HF_VISITS set distcust=0 where distcust>1
update HF_ENQ set distcust=0 where distcust>1
update HF_WON set distcust=0 where distcust>1


UPDATE HF_ENQ SET VisitDate=a.VisitDate from 
(SELECT Account_ID,min(VisitDate)VisitDate fROM HF_VISITS where visitdate is not null and new='Dealer' group by Account_ID)a
where HF_ENQ.Account_ID=a.Account_ID and new='Dealer'

UPDATE HF_ENQ SET VisitDate=a.VisitDate from 
(SELECT Account_ID,min(VisitDate)VisitDate fROM HF_VISITS where visitdate is not null and new='Eng' group by Account_ID)a
where HF_ENQ.Account_ID=a.Account_ID and new='Eng'

UPDATE HF_WON SET VisitDate=a.VisitDate from 
(SELECT Account_ID,min(VisitDate)VisitDate fROM HF_VISITS where visitdate is not null and new='Dealer' group by Account_ID)a
where HF_won.Account_ID=a.Account_ID and new='Dealer'

UPDATE HF_WON SET VisitDate=a.VisitDate from 
(SELECT Account_ID,min(VisitDate)VisitDate fROM HF_VISITS where visitdate is not null and new='Eng' group by Account_ID)a
where HF_won.Account_ID=a.Account_ID and new='Eng'

UPDATE HF_WON SET Createdon=a.Createdon from 
(SELECT Account_ID,min(Createdon)Createdon fROM HF_ENQ where Createdon is not null and new='Dealer' group by Account_ID)a
where HF_won.Account_ID=a.Account_ID and new='Dealer'

UPDATE HF_WON SET Createdon=a.Createdon from 
(SELECT Account_ID,min(Createdon)Createdon fROM HF_ENQ where Createdon is not null and new='Eng' group by Account_ID)a
where HF_won.Account_ID=a.Account_ID and new='Eng'

insert into Productive_Visits_through_visits -- Dealer
select *,row_number()over(partition by account_id order by account_id) as distcust1 from(
select *  From HF_VISITS  -- how many visits done to get 1191 customers through visits (link with 2.1)
where Account_ID in (
select distinct Account_ID From HF_ENQ
where Account_ID in (select distinct Account_ID from HF_VISITS where new='Dealer')
and  convert(date,VisitDate)<=convert(date,Createdon)and new='Dealer') and new='Dealer')a

insert into Productive_Visits_through_visits -- Eng
select *,row_number()over(partition by account_id order by account_id) as distcust1 from(
select *  From HF_VISITS  -- how many visits done to get 1191 customers through visits (link with 2.1)
where Account_ID in (
select distinct Account_ID From HF_ENQ
where Account_ID in (select distinct Account_ID from HF_VISITS where new='Eng')
and  convert(date,VisitDate)<=convert(date,Createdon)and new='Eng') and new='Eng')a

update Productive_Visits_through_visits set distcust1=0 where distcust1>1

insert into  Productive_Visits_Direct_Enquiries -- dealer

select *,row_number()over(partition by account_id order by account_id) as distcust1 from(
select *  From HF_VISITS  -- how many visits done to get 2922 customers through direct (link with 2.2)
where Account_ID in (
select distinct Account_ID from (
select distinct Account_ID From HF_ENQ
where Account_ID in (select distinct Account_ID from HF_VISITS where new='Dealer')
 and  convert(date,VisitDate)>convert(date,Createdon) and new='Dealer'

	union all
	
select distinct Account_ID From HF_ENQ
where Account_ID not in (select distinct Account_ID from HF_VISITS where new='Dealer')and new='Dealer')a) and new='Dealer'
)a

insert into  Productive_Visits_Direct_Enquiries -- Eng
select *,row_number()over(partition by account_id order by account_id) as distcust1 from(
select *  From HF_VISITS  -- how many visits done to get 2922 customers through direct (link with 2.2)
where Account_ID in (
select distinct Account_ID from (
select distinct Account_ID From HF_ENQ
where Account_ID in (select distinct Account_ID from HF_VISITS where new='Eng')
 and  convert(date,VisitDate)>convert(date,Createdon) and new='Eng'

	union all
	
select distinct Account_ID From HF_ENQ
where Account_ID not in (select distinct Account_ID from HF_VISITS where new='Eng')and new='Eng')a) and new='Eng')a


update Productive_Visits_Direct_Enquiries set distcust1=0 where distcust1>1
