--select * from ELGi_Sales_Flash_Data_Consolidated order by OrganOrdr,user_names_ordr

CREATE proc [dbo].[sp_ELGi_Sales_Flash_Data_Consolidated] as


begin

truncate table ELGi_Sales_Flash_Data_Consolidated

insert 
into ELGi_Sales_Flash_Data_Consolidated
select OrganOrdr,organ1,organ,user_names_ordr,user_names1,user_names,Division,dpstcode,dpstname,
month_budget,month_actuals,last_month_actuals,ytd_budget,ytd_actuals,ly_ytd_actuals,ly_month_actuals,
FinYear,FinMonthnum,AreaCode,null as productgroup
from (
select *, case 
when organ1='India' then 1
when organ1='SAAME' then 2
when organ1='COSEA' then 3
when organ1='America' then 4
when organ1='EUROPE' then 5
when organ1='ID' then 6
when organ1='Subsidiary' then 7
else 100
end as OrganOrdr

from (

select *,
case 
when ltrim(rtrim(user_names1))='Compressor-Channel' then 1
when ltrim(rtrim(user_names1))='Compressor-Direct' then 2
when ltrim(rtrim(user_names1))='Compressor-C&M' then 3
when ltrim(rtrim(user_names1))='Compressor-WW' then 4
when ltrim(rtrim(user_names1))='Compressor-RLY' then 5
when ltrim(rtrim(user_names1))='After Sales' then 6

when ltrim(rtrim(user_names1))='OSEA' then 1
when ltrim(rtrim(user_names1))='AUSTRALIA' then 2
when ltrim(rtrim(user_names1))='INDONESIA' then 3
when ltrim(rtrim(user_names1))='CHINA' then 4
when ltrim(rtrim(user_names1))='MED & Airends' then 5
when ltrim(rtrim(user_names1))='EZL TO CUST' then 6

when ltrim(rtrim(user_names1))='SA' then 1
when ltrim(rtrim(user_names1))='Africa' then 2
when ltrim(rtrim(user_names1))='ELGi Gulf' then 3
when ltrim(rtrim(user_names1))='ME-Billing from India' then 4
end as user_names_ordr
from (
select *,case 
when organ in ('BELAIR','ROTAIR' ) then 'EUROPE'
when organ in ('PATTONS','USA','BRAZIL') then 'AMERICA'
else organ
end as organ1
from (
select  *, case
when user_names in ('OSEA AS','OSEA Units') then 'OSEA'
when user_names in ('SA-Units','SA-Spares') then 'SA'
when user_names in ('Africa-Units','Africa-Spares') then 'Africa'
when user_names in ('GULF TO CUST - UNITS','GULF TO CUST - SPARES') then 'ELGi Gulf'
when user_names in ('ME-Units','ME-Spares') then 'ME-Billing from India'
--when ltrim(rtrim(user_names)) in ('Africa','America','CIS','Europe','Middle East & West Asia','South East Asia','Neighbouring                  ') then null
when organ='ID' then null

else user_names 
end as user_names1
from (
select organ,case
when division in ('AUSTRALIA TO CUST') then 'AUSTRALIA'
when division in ('INDONESIA TO CUST') then 'INDONESIA'
when division in ('EZL TO CUST') THEN 'CHINA'
else user_names
end as user_names,Division,dpstcode,dpstname,
month_budget,month_actuals,last_month_actuals,ly_month_actuals,
ytd_budget,ytd_actuals,ly_ytd_actuals,FinYear,FinMonthnum,AreaCode
from

(select * from ELGi_Sales_Flash_data)a)b)c)d)e)f




update ELGi_Sales_Flash_Data_Consolidated set user_names1='BELAIR' where organ='BELAIR'
update ELGi_Sales_Flash_Data_Consolidated set user_names1='ROTAIR' where organ='ROTAIR'
update ELGi_Sales_Flash_Data_Consolidated set user_names1='USA' where organ='USA'
update ELGi_Sales_Flash_Data_Consolidated set user_names1='BRAZIL' where organ='BRAZIL'
update ELGi_Sales_Flash_Data_Consolidated set user_names1='PATTONS' where organ='PATTONS'
update ELGi_Sales_Flash_Data_Consolidated set user_names1='AfterSales' where ltrim(rtrim(user_names1))='AS'


--Alter table ELGi_Sales_Flash_Data_Consolidated add productgrp varchar(25)

update ELGi_Sales_Flash_Data_Consolidated set productgrp=a.productgrp
from (select distinct dpst,Area,productgrp from Sales_Budget)a
where ELGi_Sales_Flash_Data_Consolidated.dpstcode=a.Dpst
and ELGi_Sales_Flash_Data_Consolidated.AreaCode=a.Area

update ELGi_Sales_Flash_Data_Consolidated set productgrp='SEPSAC'
where dpstcode='90009'
and productgrp is null

end
