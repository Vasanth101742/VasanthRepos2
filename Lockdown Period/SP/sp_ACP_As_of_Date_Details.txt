----------ACP_Consolidated

Truncate table ACP_Consolidated

Insert into ACP_Consolidated(Company,TranType,InvNo,SuppCode,SuppName,InvAmth,
DueDate,DueAmth,LastPaidDate,PaidAmth,DueBalh,PaymentDays)
select Company,TranType,InvNo,SuppCode,SuppName,InvAmth,
DueDate,DueAmth,LastPaidDate,PaidAmth,DueBalh,PaymentDays
from (
select a.Company,a.TranType,a.InvNo,SuppCode,SuppName,sum(a.InvAmth) InvAmth,a.DueDate,
sum(a.DueAmth) DueAmth,
b.LastPaidDate,
sum(b.PaidAmth) PaidAmth,sum(isnull(a.DueAmth,0)+isnull(b.PaidAmth,0)) as DueBalh,
datediff(dd,DueDate,isnull(b.LastPaidDate,convert(date,getdate()-1))) as PaymentDays
from
(select Company,TranType,InvNo,SuppCode,SuppName,
InvDate,sum(InvAmth) InvAmth,
case when DueDate='9999-01-01' then InvDate else DueDate end as DueDate
,sum(DueAmti) DueAmti,
sum(DueAmth) DueAmth
from ACP_As_of_Date_Details
where InvAmth is not null
group by Company,TranType,InvNo,SuppCode,SuppName,InvDate,DueDate
)a
left outer join
(
select Company,TranType,InvNo,
max(PaidDate) LastPaidDate,
sum(PaidAmth) PaidAmth
from ACP_As_of_Date_Details
where PaidAmth is not null
and tpay=2
group by Company,TranType,InvNo,InvDate,DueDate
)b
on a.Company=b.Company
and a.TranType=b.TranType
and a.InvNo=b.InvNo

group by a.Company,a.TranType,a.InvNo,a.SuppCode,a.SuppName,a.InvDate,a.DueDate,b.LastPaidDate
)a
where 
--DueBalh not between -1 and 1 and 
company like '4%'
order by Company,TranType,InvNo



update ACP_Consolidated set PaymentCategory='Early Payment > 30 days' where paymentdays <-30 and duebalh between -1 and 1
