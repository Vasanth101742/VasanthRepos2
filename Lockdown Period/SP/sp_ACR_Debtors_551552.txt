--Data Reconciliation
update acr_debtors_details_551552 set Rate_USD =a.Rate_USD 
from (
select company,InvDate,Rate_USD,Rnk1 from (
select a.company,a.InvDate,b.Rate_USD,row_number() over(partition by a.company,a.Invdate order by datediff(dd,b.Invdate,a.InvDate) desc ) as Rnk1 from
(select distinct Company,Invdate,Rate_USD from acr_debtors_details_551552
where Rate_USD is null
and company in (500,551,552,553)
)a
left outer join
(select distinct Company,Invdate,Rate_USD from acr_debtors_details_551552 
where Rate_USD is not null
and company in (500,551,552,553)
)b
on a.company=b.company
and a.Invdate<=b.Invdate
)a
where Rnk1=1
)a
where acr_debtors_details_551552.company=a.company
and acr_debtors_details_551552.InvDate=a.InvDate
and acr_debtors_details_551552.Rate_USD is null
