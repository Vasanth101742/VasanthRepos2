when b.OSTA<=3 and datediff(dd,a.TransactionDate,b.date1)>0 and b.date1>convert(date,getdate()-1) then 'Not_a_Backlog' 
when b.OSTA<=3 and datediff(dd,a.TransactionDate,b.date1)=0 then 'Valid_Backlog' 
end as BacklogStatus
from 
(SELECT distinct Description,Prodorderno,TransactionDate from  production_summary3 where remarks='Opening_Plan')a
left outer join 
(select Description,Prodorderno,OSTA,max(date1) date1 from elgi_production_details
group by Description,Prodorderno,OSTA
)b
on a.Description=b.Description
and a.Prodorderno=b.Prodorderno
--where a.Description='DPSAC'
)a
where BacklogStatus='Not_a_Backlog'
