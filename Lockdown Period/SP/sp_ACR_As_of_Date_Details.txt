insert into ACR_USD_Conversion(InvDate,Currency,BCUR,CCUR,STDT,RATE,DaysDiff,Rnk)
select InvDate,Currency,BCUR,CCUR,STDT,RATE,DaysDiff,Rnk from (
select *,datediff(dd,STDT,InvDate) as DaysDiff,
row_number() over (partition by InvDate,currency order by datediff(dd,STDT,InvDate)) as Rnk 
from (
select distinct t_docd as InvDate,t_ccur as Currency from TTFACR200ALL 
union
select distinct convert(date,docd) as InvDate,ccur as Currency from TTFACR200551552 
--where currency='MYR'
)a
left outer join 
(select BCUR,CCUR,convert(date,STDT) STDT,max(RATE) RATE from TTCMCS008ALL 
where RTYP='SLS' 
and ccur='USD'
group by BCUR,CCUR,convert(date,STDT))b
on a.currency=b.bcur
and a.InvDate>=b.STDT
)a
where Rnk=1
and RATE<>1
order by InvDate
