--Method 1:
update Triggers_Inventory_Production set MonthType='Backlog' where date1 < dateadd(dd,1,eomonth(getdate()-1,-1))
update Triggers_Inventory_Production set MonthType='CurrentMonth' where date1 between dateadd(dd,1,eomonth(getdate()-1,-1)) and eomonth(getdate()-1)

--Method 2: For Production Orders
