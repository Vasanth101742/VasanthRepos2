when A.SOPS = 2 then  'MULTI BIN'
when A.SOPS = 3 then  'MILK RUN'

update Lean_Allocation_Tmp set BINQTY=SQTY,SLOQ=a.SLOQ
from(
select distinct COMPANY,DIVI,DEPT,ITEM,SQTY,SLOQ
from TZZPUR511ALL a
where COMPANY like '4%'
and SQTY =(
select max(SQTY) SQTY from TZZPUR511ALL b
where a.COMPANY=b.COMPANY
and a.DIVI=b.DIVI
and a.DEPT=b.DEPT
and a.ITEM=b.ITEM
--and item='         410431001'
group by COMPANY,DIVI,DEPT,ITEM)
)a

update Lean_Allocation_Tmp set ProdAllocation=a.AllocatedQty
from(
select COMPANY,T$ITEM as ITEM,null as Category,'Production Allocation' as AllocationType,
sum(T$QANA) as AllocatedQty--,sum(T$PRIC) as PRIC, sum(T$PRIC*T$QANA) as Amnt

)a

from(select COMPANY,T$ITEM as ITEM,null as Category,'Sales Allocation' as AllocationType,

update Lean_Allocation_Tmp set BinAllocation=SLOQ*BINQTY where category in ('MULTI BIN') and AllocationType is null


update Lean_Allocation_Tmp set RegAllocation=BINQTY 
where category in ('SINGLE BIN') and AllocationType='ALLOCATION'
update Lean_Allocation_Tmp set RegAllocation=2*BINQTY+ProdAllocation+SalesAllocation 
where category in ('MR-MOTORS') and AllocationType='ALLOCATION'
update Lean_Allocation_Tmp set RegAllocation=ProdAllocation+SalesAllocation 
where category in ('MILK RUN') and AllocationType='ALLOCATION'
update Lean_Allocation_Tmp set RegAllocation=BINQTY
where category in ('TWO BIN') and AllocationType='ALLOCATION' and BINQTY>(ProdAllocation+SalesAllocation)
update Lean_Allocation_Tmp set RegAllocation=ceiling((ProdAllocation+SalesAllocation)/BINQTY)*BINQTY
where category in ('TWO BIN') and AllocationType='ALLOCATION' and BINQTY<(ProdAllocation+SalesAllocation)
update Lean_Allocation_Tmp set RegAllocation=ceiling((sloq*BINQTY+ProdAllocation+SalesAllocation)/BINQTY)*BINQTY
where category in ('MULTI BIN') and AllocationType='ALLOCATION' 


update Lean_Allocation_Tmp set StockAllocation=BINQTY where category in ('SINGLE BIN') and AllocationType='STOCK'
update Lean_Allocation_Tmp set StockAllocation=2*BINQTY+ProdAllocation+SalesAllocation where category in ('MR-MOTORS') and AllocationType='STOCK'
update Lean_Allocation_Tmp set StockAllocation=BINQTY+ProdAllocation+SalesAllocation where category in ('MILK RUN') and AllocationType='STOCK'
update Lean_Allocation_Tmp set StockAllocation=ceiling((BINQTY+ProdAllocation+SalesAllocation)/BINQTY)*BINQTY
where category in ('TWO BIN') and AllocationType='STOCK'
update Lean_Allocation_Tmp set StockAllocation=SLOQ*BINQTY where category in ('MULTI BIN') and AllocationType='STOCK'

select COMPANY,ITEM,case when Rnk=1 then StockQty else null end as StockQty,LeanType,CATEGORY,AllocationType,SUPPCODE,BUYRCODE,sloq,
BINQTY,ProdAllocation,SalesAllocation,BinAllocation,RegAllocation,StockAllocation
from (
select * ,row_number() over(partition by COMPANY,ITEM,BUYRCODE order by COMPANY,ITEM,BUYRCODE) as Rnk
from (
select isnull(a.COMPANY,b.COMPANY) COMPANY,
isnull(a.ITEM,b.ITEM) ITEM,b.StockQty,LeanType,CATEGORY,AllocationType,SUPPCODE,BUYRCODE,sloq,
BINQTY,ProdAllocation,SalesAllocation,BinAllocation,RegAllocation,StockAllocation
from 
(
select company,ITEM,LeanType,CATEGORY,AllocationType,SUPPCODE,BUYRCODE,
sum(sloq) sloq,sum(BINQTY) BINQTY,
sum(ProdAllocation) ProdAllocation,sum(SalesAllocation) SalesAllocation,
sum(BinAllocation) BinAllocation,sum(RegAllocation) RegAllocation,sum(StockAllocation) StockAllocation
from Lean_Allocation_Tmp
group by company,ITEM,LeanType,CATEGORY,AllocationType,SUPPCODE,BUYRCODE
)a
full outer join 
(select COMPANY,ITEM,StockQty from Excess_Inventory2_Temp)b
on a.COMPANY=b.COMPANY
and ltrim(rtrim(a.ITEM))=ltrim(rtrim(b.ITEM))
)a1
--where ltrim(rtrim(ITEM))='000906217'
)a2
