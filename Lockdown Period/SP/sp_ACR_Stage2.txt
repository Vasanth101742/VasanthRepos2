and ACR_Tmp1.currate <> 1.00)

where (diffdays >= 91 and diffdays <= 180) 
and @gdate  > convert(datetime,convert(varchar,ACR_Tmp1.duedt,101)) 

update ACR_Tmp1 set DAmtHCGT180 = ABalHome  
where diffdays >= 181 and @gdate > convert(datetime,convert(varchar,ACR_Tmp1.duedt,101)) 

update ACR_Tmp1 set undueamt = aBalHome 
where @gdate  < convert(datetime,convert(varchar,ACR_Tmp1.duedt,101))
