from InventoryDetails

case when BOUT=1 then 'Yes'

from TWHLTC500ALL
where OSER=2 and QHND<>0
)b
on a.COMPANY=b.company
and a.cwar=b.cwar
and ltrim(rtrim(a.ITEM))=ltrim(rtrim(b.ITEM))
and a.TRDT=b.IDAT
left outer join (select distinct COMPANY,ORNO,PONO,PDNO from TZZSFC289ALL
where concat(company,PDNO) in (select distinct concat(COMPANY,PDNO) from TTISFC001ALL)
)c
on b.company=c.company

left outer join (select distinct Company,WareHouse,ItemCode,
OrderDate,OrderNum,PositionNum,CustCode,CustName from Sales_Inflow where invnum=0)d
on c.COMPANY=d.Company
and ltrim(rtrim(c.ORNO))=ltrim(rtrim(d.OrderNum))
and ltrim(rtrim(c.PONO))=ltrim(rtrim(d.PositionNum))



update FG_Inventory_Details set Description=b.Description
from (
select distinct a.company,Description,prd_from,prd_to from ELGi_Production_Master a
left outer join ELGi_AssemblyLine_Master b
on b.code = a.assemblycode
)b
where case when FG_Inventory_Details.PDNO like '%[a-z]%' then '0' else FG_Inventory_Details.PDNO end 
between case when b.prd_from like '%[a-z]%' then '0' else b.prd_from end 
and case when b.prd_to like '%[a-z]%' then '0' else b.prd_to end  
and FG_Inventory_Details.company = b.company




update FG_Inventory_Details set QA_BLK=g.QA_BLK
from (select distinct COMPANY,CWAR,ITEM,SERL,
case when SERL is not null then 'Yes'
when SERL is null then 'No' end as QA_BLK
from TWHINH211ALL where QTBU>0
and ltrim(rtrim(SERL)) in (select distinct ltrim(rtrim(MSER)) from FG_Inventory_Details)
)g
where FG_Inventory_Details.COMPANY=g.company
and ltrim(rtrim(FG_Inventory_Details.cwar))=ltrim(rtrim(g.cwar))
and ltrim(rtrim(FG_Inventory_Details.ITEM))=ltrim(rtrim(g.ITEM))
and ltrim(rtrim(FG_Inventory_Details.MSER))=ltrim(rtrim(g.SERL))


update FG_Inventory_Details set QSKT=null,Amnt=null,Amnt_Mn=null where Rnk1>1

Update FG_Inventory_Details Set LastRunDate=getdate()-1
