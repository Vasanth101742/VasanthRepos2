Insert into inflow_billing_Temp1
select * from(
select distinct organ,Vertical1,dpst,AreaCode,FinPeriod,
dense_rank() over (order by organ) as OrganRnk,
dense_rank() over (order by Vertical1) as Vertical1Rnk,
dense_rank() over (partition by organ,vertical1 order by dpst) as dpstRnk,
dense_rank() over (partition by organ,vertical1,dpst order by AreaCode) as AreaRnk,
dense_rank() over (order by FinPeriod) as FinPeriodRnk
from Inflow_billing
where  dpst is not null
and AreaCode is not null and FinPeriod is not null

)a
order by OrganRnk,Vertical1Rnk,dpstRnk,AreaRnk,FinPeriodRnk


set @OrganCnt =(select max(OrganRnk) from inflow_billing_Temp1)
set @Count1=1
while (@Count1<=@OrganCnt)
begin

set @Organ=(select distinct organ from inflow_billing_Temp1 where OrganRnk=@Count1)
set @Vertical1Cnt=(select max(Vertical1Rnk) from inflow_billing_Temp1)
set @Count2=1
while (@Count2<=@Vertical1Cnt)
begin

set @Vertical1=(select distinct Vertical1 from inflow_billing_Temp1 
where OrganRnk=@Count1 and Vertical1Rnk=@Count2)
set @dpstCnt=(select max(dpstRnk) from inflow_billing_Temp1)
set @Count3=1
while (@Count3<=@dpstCnt)
begin


set @dpst=(select distinct dpst from inflow_billing_Temp1 
where OrganRnk=@Count1 and Vertical1Rnk=@Count2 and dpstRnk=@Count3)
set @AreaCnt=(select max(AreaRnk) from inflow_billing_Temp1)
set @Count4=1
while (@Count4<=@AreaCnt)
begin



set @AreaCode=(select distinct AreaCode from inflow_billing_Temp1 
where OrganRnk=@Count1 and Vertical1Rnk=@Count2 and dpstRnk=@Count3 and 
AreaRnk=@Count4)
set @FinPeriodCnt=(select max(FinPeriodRnk) from inflow_billing_Temp1)
set @Count5=1
while (@Count5<=@FinPeriodCnt)
--while (@Count5<=10)
begin


set @FinPeriod=(select distinct FinPeriod from inflow_billing_Temp1 
where --OrganRnk=@Count1 and Vertical1Rnk=@Count2 and dpstRnk=@Count3 and AreaRnk=@Count4 and 
FinPeriodRnk=@Count5)

set @PrevFinPeriod=(select distinct FinPeriod from inflow_billing_Temp1 
where --OrganRnk=@Count1 and Vertical1Rnk=@Count2 and dpstRnk=@Count3 and AreaRnk=@Count4 and 
FinPeriodRnk=@Count5-1)

@Organ,@Vertical1,@dpst,@AreaCode,@FinPeriod

select a.Organ,a.Vertical1,a.dpst,a.Area,a.FinPeriod,1 as Rnk from 
(select Organ,Vertical1,dpst,Area,FinPeriod from Inflow_billing_Temp2 
where dpst is not null and Area is not null
and FinPeriod<>''

and a.Area=b.AreaCode

update Inflow_billing set PrevinflowOrders=a.infloworders,PrevOrderQty=a.OrderQty,PrevInflow=a.Inflow,
PrevSalesOrders=a.SalesOrders,PrevDeliveryQty=a.deliveryQty,PrevSales=a.sales,

select Organ,Vertical1,dpst,AreaCode,--FinPeriod
sum(isnull(infloworders,0)) infloworders,sum(isnull(OrderQty,0)) OrderQty,sum(isnull(Inflow,0)) Inflow,
