CREATE proc [dbo].[SP_GR_GSC_SupplyPlan]
as
begin

truncate table [GR_GSC_SUPPLYPLAN]


update tzzsls158490 set t$item=replace(t$item,' ','')


insert into [GR_GSC_SUPPLYPLAN]

select plan_recp_date,blocked,disc_origin,plan_del_date,Sales_Inv_Address
,Cust_Inv_Address,Sales_GSTN_NO,Cust_GSTN_NO,SALES_SEQ,Cust_SEQ,
cust_state,Sales_state,state,Old_orderno,Old_positionno,BP_SEQ_NO,Tax_Desc,HSN_CODE,GSTN_NO,PRICE_ORIGIN,
Ref,dpst,dpstname,PositionNum,OrderNum,ordertype,origin,Baan_orderno,Baan_positionno,
DeliveryDate,ItemCode,Item_desc,TaxCode,Price,DiscPerc1,DiscPerc2,buyer,Indent,OrderDate,CustCode,Custname,[value],ord_qty,
delqty,penqty,STOCK_490_act,STOCK_490_onhand,Next_Due_Date1,Next_Due_Date2,Reason1,Reason2,
Supplier,t_oamt,PlanRecDate,Lead_days,PlannedDelDate,ItemType,ShipQty,Allocation,ShipmentNO,QtyBlock,Wh_orderDate,
Ship_Status,SlNo,STock_Block,Seq,ExpecDate,CurrentDate,
OverdueStatus,[Over Due Days],FinalPrice,Amount,Order_status,[OD Com],Division,OrderLine

,case when  Order_status='Overdue' and Ship_Status='Waiting For Invoice' then getdate() + 3
when  Order_status='Overdue' and Ship_Status='Stock Available' then getdate()+8
when  Order_status='Overdue' and Ship_Status in ('In-Packing / Billing','In Packing','Kits-WIP') then getdate()+6
when  Order_status='Overdue' and Ship_Status in ('Fully Advised') then getdate()+6

else ''
end 'ExpecDay','','','',Cust_Req_Date

from (
select *,case when Division='Subsidiaries' then 4
when  Division='Industrial' then 1
when  Division='OFSAC' then 2
when  Division='Portable' then 3
when  Division='Direct Export' then 5
end 'OrderLine'  from (

select *,round((case when  sales_gstn_no='EXPORT' then value else t_oamt end),2)'FinalPrice'
,round((case when  sales_gstn_no='EXPORT' then value/1000000 else t_oamt/1000000 end),2)'Amount',

case when [Over Due Days]>0 and isnull(wh_orderdate,'1900-01-01')>'2000-01-01' then 'Overdue'
when  [Over Due Days]<=0   then 'Within Due'
when  [Over Due Days]>0 and isnull(wh_orderdate,'1900-01-01')<='2000-01-01'  then 'Not Released'
end'Order_status',case 
when [Over Due Days] between 1 and 10 then '0-10Days' 
when [Over Due Days] between 11 and 20 then '11-20Days'
when [Over Due Days] between 21 and 30 then '21-30Days'
when [Over Due Days] between 31 and 60 then '31-60Days'
when [Over Due Days] between 61 and 90 then '61-90Days'
when [Over Due Days] between 91 and 150 then '91-150Days'
when [Over Due Days] >150 then '150 GreatDays'
when [Over Due Days]<=0 and abs([Over Due Days]) between 0 and 7  then 'Up Week 1'
when [Over Due Days]<0 and abs([Over Due Days]) between 8 and 14  then 'Up Week 2' 
when [Over Due Days]<0 and abs([Over Due Days]) between 15 and 21  then 'Up Week 3' 
when [Over Due Days]<0 and abs([Over Due Days]) between 22 and 28  then 'Up Week 4' 
when [Over Due Days]<0 and abs([Over Due Days]) between 29 and 35  then 'Up Week 5' 
when [Over Due Days]<0 and abs([Over Due Days]) between 36 and 42  then 'Up Week 6' 
when [Over Due Days]<0 and abs([Over Due Days]) between 43 and 49  then 'Up Week 7' 
when [Over Due Days]<0 and abs([Over Due Days]) between 49 and 56  then 'Up Week 8' 
when [Over Due Days]<0 and abs([Over Due Days]) >=57  then 'Great than 8' 
end 'OD Com',

case when g.Dpst in (70016,70028,70045,70056,70065,70085,70096) then 'Subsidiaries'  
when g.Dpst in (90001, 90013,90028,90031,90063,90075,90008,90010,90016,90020,90021,90022,90030,90033,90035,90037,90038,90094) then 'Industrial'
when g.Dpst in (90082 , 90083 , 90084) then 'OFSAC'
when g.Dpst in (90076 , 90077) then 'Portable'
when g.Dpst in(90213,90231,90256,90257,90267) then 'Direct Export'
else 'Others'
end 'Division'

  from 

(

select *,((DATEDIFF(dd, c.PlannedDelDate,c.CurrentDate) + 1)
    -(DATEDIFF(wk,  c.PlannedDelDate,c.CurrentDate) * 2)
    -(CASE WHEN DATEPART(dw, c.OrderDate) = 1 THEN 1 ELSE 0 END)
    -(CASE WHEN DATEPART(dw, c.CurrentDate) = 7 THEN 1 ELSE 0 END))-1 as 'Over Due Days'
 from 

(
select a.*,convert(date,getdate()) as 'CurrentDate',case when a.STOCK_490_onhand=0 then 'No' else 'Yes' end 'OverdueStatus'
from gr_PendingOrders  a

where 
a.dpst in (70016,70028,70045,70056,70065,70085,70096,90001,90013,90028,90031,90063,
				90075,90076,90077,90082,90083,90084,90213,90231,90256,90257,90267
				,90008,90010,90016,90020,90021,90022,90030,90033,90035,90037,90038,90094) 
				and 
convert(date,a.OrderDate)>'20220331'-- and  isnull(wh_orderdate,'1900-01-01')>'1900-01-01'

and
 (replace(a.OrderNum,' ','') +(replace(a.ItemCode,' ',''))) not in ('292051258013436408',
'292050433M008906415',
'292051258012374519',
'430000013000998065',
'430000013B676103',
'292052298B009400010129',
'292052298X0180912',
'292055109A990921',
'292054810B015400780001',
'292054759Z012550',
'292054764Z012555',
'292055039972300030',
'292054758Z012555',
'292055186972300030',
'292057118000960087',
'292057118090445790',
'292058877B006700770023',
'292060311411400301',
'243003484A410614',
'251011905B011600960153',
'292061457B011106740001',
'292064082022459499',
'292063744022459499',
'243003491412400907',
'34300958600021002B',
'34300946700021002B',
'242024582B004700770022',
'292063765B005706740002',
'292063532B014606410052',
'343009627105411BSC',
'292065524015401697',
'251012603A990288',
'242026082012485146',
'292065030012473189',
'292065524972300060',
'242026072012485146',
'242025793012485146',
'292064430B014606410052')
)c

) g


 ) a
 
 
 )a
 

 --where Order_status='Overdue' 
 
order by a.[Over Due Days] desc



update a set Comment=isnull(b.T$cmnt,'Data Not Available') from GR_GSC_SUPPLYPLAN a
left join tzzsls158490 b on b.t$item=a.itemcode
where  ship_status  in ('Partially Available','Partialy Advised','Stock Not Available','Stock Not Available - HO Manf Parts'
						,'Stock Not Available - ACP Manf Parts') 
and  isnull(T$CDAT,'1970-01-01 00:00:00.000')='1970-01-01 00:00:00.000' and Order_status='Overdue' 


update a set ExpecDay=(ComDATE+9) from  GR_GSC_SUPPLYPLAN a 
left join (select t$item,max(T$CDAT)'ComDATE' from tzzsls158490 group by t$item) b on b.t$item=a.itemcode
where  ship_status  in ('Partially Available','Partialy Advised','Stock Not Available','Stock Not Available - HO Manf Parts','Stock Not Available - ACP Manf Parts'
) and a.Comment='' and Order_status='Overdue' 


update GR_GSC_SUPPLYPLAN set ExpecDay= PlannedDelDate,Comment=Order_status  where Order_status <>'Overdue'

update a set areacode=b.areacode,  areaname=b.areaname from GR_GSC_SupplyPlan a
inner join Pending_Orders b on b.OrderNum=a.OrderNum and b.PositionNum=a.PositionNum and b.CustCode=a.CustCode

update a set Baan_orderno=isnull(b.Zone,'Others') from [GR_GSC_SUPPLYPLAN] a
left join vw_AREA_MASTER b  on b.Areacode=a.areacode




end
