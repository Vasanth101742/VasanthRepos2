OPON,LOBL,DTRL,LORL,BPID,BPRT,RLTY,RLAM,ERDT,PSTC,PHSN,REFCNTD,REFCNTU) 
select * from TTDSLS421ALL_ver2


update Credit_Block_Details set Area=a.Area,dpst=a.Dpst
from (select distinct Company,OrderNum,DPST,Area from Sales_Header where company in (401,490,600))a
where Credit_Block_Details.company=a.company
and Credit_Block_Details.ORNO=a.OrderNum

update Credit_Block_Details set AreaName=a.AreaName
from (select * from vw_AREA_MASTER)a
where Credit_Block_Details.area=a.Areacode

update Credit_Block_Details set DPSTName=a.DPSTName
from (select * from vw_DPST_MASTER)a
where Credit_Block_Details.dpst=a.DPSTcode

update Credit_Block_Details set CustName=a.custname1
from (select * from vw_Business_Partner)a
where Credit_Block_Details.BPID=a.custcode

update Credit_Block_Details set creditLimit=a.CRLR
from (select company,ITBP,CRLR from TTCCOM112ALL)a
where ltrim(rtrim(Credit_Block_Details.BPID))=ltrim(rtrim(a.ITBP))


end
