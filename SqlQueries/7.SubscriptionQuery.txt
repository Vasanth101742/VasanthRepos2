select * from vw_subscriptionDetails

CREATE VIEW vw_subscriptionDetails as
SELECT *,row_number()over(order by reportname) as rank from (  
Select reportname,[to],cc,[subject],laststatus,lastruntime,cast([next run date] as datetime) as [next run date],[next run time] ,mail_status  
from (  
SELECT  
c.Name AS ReportName,  
'Next Run Date' = CASE next_run_date WHEN 0 THEN null  
ELSE  substring(convert(varchar(15),next_run_date),1,4) + '/' +  
substring(convert(varchar(15),next_run_date),5,2) + '/' +  
substring(convert(varchar(15),next_run_date),7,2)  END,  
 
'Next Run Time' = isnull(CASE len(next_run_time)  
WHEN 3 THEN cast('00:0'  + Left(right(next_run_time,3),1) +':' + right(next_run_time,2) as char (8))  
WHEN 4 THEN cast('00:' + Left(right(next_run_time,4),2)  +':' + right(next_run_time,2) as char (8))  
WHEN 5 THEN cast('0' + Left(right(next_run_time,5),1) +':' + Left(right(next_run_time,4),2)  +':' + right(next_run_time,2) as char (8))  
WHEN 6 THEN cast(Left(right(next_run_time,6),2) +':' + Left(right(next_run_time,4),2) +':' + right(next_run_time,2) as char (8))  
END,'NA'),  
Convert(XML,[ExtensionSettings]).value('(//ParameterValue/Value[../Name="TO"])[1]','varchar(4000)') as [To] ,
Convert(XML,[ExtensionSettings]).value('(//ParameterValue/Value[../Name="CC"])[1]','varchar(4000)') as [CC] ,
Convert(XML,[ExtensionSettings]).value('(//ParameterValue/Value[../Name="RenderFormat"])[1]','nvarchar(4000)') as [Render Format] ,
Convert(XML,[ExtensionSettings]).value('(//ParameterValue/Value[../Name="Subject"])[1]','nvarchar(4000)') as [Subject] ,  
  
---Example report parameters: StartDateMacro, EndDateMacro &amp; Currency.  
  
Convert(XML,[Parameters]).value('(//ParameterValue/Value[../Name="StartDateMacro"])[1]','nvarchar(4000)') as [Start Date] ,
Convert(XML,[Parameters]).value('(//ParameterValue/Value[../Name="EndDateMacro"])[1]','nvarchar(4000)') as [End Date] ,
Convert(XML,[Parameters]).value('(//ParameterValue/Value[../Name="Currency"])[1]','nvarchar(4000)') as [Currency] ,
[LastStatus] ,[EventType] ,[LastRunTime] ,[DeliveryExtension] ,[Version] ,
case when laststatus like '%Mail sent%' or laststatus like '%The File%' then 'success'  
when  laststatus not like '%Mail sent%' and  laststatus  not like '%The File%' then 'Failed' end  as mail_status
  
FROM   
  
ReportingService_916a074721bc43a0869d72c211bd6a18.dbo.[Catalog] c  
INNER JOIN ReportingService_916a074721bc43a0869d72c211bd6a18.dbo.[Subscriptions] S 
ON c.ItemID = S.Report_OID  
INNER JOIN ReportingService_916a074721bc43a0869d72c211bd6a18.dbo.ReportSchedule R 
ON S.SubscriptionID = R.SubscriptionID  
INNER JOIN msdb.dbo.sysjobs J 
ON Convert(nvarchar(128),R.ScheduleID) = J.name  
INNER JOIN msdb.dbo.sysjobschedules JS 
ON J.job_id = JS.job_id
)a  
  
group by  reportname,[to],cc,[subject],laststatus,lastruntime,[next run date],[next run time],mail_status)a  
WHERE  [to] is not null
