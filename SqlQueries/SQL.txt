select ASSEMBLY,PLINE,METRIC,MONTH,YEAR,FDATE,TDATE,
case 
when column_name1='india_target' or column_name2='india_cl' then 'India and India Like'
when column_name1='subs_target' or column_name2='subs_cl' then 'Subsidiary'
when column_name1='overall_target' or column_name2='overall_cl' then 'Overall'
end as Market,Targets,CLimits 
From(
SELECT ASSEMBLY,PLINE,METRIC,MONTH,YEAR,FDATE,TDATE,india_target,subs_target,overall_target,
india_cl,subs_cl,overall_cl
FROM alc_metrics
)a
unpivot
(
Targets for column_name1 in (india_target,subs_target,overall_target )
)as k
unpivot
(
CLimits for column_name2 in (india_cl,subs_cl,overall_cl)
) as K1

order by fdate ,tdate,metric,assembly

cumulative calculation



 

select * from(
select distinct P_ID,isnull(P_Team,'Sales') as P_Team,
P_User_ID,P_Name,P_Email,
segment_id,Segment,
ordering,metrics_levels,Act_Count,
case when ordering in (1,2,3) then sum(Act_Count)over(partition by isnull(p_id,0),segment_id order by ordering desc  rows between UNBOUNDED PRECEDING and CURRENT ROW)
when ordering in  (4,5,6) then Act_Count 
end as Req_Counts
from(
select 
P_ID,
P_Team,
P_User_ID,P_Name,P_Email,
Segment_id,isnull(segment,'Others') Segment,
ordering,metrics_levels,sum(Act_Count) Act_Count
from CRM_Parent_Child_ver2
inner join crm_access acc
on acc.pemail=CRM_Parent_Child_ver2.P_Email
where P_Email =acc.pemail
and concat('ELGI\',substring(acc.u_email,1,charindex('@',acc.u_email)-1)) = @email

and segment_id in(@Segment)


--where P_User_ID=226
group by P_ID,P_Team,
P_User_ID,P_Name,P_Email,
Segment_id,Segment,
ordering,metrics_levels
)a
)b
--where ordering is null or ordering <5
order by P_ID,P_Team,
P_User_ID,P_Name,P_Email,
Segment_id,Segment,
ordering,metrics_levels
