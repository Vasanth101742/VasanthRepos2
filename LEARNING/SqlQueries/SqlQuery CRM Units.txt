from appointment
where len(requiredattendees)>2 and len(elgi_customertypesalesname)>2
and elgi_latestvisitdate>='2018-10-01'
)a
full outer join
(select opportunityid,

--statecodename,elgi_probabilityname,estimatedclosedate,
--actualclosedate,createdon,
customerid--,customeridname
from CRM_OPPURTUNITY
where elgi_enquirystatusname<>'On Hold'
and elgi_businessverticalname='Industrial'
and elgi_sourceofenquiryname not in ('Website')
and createdon>='2018-10-01'
)b
on a.requiredattendees=b.customerid
left outer join 
(select accountid
--,name,elgi_cityname,elgi_statename,
--elgi_customerclustersalesname,
--elgi_customerclassificationname,--a.elgi_lineofbusinessname,
--b.elgi_description
from ACCOUNT a
left outer join
ELGI_LINEOFBUSINESS b
on a.elgi_lineofbusinessname=b.elgi_name
) c
on isnull(a.requiredattendees,b.customerid)=c.accountid
left outer join 
(select a.opportunityid,a.quoteid,
--a.elgi_businessverticalname,a.statuscodename, 
b.elgi_salesofficename,--b.priceperunit,b.quantity,b.baseamount,
b.productid,--b.productidname,
--c.elgi_salesofficecode,
d.elgi_itemcode--,e.elgi_productcategoryname
from QUOTE a
left outer join QUOTEDETAILS b
on a.quoteid=b.quoteid
left outer join 
(select * from ELGI_SALESOFFICES where elgi_salesofficecode in ('90008','90009','90016',
'90020','90022','90030','90032','90093','90094') ) c
on ltrim(rtrim(b.elgi_salesofficename))=ltrim(rtrim(c.elgi_name))
left outer join Product d
on b.productid=d.productid
left outer join (
select * from(
select elgi_itemcode,elgi_productcategoryname,
ROW_NUMBER() over (partition by elgi_productcategoryname order by elgi_productcategoryname) as Rnk
from ELGI_DEALERPRICELISTMASTER)a
where Rnk=1
) e
on ltrim(rtrim(d.elgi_itemcode))=ltrim(rtrim(e.elgi_itemcode))
where a.opportunityid in 
(select distinct  opportunityid from CRM_OPPURTUNITY
where elgi_enquirystatusname<>'On Hold'
and elgi_businessverticalname='Industrial'
and elgi_sourceofenquiryname not in ('Website'))
and a.statuscodename not in ('Revised')
)Quote
on b.opportunityid=Quote.opportunityid

--)complete


update CRM_Units_Fact set elgi_customervisitid=a.elgi_customervisitid,
elgi_customertypesalesname=a.elgi_customertypesalesname,
elgi_latestvisitdate=a.elgi_latestvisitdate,
elgi_monthstartingdate=a.elgi_monthstartingdate,
elgi_dealer=a.elgi_dealer,
elgi_dealerengineer=a.elgi_dealerengineer,

elgi_statusofvisitsalesname=a.elgi_statusofvisitsalesname
from (
select substring(requiredattendees,14,36) requiredattendees
,elgi_customervisitid,elgi_customertypesalesname,elgi_latestvisitdate,
elgi_monthstartingdate,elgi_dealer,elgi_dealerengineer,elgi_statusofvisitsales,
elgi_statusofvisitsalesname,owneridname
from appointment

and elgi_latestvisitdate>='2018-10-01'
)a
where CRM_Units_Fact.requiredattendees=a.requiredattendees

update CRM_Units_Fact set opportunityid=b.opportunityid,
Opportunity_Name=b.Opportunity_Name,
elgi_enquiryid=b.elgi_enquiryid,
elgi_businessverticalname=b.elgi_businessverticalname,
elgi_sourceofenquiryname=b.elgi_sourceofenquiryname,
elgi_dealername=b.elgi_dealername,
elgi_dealerengineername=b.elgi_dealerengineername,
owneridname=b.owneridname,
elgi_opportunitystatusname=b.elgi_opportunitystatusname,
elgi_enquirystatusname=b.elgi_enquirystatusname,
statecodename=b.statecodename,
elgi_probabilityname=b.elgi_probabilityname,
estimatedclosedate=b.estimatedclosedate,
actualclosedate=b.actualclosedate,createdon=b.createdon,
customerid=b.customerid,
createdbyname=b.createdbyname
from (select opportunityid,elgi_enquiryid as Opportunity_Name,
elgi_enquiryid,elgi_businessverticalname,
elgi_sourceofenquiryname,elgi_dealername,elgi_dealerengineername,
owneridname,elgi_opportunitystatusname,elgi_enquirystatusname,
statecodename,elgi_probabilityname,estimatedclosedate,
actualclosedate,createdon,createdbyname,
customerid--,customeridname
from CRM_OPPURTUNITY
where elgi_enquirystatusname<>'On Hold'
and elgi_businessverticalname='Industrial'
and elgi_sourceofenquiryname not in ('Website')
and createdon>='2018-10-01'
)b
where CRM_Units_Fact.customerid=b.customerid

update CRM_Units_Fact set 
name=c.name,elgi_cityname=c.elgi_cityname,
elgi_statename=c.elgi_statename,
elgi_customerclustersalesname=c.elgi_customerclustersalesname,
elgi_customerclassificationname=c.elgi_customerclassificationname,
elgi_description=c.elgi_description,
accountnumber=c.accountnumber
from (select accountid,accountnumber,
name,elgi_cityname,elgi_statename,
elgi_customerclustersalesname,
elgi_customerclassificationname,--a.elgi_lineofbusinessname,
b.elgi_description
from ACCOUNT a
left outer join
ELGI_LINEOFBUSINESS b
on a.elgi_lineofbusinessname=b.elgi_name
) c
where isnull(requiredattendees,customerid)=c.accountid


update CRM_Units_Fact set elgi_businessverticalname=Quote.elgi_businessverticalname,
Quote_Name=Quote.Quote_Name,
statuscodename=Quote.statuscodename,
priceperunit=Quote.priceperunit,
quantity=Quote.quantity,
baseamount=Quote.baseamount,
productidname=Quote.productidname,
elgi_salesofficecode=Quote.elgi_salesofficecode,
elgi_productcategoryname=Quote.elgi_productcategoryname
from 
(select a.opportunityid,a.quoteid,a.name as Quote_Name,
a.elgi_businessverticalname,a.statuscodename, 
b.elgi_salesofficename,
b.priceperunit,b.quantity,b.baseamount,
b.productid,
b.productidname,c.elgi_salesofficecode,
d.elgi_itemcode
,e.elgi_productcategoryname
from QUOTE a
left outer join QUOTEDETAILS b
on a.quoteid=b.quoteid
Inner join 
(select * from ELGI_SALESOFFICES where elgi_salesofficecode in ('90008','90009','90016',
'90020','90022','90030','90032','90093','90094') ) c
on ltrim(rtrim(b.elgi_salesofficename))=ltrim(rtrim(c.elgi_name))
left outer join Product d
on b.productid=d.productid
left outer join (
select * from(
select elgi_itemcode,elgi_productcategoryname,
ROW_NUMBER() over (partition by elgi_productcategoryname order by elgi_productcategoryname) as Rnk
from ELGI_DEALERPRICELISTMASTER)a
where Rnk=1
) e
on ltrim(rtrim(d.elgi_itemcode))=ltrim(rtrim(e.elgi_itemcode))
where a.opportunityid in 
(select distinct  opportunityid from CRM_OPPURTUNITY
where elgi_enquirystatusname<>'On Hold'
and elgi_businessverticalname='Industrial'
and elgi_sourceofenquiryname not in ('Website'))
and a.statuscodename not in ('Revised')
)Quote
where CRM_Units_Fact.opportunityid=Quote.opportunityid



update CRM_Units_Fact set 
elgi_dealerofferid=Offers.elgi_dealerofferid,
elgi_dealeroffername=Offers.elgi_dealeroffername,
elgi_businessverticalname=Offers.elgi_businessverticalname,
statuscodename=Offers.elgi_revisedflagname,
priceperunit=Offers.elgi_priceperunit,
quantity=Offers.elgi_quantity,
baseamount=Offers.elgi_amount,
productidname=Offers.elgi_existingproductname,
elgi_salesofficecode=Offers.elgi_salesofficecode,
elgi_productcategoryname=Offers.elgi_productcategoryname
from (select a.elgi_enquiry,elgi_dealerofferid,b.elgi_dealeroffername,
a.elgi_businessverticalname,a.elgi_revisedflagname,--a.statuscodename, 
b.elgi_salesofficename,b.elgi_priceperunit,b.elgi_quantity,b.elgi_amount,b.elgi_existingproduct,b.elgi_existingproductname,
c.elgi_salesofficecode,d.elgi_itemcode,e.elgi_productcategoryname
from ELGI_DEALEROFFER a
left outer join ELGI_DEALEROFFERPRODUCT b
on a.elgi_dealerofferid=b.elgi_dealeroffer
left outer join 
(select * from ELGI_SALESOFFICES where elgi_salesofficecode in ('90008','90009','90016',
'90020','90022','90030','90032','90093','90094') ) c
on ltrim(rtrim(b.elgi_salesofficename))=ltrim(rtrim(c.elgi_name))
left outer join Product d
on b.elgi_existingproduct=d.productid
left outer join (
select * from(
select elgi_itemcode,elgi_productcategoryname,
ROW_NUMBER() over (partition by elgi_productcategoryname order by elgi_productcategoryname) as Rnk
from ELGI_DEALERPRICELISTMASTER)a
where Rnk=1
) e
on ltrim(rtrim(d.elgi_itemcode))=ltrim(rtrim(e.elgi_itemcode))
where a.elgi_enquiry in 
(select distinct  opportunityid from CRM_OPPURTUNITY
where elgi_enquirystatusname<>'On Hold'
and elgi_businessverticalname='Industrial'
and elgi_sourceofenquiryname not in ('Website'))
and a.elgi_revisedflagname not in ('Yes')
)Offers
where CRM_Units_Fact.opportunityid=Offers.elgi_enquiry


end
