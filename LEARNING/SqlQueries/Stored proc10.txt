insert into tb4
select isnull(section_id,0) as section_id,Engg_User_id,Metrics_levels,ordering,Act_Count
from(
select section_id, Engg_User_id,Metrics_levels, count(Metrics_levels) as Act_Count,
case when Metrics_levels='Prospects' then 1
when Metrics_levels='Enquiry Generation' then 2
when Metrics_levels='Enquiry Conversion' then 3
when Metrics_levels='Order_won' then 4
when Metrics_levels='Order_lost' then 5
when Metrics_levels='Order_dropped' then 6 end as ordering
from (
select a.section_id,a.user_id Engg_User_id,
case when a.type  ='lead' and a.stage_id  =1 then 'Prospects'
when a.stage_id  =4 and a.type  ='opportunity' then 'Enquiry Generation'
when a.stage_id  =5 and a.type  ='opportunity' then 'Enquiry Conversion'
when a.stage_id  =7 and a.type  ='opportunity' then 'Order_won'
when a.stage_id  =8 and a.type  ='opportunity' then 'Order_lost'
when a.stage_id  =6 and a.type  ='opportunity' then 'Order_dropped'
end as Metrics_levels
                           
from crm_lead a
where a.stage_id in(1,4,5,6,7,8)
--and a.section_id=4 for analysis purpose
)a
group by section_id,Engg_User_id,Metrics_levels
)b
order by section_id,Engg_User_id,ordering

--since Source data doesnt have all metrics like prospects,EG,EC,Order won,Order lost, Order dropped by default
--for each sales person, below strategy is used to get all metrics for each person
--in order to avoid to Cumulative count mismatch between Managers data and sum of his subordinates data.

declare @counts int
declare @section_id int
declare @Engg_User_id int
declare @rownum int=1

set @counts=
(select count(*) as counts from (
select * from(
select *,row_number() over(order by engg_user_id) Rownum from
(select distinct section_id,engg_user_id from tb4 )a
)b
)c
)


insert into All_Metrics1
select *  from(
select *,row_number() over(order by engg_user_id) Rownum from
(select distinct section_id,engg_user_id from tb4 )a
)b


while (@rownum<=@counts)
begin
set @section_id =(select section_id from All_Metrics1 where rownum=@rownum)
set @Engg_User_id =(select Engg_User_id from All_Metrics1 where rownum=@rownum)
--select @section_id
insert into  All_Metrics2(section_id,Engg_User_id,ordering,Metrics_levels) values
(@section_id,@Engg_User_id,1,'Prospects'),
(@section_id,@Engg_User_id,2,'Enquiry Generation'),
(@section_id,@Engg_User_id,3,'Enquiry Conversion'),
(@section_id,@Engg_User_id,4,'Order_won'),
(@section_id,@Engg_User_id,5,'Order_lost'),
(@section_id,@Engg_User_id,6,'Order_dropped')


update All_Metrics2 set section_id=0
where section_id is null and Engg_User_id=@Engg_User_id

set @rownum=@rownum+1

end



insert into tb4a
select a.section_id,a.Engg_User_id,a.ordering,a.Metrics_levels,isnull(b.act_count,0) act_count from All_Metrics2 a 
left outer join tb4 b
on 
a.section_id=b.section_id 
and a.Engg_User_id=b.Engg_User_id
and a.ordering=b.ordering
