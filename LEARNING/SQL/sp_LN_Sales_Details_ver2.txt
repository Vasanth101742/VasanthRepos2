(company,dpst,trantype,OrderNo,PositionNo,Invoice,custcode,Item,t_rate_1,t_rate_2,ED_AMOUNT,t_amth_1,t_txah_1,
Foreign_currency_amount,UnitPrice,DeliveryQty,

SELECT b.company,b.t_cofc,b.Trantype,b.OrderNum,b.PositionNum,
b.invnum,b.CustCode,b.t_item ,b.t_rate_1 ,b.t_rate_2 ,
sum(b.t_txai) t_txai,sum(isnull(b.t_amth_1,0)),sum(isnull(b.t_txah_1,0)),

convert(date,a.t_idat),a.t_idat,a.t_ccur
FROM (select Company,Trantyp,invnum,t_idat,t_ccur from CI_Header
where t_idat>=@InvoiceDate
) a
left outer join CI_Lines b
on a.Company=b.Company 
and ltrim(rtrim(a.Trantyp)) =ltrim(rtrim(b.Trantype ))
and ltrim(rtrim(a.invnum)) =ltrim(rtrim(b.invnum)) 

group by b.company,b.t_cofc,b.Trantype,b.OrderNum,b.PositionNum,b.invnum,b.CustCode,

on a.currency=b.bcur
and a.invoice_date>=b.STDT
)a
where Rnk=1
