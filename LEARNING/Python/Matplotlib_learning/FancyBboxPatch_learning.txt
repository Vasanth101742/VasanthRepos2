import pandas as pd
import matplotlib.pyplot as plt
from matplotlib.patches import FancyBboxPatch



# ----------------------------
# DATA
# ----------------------------

df = dataset.copy()
df.columns = ["ProductGroup", "Sales", "YoY"]
df = df.dropna(subset=["ProductGroup", "Sales"])


df["ProductGroup"] = df["ProductGroup"].astype(str)
df["Sales"] = pd.to_numeric(df["Sales"], errors="coerce")
df["YoY"] = pd.to_numeric(df["YoY"], errors="coerce")



df = df.dropna(subset=["Sales"])

df = (
    df.sort_values("Sales", ascending=False)
      .head(10)
      .reset_index(drop=True)
)


max_sales = df["Sales"].max()



# ----------------------------
# FIGURE
# ----------------------------

fig, ax = plt.subplots(figsize=(16,10))

fig.patch.set_facecolor("white")
ax.set_facecolor("white")


# IMPORTANT
ax.set_xlim(0,110)

# More vertical room used
row_gap = 1.4

ax.set_ylim(
    0,
    len(df)*row_gap + 3
)

ax.axis("off")


# ----------------------------
# HEADER
# ----------------------------

header_y = len(df)*row_gap + 2

ax.text(
    1.5,
    header_y,
    "●",
    fontsize=20,
    color="#ED143D",
    va="center"
)



ax.text(
    4.5,
    header_y,
    "Level 1 Groups",
    fontsize=30,
    fontweight="bold",
    color="#6B7280",
    va="center"
)



ax.plot(
    [1,119],
    [len(df)*row_gap + 1.2,
     len(df)*row_gap + 1.2],
    color="#E5E7EB",
    linewidth=1.5
)



# ----------------------------
# LAYOUT
# ----------------------------



label_x = 3
track_x = 35
track_width = 50
value_x = 95
yoy_x = 108



# ----------------------------
# ROWS
# ----------------------------



for idx, row in df.iterrows():

    y = len(df)*row_gap - idx*row_gap

    sales = row["Sales"]
    yoy = row["YoY"]

    ratio = sales / max_sales

    # Product name

    ax.text(
        label_x,
        y,
        row["ProductGroup"],
        fontsize=23,
        color="#333333",
        va="center"
    )



    # Grey track

    track = FancyBboxPatch(
        (track_x, y-0.28),
        track_width,
        0.56,
        boxstyle="round,pad=0.02,rounding_size=0.28",
        linewidth=0,
        facecolor="#E5E7EB"
    )

    ax.add_patch(track)


    # Red bar

    fill = FancyBboxPatch(
        (track_x, y-0.28),
        track_width*ratio,
        0.56,
        boxstyle="round,pad=0.02,rounding_size=0.28",
        linewidth=0,
        facecolor="#ee3124"
    )


    ax.add_patch(fill)


    # Value

    ax.text(
        value_x,
        y,
        f"{sales/1000000:.1f}",
        fontsize=25,
        fontweight="bold",
        color="#222222",
        ha="right",
        va="center"
    )


    # YoY

    if pd.notna(yoy):

        if yoy > 0:
            arrow = "▲"
            color = "#22C55E"

        elif yoy < 0:

            arrow = "▼"
            color = "#EF4444"

        else:

            arrow = "►"
            color = "#9CA3AF"


        ax.text(
            yoy_x,
            y,
            f"{arrow} {abs(yoy)*100:.0f}%",
            fontsize=25,
            fontweight="bold",
            color=color,
            ha="right",
            va="center"
        )



# ----------------------------
# FINAL
# ----------------------------

plt.subplots_adjust(
    left=0.01,
    right=0.99,
    top=0.99,
    bottom=0.01
)


plt.show()
