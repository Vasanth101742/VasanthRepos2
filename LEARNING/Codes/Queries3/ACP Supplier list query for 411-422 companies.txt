select company,ifbp as suppcode,b.custname1 as SuppName
from ttfacp200all a
left outer join vw_business_partner b
on ltrim(rtrim(a.ifbp))=ltrim(rtrim(b.custcode))
where company between 411 and 422
--and ifbp='SU1A01930'
group by company,ifbp,b.custname1
order by company,ifbp,b.custname1


select distinct company,ifbp as suppcode,b.custname1 as SuppName from ttfacp200all a
left outer join vw_business_partner b
on ltrim(rtrim(a.ifbp))=ltrim(rtrim(b.custcode))
where ifbp in
(
select ifbp
from ttfacp200all
where company between 411 and 422
group by ifbp
)
and company not between 411 and 422
and company like '4%'
