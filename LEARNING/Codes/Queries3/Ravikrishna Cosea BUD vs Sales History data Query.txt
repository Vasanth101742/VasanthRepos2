select Organ,DPSTCode,DPSTName,AreaCode,AreaName,--Category,ProductLine,
ItemCode,--Item_desc,
sum(BudQty) BudQty,
sum(SalesQty) SalesQty,
sum(Budget) Budget_INR,
sum(Sales_amount) Sales_amount_INR,
sum(Budget_USD) Budget_USD,
sum(Sales_USD) Sales_USD,
sum(Budget_Foreign_Currency) Budget_Foreign_Currency,
sum(Sales_Foreign_currency) Sales_Foreign_currency,
sum(case when areacode='631' then isnull(Budget_Foreign_Currency,0) else isnull(Budget_USD,0) end )/1000 as budget,
sum(case when areacode='631' then isnull(Sales_Foreign_currency,0) else isnull(Sales_USD,0) end )/1000 as sales_amount,
Cal_year,Cal_month,Fin_year,Fin_month,substring(datename(mm,dateadd(mm,Cal_month,-1)),1,3) as Month_Name
from Bud_sales_Inflow_COSEA 
where Fin_year>=2017
--and itemcode='S011292'
group by Organ,DPSTCode,DPSTName,AreaCode,AreaName,Category,ProductLine,ItemCode,Item_desc,
Cal_year,Cal_month,Fin_year,Fin_month
order by Fin_year,Fin_month
