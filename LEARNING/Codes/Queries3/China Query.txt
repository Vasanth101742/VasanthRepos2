custname,dpst,dpstname,Organ,areacode,areaname,Foreign_currency_amount,UnitPrice,DeliveryQty,cal_Month,cal_year,
Monthname,Sales_amount,SaleOrderType,Sales_USD--,Rate_USD

select company,trantype,Invoice,Invoice_Date,Item,Item_desc,custcode,custname,dpst,dpstname,areacode,areaname,
currency,t_rate_1 as CurrRate,
sum(Sales_amount) as Sales_INR,
Rate_USD,
sum(Sales_USD) Sales_USD,
Fin_year,Fin_month,Monthname,organ
from ELGi_Sales_Details
where Fin_year=2019
and Vertical1 like '%china%'
and sales_category not in (1)
group by 
company,trantype,Invoice,Invoice_Date,Item,Item_desc,custcode,custname,dpst,dpstname,areacode,areaname,
currency,t_rate_1,Rate_USD,
Fin_year,Fin_month,Monthname,organ
order by Invoice_Date
