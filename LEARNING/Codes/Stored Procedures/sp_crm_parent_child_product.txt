insert into tb5_ver4
select * from (select * From tb5_ver2  where act_count <>0)a
inner join 
(select f.stage_id as stage_id,a.id as category_id,a.name as categoryname,
b.id as product_id,b.name productname,
d.id as order_line_id,e.id as order_id,f.id as lead_id ,
d.product_uom_qty as productqty,
f.enquiry_date as prospect_date,
case when (stage_id >=4 and date_enquiry_generation is not null) then f.date_enquiry_generation else f.enquiry_date end as EG_date,
--case when (stage_id >=4 and date_enquiry_generation is not null) then f.enquiry_date else null end as EG_date,
--case when stage_id >=4 then f.enquiry_date else null end as EG_date,
--case when stage_id =5 then e.date_order else null end as EC_date,
e.date_order as EC_date,
case when f.stage_id =7 then e.date_confirm else null end as won_date
From product_category a

left outer join product_template b
on a.id=b.categ_id 

left outer join product_master c
on b.id=c.product_tmpl_id 

left outer join sale_order_line d
on d.product_id =c.id 

left outer join sale_order e
on d.order_id =e.id 

left outer join Crm_Lead f
on f.id=e.opportunity_id 

where stage_id in (1,4,5,7)
and f.user_id in (select id from res_users where active=1)
and a.name <>'ACCESSORY'

)b
on a.crm_lead_id=b.lead_id
