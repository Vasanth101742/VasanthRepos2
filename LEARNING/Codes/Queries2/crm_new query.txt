select * from 
(
select a.section_id,a.user_id,a.id as lead_id ,a.stage_id,
case when a.type  ='lead' and a.stage_id  =1 then 'Prospects'
when a.stage_id  =4 and a.type  ='opportunity' then 'Enquiry Generation'
when a.stage_id  =5 and a.type  ='opportunity' then 'Enquiry Conversion'
when a.stage_id  =7 and a.type  ='opportunity' then 'Order_won'
when a.stage_id  =8 and a.type  ='opportunity' then 'Order_lost'
when a.stage_id  =6 and a.type  ='opportunity' then 'Order_dropped'
end as Metrics_levels,
enquiry_date as prospects_date,b.date_order as ec_date,b.date_confirm as won_date,
b.state,
--case when b.state='progress'then 'won' else 'EC'end as status
case when enquiry_date is not null and date_order is not null and date_confirm is not null and b.state in ('progress','manual') then 'WON'
when  enquiry_date is not null and date_order is not null and date_confirm is not null and b.state in ('draft','sent') then 'TobeWon'
when  enquiry_date is not null and date_order is not null and date_confirm is not null and b.state in ('lost','drop','cancel') then 'NotWon'
when  enquiry_date is not null and date_order is not null and date_confirm is null then 'EC'
when  enquiry_date is not null and date_order is null and date_confirm is null then 'Prospects'
end as Status,
--datediff(dd,enquiry_date,isnull(date_order,enquiry_date)) as Prospects_to_EC_Days,
case when date_order is not null then 
case when datediff(dd,enquiry_date,date_order) < 0 then datediff(dd,enquiry_date,date_order)*(-1)
else datediff(dd,enquiry_date,date_order) end
else '-9999'
end
as Prospects_to_EC_Days,
case when date_confirm is not null then 
case when datediff(dd,date_order,date_confirm) <0 then datediff(dd,date_order,date_confirm)*(-1)
else datediff(dd,date_order,date_confirm) end
else '-9999'
end
as EC_to_Won_Days
from Crm_Lead a
left join sale_order b
on ltrim(rtrim(a.id))=ltrim(rtrim(b.opportunity_id ))

where a.stage_id in(1,4,5,6,7,8)
and a.user_id in(select distinct id from res_users where active=1)
)b

on a.L2_ID=b.section_id
and a.Engg_User_ID=b.user_id
where L4_ID is null
