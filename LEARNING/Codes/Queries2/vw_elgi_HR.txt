CREATE view vw_elgi_HR as

SELECT 	GLOBAL, 
	A.Emp_Name,
	OLD_BAAN_EMPCODE, 
	CAST(coalesce (A.Emp_Code,OLD_BAAN_EMPCODE) AS INT)  EMP_CODE,
	--a.emp_code,
	Dense_rank() OVER(PARTITION BY LTRIM(RTRIM(A.EMP_CODE)),finmonthno
order by global ) AS ERANK,
	A.Designation, 
	Org_Unit, 
	isnull(Division_Name,'') Division_Name, 
	isnull(Function_Name,'') Function_Name,
	isnull(Department,'') Department, 
	isnull(ltrim(rtrim(A.Band)),'') Band,
	Location,
	Emp_Status, 
	Reason, 
	Seperation_Date, 
	Transfer_Date, 
	New_Division, 
	New_Function, 
	New_Department, 
	EE_Subgroup, 
	Start_Date, 
	Data_Load_Date, 
	num, 
	FinMonthNo, 
	FinMonthName, 
	FinYear, 
	Fin_Year
FROM (
	SELECT distinct 'DOMESTIC' AS GLOBAL, 
		Emp_Name,
		' ' AS OLD_BAAN_EMPCODE, 
		Emp_Code, 
		Band, 
		Designation, 
		Org_Unit, 
		ltrim(rtrim(Division_Name)) as Division_Name,
		ltrim(rtrim(Function_Name)) as Function_Name, 
		ltrim(rtrim(Department)) as Department,
		Location,
		Emp_Status, 
		Reason, 
		Seperation_Date, 
		Transfer_Date, 
		New_Division, 
		New_Function, 
		New_Department, 
		EE_Subgroup, 
		Start_Date, 
		Data_Load_Date, 
		num, 
		FinMonthNo, 
		FinMonthName, 
		FinYear, 
		Fin_Year 

	FROM 	HR_Master
	where emp_code not  between 800000 and 899999

	UNION ALL

	SELECT distinct 'INTERNATIONAL' AS GLOBAL, 
		Emp_Name,
		F.Emp_Code AS OLD_BAAN_EMPCODE, 
		I.NEW_ECODE AS Emp_Code, 
		f.Band, 
		Designation, 
		Org_Unit, 
		case when ltrim(rtrim(DIVNM)) is null then ltrim(rtrim(f.Division_Name))
		else ltrim(rtrim(DIVNM)) end as Division_Name,
		case when ltrim(rtrim(FUNNM)) is null then ltrim(rtrim(f.Function_Name)) 
		else ltrim(rtrim(FUNNM)) end as Function_Name, 
		case when ltrim(rtrim(DEPNM)) is null then ltrim(rtrim(f.Department))
		else ltrim(rtrim(DEPNM)) end as Department,
		Location, 
		Emp_Status, 
		Reason, 
		'' as Seperation_Date, 
		Transfer_Date, 
		New_Division, 
		New_Function, 
		New_Department, 
		EE_Subgroup, 
		Start_Date, 
		Data_Load_Date, 
		num, 
		FinMonthNo, 
		FinMonthName, 
		FinYear, 
		Fin_Year 

	FROM 	HR_EMP F 
		left JOIN HR_OLD_NEW_EMPCODE I
		ON F.Emp_Code=I.OLD_ECODE
		left join
		(
		select 	a.ECODE,a.DIVCD,b.DIVNM,a.FUNCD,b.FUNNM,a.DEPCD,b.DEPNM 
		from 	tthhrd113 a
			inner join  tthhrd085 b
				on a.DIVCD=b.DIVCD
				and a.FUNCD=b.FUNCD
				and a.DEPCD=b.DEPCD
		) J

		--on ltrim(rtrim(I.NEW_ECODE))=ltrim(rtrim(J.ECODE))
		on ltrim(rtrim(f.Emp_Code))=ltrim(rtrim(J.ECODE))
		
		union all 
		
		
		select 'INTERNATIONAL' AS GLOBAL, 
		ENAME,a.ECODE,a.ECODE,'','','',
		--b.DIVNM,b.FUNNM,b.DEPNM,
		'','','',
		'','','',DDATE as Seperation_Date,
		 '','','','','','',
			--'','','','','' ,'',
			'2016-02-29 00:00:00.000' as Data_Load_Date,
	   1 as num,
    case when month('2016-02-29 00:00:00.000') between 1 and 3 then month('2016-02-29 00:00:00.000') + 9 else month('2016-02-29 00:00:00.000') - 3 end as FinMonthNo,
datename(mm,'2016-02-29 00:00:00.000')  as FinMonthName,
case when month('2016-02-29 00:00:00.000') between 1 and 3 then year('2016-02-29 00:00:00.000') - 1 else year('2016-02-29 00:00:00.000')  end as FinYear,
CASE WHEN month('2016-02-29 00:00:00.000') > 3 THEN (CONVERT(varchar, year('2016-02-29 00:00:00.000')) + '-' + CONVERT(varchar, year('2016-02-29 00:00:00.000') + 1)) 
                      ELSE (CONVERT(varchar, year('2016-02-29 00:00:00.000') - 1) + '-' + CONVERT(varchar, year('2016-02-29 00:00:00.000'))) END AS Fin_Year

			
			from tthhrd031100 a
--inner join
--(
--		select 	a.ECODE,a.DIVCD,b.DIVNM,a.FUNCD,b.FUNNM,a.DEPCD,b.DEPNM 
--		from 	tthhrd113 a
--			inner join  tthhrd085 b
--				on a.DIVCD=b.DIVCD
--				and a.FUNCD=b.FUNCD
--				and a.DEPCD=b.DEPCD
--		) b
--		on a.DIVCD=convert(varchar,b.DIVCD)
				--and ltrim(rtrim(a.FUNCD))=ltrim(rtrim(b.FUNCD))
				--and ltrim(rtrim(a.DEPCD))=ltrim(rtrim(b.DEPCD))

		where a.ECODE>799999 and a.ECODE<900000
and month(DDATE)= (select case when month(getdate()-1)=1 then 12 else month(getdate()-1)-1 end)
and year(DDATE) = (select case when month(getdate()-1)=1 then year(getdate())-1 else year(getdate()) end )


 )A
