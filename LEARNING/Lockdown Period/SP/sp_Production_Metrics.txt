begin

Truncate table Production_Metrics_Temp

insert into Production_Metrics_Temp(company,warehouse,item,Ordernum,Date1,Planned_qty)
select * from (
select a.COMPANY,a.CWAR,a.MITM,a.PDNO,RDLD,
sum(case when RDLD='9999-01-01 00:00:00.000' then null else (case when osta in (1,2,3) then qrdr else 0 end) end) as Planned_qty
from TTISFC001ALL a 
where a.PDNO not like '%[a-z]%'
--and convert(date,RDLD) 
--between dateadd(MM,datediff(MM,0,getdate()),0) and
--<=dateadd(MS,-3,dateadd(mm,0,dateadd(mm,datediff(mm,0,getdate())+1,0)))
group by a.COMPANY,a.CWAR,a.MITM,a.PDNO,RDLD
)a1
where Planned_qty>0


--WIP (Till Yesterday)

insert into Production_Metrics_Temp(company,warehouse,item,Ordernum,Date1,WIP_qty)
select * from (
select a.COMPANY,a.CWAR,a.MITM,a.PDNO,
b.date as WIP_Date,
sum(case when ADLD='9999-01-01 00:00:00.000' then 0 else (case when a.OSTA IN(4,6) then qrdr else 0 end) end) as WIP_qty
from  TTISFC001ALL a 
left outer join
(
select * from (
select PDNO,OSTA,convert(date,date) as date,row_number() over (partition by pdno order by date) dateRank 
from TTICST151ALL
where OSTA in (4,6)
--and PDNO='230009816' --for testing Purpose
)b1
where dateRank=1
)b
on a.PDNO=b.PDNO

where a.PDNO not like '%[a-z]%'
and b.date< getdate()-1
group by a.COMPANY,a.CWAR,a.MITM,a.PDNO,b.date
)a
where WIP_qty<>0

from (select company,warehouse,item,Ordernum,date1 as PlannedDate,
case when month(date1)=month(getdate()-1)  and date1 <=getdate()-1 then 'Normal'
when month(date1)<month(getdate()-1) then 'Backlog'
when month(date1)=month(getdate()-1) and date1 > getdate()-1 then 'Advance'
end as OrderType
from Production_Metrics_Temp
where Planned_qty<>0
group by company,warehouse,item,Ordernum,date1
)a
where Production_Metrics_Temp.company=a.company
and Production_Metrics_Temp.Ordernum=a.Ordernum

Insert into Production_Metrics_Temp2(Description,company,warehouse,item,itemdescr,Ordernum,Date1,
MonthPlanQty,Actual_qty,WIP_qty,Planned_qty,Future_Planned_qty,FG_qty,AssemblyLine,CurrentDate,
weekNo,CalMonth,CalYear)
select Description,company,warehouse,item,itemdescr,Ordernum,Date1,
MonthPlanQty,Actual_qty,WIP_qty,Planned_qty,Future_Planned_qty,FG_qty,AssemblyLine,CurrentDate,
weekNo,CalMonth,CalYear from (
select * from (
select Description,company,warehouse,item,itemdescr,Ordernum,Date1,
MonthPlanQty,Actual_qty,WIP_qty,
case when Date1<getdate()-1 then null else Planned_qty end as Planned_qty,
case when Date1<getdate()-1 then null else Planned_qty end as Future_Planned_qty,
FG_qty,AssemblyLine,
convert(date,getdate()-1) as CurrentDate,
case when Date1<getdate()-1 then -100 else datepart(ISO_WEEK,Date1) end as weekNo,
datepart(mm,Date1) as CalMonth,datepart(YYYY,Date1) as CalYear
from Production_Metrics_Temp )a
where year(Date1)=@CalYear
and month(Date1)=@CalMonth
--and ltrim(rtrim(AssemblyLine))  in (@AssemblyLine )

union all
--Current Month Backlog Planned orders
select * from (
select Description,company,warehouse,item,itemdescr,Ordernum,Date1,
0 as MonthPlanQty,0 as Actual_qty,0 as WIP_qty,Planned_qty,Planned_qty as Future_planned_qty,
0 as FG_qty,AssemblyLine,
convert(date,getdate()-1) as CurrentDate,
--datepart(ISO_WEEK,Date1) as weekNo,datepart(mm,Date1) as CalMonth,datepart(YYYY,Date1) as CalYear
-1 as weekNo,1 as CalMonth,datepart(YYYY,Date1) as CalYear
from Production_Metrics_Temp )a
where Date1<=getdate()-1
and year(Date1)=@CalYear
and month(Date1)=@CalMonth
and Future_planned_qty<>0
--and ltrim(rtrim(AssemblyLine))  in (@AssemblyLine )


union all
--upto Previous Month Backlog Planned orders
select * from (
select Description,company,warehouse,item,itemdescr,Ordernum,Date1,
0 as MonthPlanQty,0 as Actual_qty,0 as WIP_qty,
--0 as Planned_qty,
Planned_qty,
Planned_qty as Future_planned_qty,
0 as FG_qty,AssemblyLine,
convert(date,getdate()-1) as CurrentDate,
--datepart(ISO_WEEK,Date1) as weekNo,datepart(mm,Date1) as CalMonth,datepart(YYYY,Date1) as CalYear
-1 as weekNo,1 as CalMonth,datepart(YYYY,Date1) as CalYear
from Production_Metrics_Temp )a
where year(Date1)=@CalYear
and month(Date1)<@CalMonth
and Future_planned_qty<>0
--and ltrim(rtrim(AssemblyLine))  in (@AssemblyLine )


union all
--Future Planned orders
select * from (
select Description,company,warehouse,item,itemdescr,Ordernum,Date1,
0 as MonthPlanQty,0 as Actual_qty,0 as WIP_qty,
--0 as Planned_qty,
Planned_qty,
Planned_qty as Future_Planned_qty,
0 as FG_qty,AssemblyLine,
convert(date,getdate()-1) as CurrentDate,--datepart(ISO_WEEK,Date1) as weekNo
datepart(ISO_WEEK,Date1) as weekNo,datepart(mm,Date1) as CalMonth,datepart(YYYY,Date1) as CalYear
from Production_Metrics_Temp )a
where Date1>=(select dateadd(dd,1,eomonth(getdate()-1)))
and Future_Planned_qty<>0
--and ltrim(rtrim(AssemblyLine))  in (@AssemblyLine )
)a
--where Ordernum='200019888' --for testing purpose
order by date1




end
