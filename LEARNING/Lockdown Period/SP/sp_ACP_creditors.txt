when a.DueDate<>'9999-01-01 00:00:00.000' and  datediff(dd,a.DueDate,@AS_OF_DATE) between 46 and 90 then '46-90'
when a.DueDate<>'9999-01-01 00:00:00.000' and  datediff(dd,a.DueDate,@AS_OF_DATE) between 91 and 120 then '91-120'

--and DOCD<=@AS_OF_DATE  --Docdt based
group by COMPANY,TTYP,NINV,DOCD,DOCN,IFBP,LINE,CCUR,RATE$1,YEAR,PROD,DUED,ORNO,REFR

(select COMPANY,IFBP,TTYP,NINV,LINE,ccur as Currency,

--and DOCD<=@AS_OF_DATE --Docdt based

)b
on a.company=b.company
and a.NINV=b.NINV
and a.TTYP=b.TTYP
and a.IFBP=b.IFBP
and a.LINE=b.LINE

group by a.DOCD,a.COMPANY,a.TTYP,a.NINV,a.IFBP,a.DueDate,
FinYear,FinMonth,CalYear,CalMonth,a.CurRate,a.Currency,a.LINE,a.ORNO,a.REFR
)a
group by DocDt,COMPANY,TranType,InvNo,SuppCode,SuppName,SuppAreaCode,SuppAreaName,DPST,DPSTName,Description,InvAmnt,PaidAmnt,
BalanceAmnt,DueDate,Ageing,AgeingOrdr,FinYear,FinMonth,CalYear,CalMonth,CurRate,Currency,InvAmnti,PaidAmnti,lINE,BalanceAmnti,

update ACP_Creditors_stage set Rate_USD= a.CurRate
from (select distinct DocDt,currency,CurRate from ACP_Creditors_stage 
where currency='USD'
and company not in (500,551,552,553)
and CurRate>1
)a
where ACP_Creditors_stage.DocDt=a.DocDt
and ACP_Creditors_stage.currency='USD'
and ACP_Creditors_stage.company not in (500,551,552,553)



--Data Reconciliation
update ACP_Creditors_stage set Rate_USD =a.Rate_USD 
from (
select DocDt,Rate_USD,Rnk1 from (
select a.DocDt,b.Rate_USD,row_number() over(partition by a.DocDt order by datediff(dd,b.DocDt,a.DocDt) desc ) as Rnk1 from
(select distinct DocDt,Rate_USD from ACP_Creditors_stage 
where Rate_USD is null
and company not in (500,551,552,553)
)a
left outer join
(select distinct DocDt,Rate_USD from ACP_Creditors_stage 
where Rate_USD is not null
and company not in (500,551,552,553)
)b
on a.DocDt<=b.DocDt
)a
where Rnk1=1
)a
where ACP_Creditors_stage.DocDt=a.DocDt
and ACP_Creditors_stage.Rate_USD is null
and ACP_Creditors_stage.company not in (500,551,552,553)

--End: Part 1: Rate_USD Update for INR to USD----------------------
