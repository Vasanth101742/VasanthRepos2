update ATS_Inventory_vs_Lean_Temp set

--,Cumul_Inward_Value=a.T$DAMT
from (
select COMPANY,ITEM,sum(QAPR) QAPR,
sum(QREC) QREC,
sum(T$DAMT) T$DAMT from 
(select COMPANY,ITEM,RCNO,RCLN,ORNO,PONO,SEQN,
sum(QAPR) QAPR,
sum(QREC) QREC
 from TWHINH312ALL
where company like '6%'
and CWAR in ('651','652','653','654','655','667','674','675','676')
and year(ARDT)=year(getdate()-1)
and month(ARDT)=month(getdate()-1)
and LSTA=30
group by COMPANY,ITEM,RCNO,RCLN,ORNO,PONO,SEQN)a
inner join
(select T$RCNO,T$RSEQ,T$ORNO,T$PONO,T$SQNB,sum(T$DAMT) T$DAMT from TTDPUR406ALL
where company like '6%'
group by T$RCNO,T$RSEQ,T$ORNO,T$PONO,T$SQNB
)b
on a.RCNO=b.T$RCNO
and a.RCLN=b.T$RSEQ
and a.ORNO=b.T$ORNO
and a.PONO=b.T$PONO
and a.SEQN=b.T$SQNB
group by COMPANY,ITEM
)a
where ATS_Inventory_vs_Lean_Temp.company=a.company
and ltrim(rtrim(ATS_Inventory_vs_Lean_Temp.item))=ltrim(rtrim(a.ITEM))

--,sum(T$DAMT) T$DAMT --Comment this
from 
(select COMPANY,ITEM,RCNO,RCLN,ORNO,PONO,SEQN,
sum(QAPR) QAPR ,sum(QREC) QREC
from TWHINH312ALL
where company like '6%'
and CWAR in ('651','652','653','654','655','667','674','675','676')
and year(ARDT)=year(getdate()-1)
and month(ARDT)=month(getdate()-1)
and LSTA=30
group by COMPANY,ITEM,RCNO,RCLN,ORNO,PONO,SEQN)a
inner join
(select T$RCNO,T$RSEQ,T$ORNO,T$PONO,T$SQNB,sum(T$DAMT) T$DAMT from TTDPUR406ALL
where company like '6%'
group by T$RCNO,T$RSEQ,T$ORNO,T$PONO,T$SQNB
)b
on a.RCNO=b.T$RCNO
and a.RCLN=b.T$RSEQ
and a.ORNO=b.T$ORNO
and a.PONO=b.T$PONO
and a.SEQN=b.T$SQNB
where ltrim(rtrim(ITEM)) not in (select distinct ltrim(rtrim(ITEM)) from ATS_Inventory_vs_Lean_Temp)
group by COMPANY,ITEM
