CREATE proc sp_Consumption as

begin

declare @Date1 date=(select dateadd(mm,-4,eomonth(getdate()-1)))

Truncate table Consumption

insert into Consumption (COMPANY,ItemCode,Production_Consumption,Sales_Consumption,CalMonth,CalYear,TRDT)
select a.COMPANY,ITEM,a.Production_Consumption,a.Sales_Consumption,CalMonth,CalYear,TRDT
from 
(
select company,item,sum(QSTK) Sales_Consumption,0 as Production_Consumption,
month(TRDT) as CalMonth,year(TRDT) as CalYear,convert(date,TRDT) as TRDT
from TWHINR110ALL
where kost=5
and koor=3
and company like '4%'
--and concat(RIGHT('00'+CAST(month(TRDT) AS VARCHAR(2)),2),cast(year(TRDT) as varchar(4))) in (@MMYYYY)
and TRDT >@Date1
group by convert(date,TRDT),company,item,month(TRDT),year(TRDT),convert(date,TRDT)

--union
--select company,item,0,sum(QSTK) Production_Consumption,month(TRDT) as CalMonth,year(TRDT) as CalYear,convert(date,TRDT) as TRDT
--from TWHINR110ALL
--where kost=5
--and koor=1
--and company like '4%'
--and TRDT >@Date1
--group by convert(date,TRDT),company,item,month(TRDT),year(TRDT),convert(date,TRDT)

)a


insert into Consumption (COMPANY,ItemCode,Production_Consumption,Sales_Consumption,CalMonth,CalYear,TRDT)
select a.COMPANY,ITEM,a.Production_Consumption,a.Sales_Consumption,CalMonth,CalYear,TRDT
from 
(
--select company,item,sum(QSTK) Sales_Consumption,0 as Production_Consumption,
--month(TRDT) as CalMonth,year(TRDT) as CalYear,convert(date,TRDT) as TRDT
--from TWHINR110ALL
--where kost=5
--and koor=3
--and company like '4%'
----and concat(RIGHT('00'+CAST(month(TRDT) AS VARCHAR(2)),2),cast(year(TRDT) as varchar(4))) in (@MMYYYY)
--and TRDT >@Date1
--group by convert(date,TRDT),company,item,month(TRDT),year(TRDT),convert(date,TRDT)

--union

select company,item,0 as Sales_Consumption,sum(QSTK) Production_Consumption,month(TRDT) as CalMonth,year(TRDT) as CalYear,convert(date,TRDT) as TRDT
from TWHINR110ALL
where kost=5
and koor=1
and company like '4%'
and TRDT >@Date1
group by convert(date,TRDT),company,item,month(TRDT),year(TRDT),convert(date,TRDT)

)a


update Consumption set ItemDescr=a.DSCA
from TTCIBD001465 a
where ltrim(rtrim(Consumption.ItemCode))=ltrim(rtrim(a.ITEM))
and Consumption.COMPANY=465

update Consumption set ItemDescr=a.DSCA
from ttcibd001470 a
where ltrim(rtrim(Consumption.ItemCode))=ltrim(rtrim(a.ITEM))
and Consumption.COMPANY=470

update Consumption set ItemDescr=a.DSCA
from TTCIBD001490 a
where ltrim(rtrim(Consumption.ItemCode))=ltrim(rtrim(a.ITEM))
and Consumption.COMPANY=490

update Consumption set ItemDescr=a.t_dsca
from TTCIBD001400 a
where ltrim(rtrim(Consumption.ItemCode))=ltrim(rtrim(a.t_item))
and Consumption.ItemDescr is null



update Consumption set BuyrCode=b.BUYR,SuppCode=b.OTBP
from (
select distinct company ,item,BUYR,OTBP from ttdipu001all 
where company =465 and 
len(ltrim(rtrim(concat(BUYR,otbp))))>2
) b
where ltrim(rtrim(Consumption.ItemCode))=ltrim(rtrim(b.ITEM))
and Consumption.Company= b.company
and Consumption.Company=465



update Consumption set BuyrCode=b.BUYR,SuppCode=b.OTBP
from (
select distinct company ,item,BUYR,OTBP from ttdipu001all 
where company =470 and 
len(ltrim(rtrim(concat(BUYR,otbp))))>2
) b
where ltrim(rtrim(Consumption.ItemCode))=ltrim(rtrim(b.ITEM))
and Consumption.Company= b.company
and Consumption.Company=470

select distinct company ,item,BUYR,OTBP from ttdipu001all 
where company =490 and 
len(ltrim(rtrim(concat(BUYR,otbp))))>2
) b

and Consumption.Company= b.company
and Consumption.Company=490

from (
select distinct company ,item,BUYR,OTBP from ttdipu001all 
where company not in (465,470,490) 
and COMPANY like '4%'
and len(ltrim(rtrim(concat(BUYR,otbp))))>2
) b
where ltrim(rtrim(Consumption.ItemCode))=ltrim(rtrim(b.ITEM))
and Consumption.BuyrCode is null

update Consumption set buyrname=c.emp_name
from
(select isnull(new_emp_code,emp_code) as new_emp_code, emp_name from employee_details) c
where consumption.buyrcode=c.new_emp_code 

update Consumption set SuppName=d.custname1
from vw_Business_Partner d
where Consumption.SuppCode=d.custcode



end

select * from Consumption
