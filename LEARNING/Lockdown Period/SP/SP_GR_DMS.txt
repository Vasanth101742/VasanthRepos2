--exec [SP_GR_DMS]
CREATE proc [SP_GR_DMS]
as
begin
select distinct *,g1.rca_id from (
select distinct *,case when a.DefectAging between 0 and 45 then '0-45 Days' when a.DefectAging between 46 and 90 then '46-90 Days'  when a.DefectAging>90 then 'Above 90 Days' end 'AgingCom'
,case when a.TargetDefectAging <0 then 'In Due' when a.TargetDefectAging between 0 and 45 then '0-45 Days' when a.TargetDefectAging between 46 and 90 then '46-90 Days'  when a.TargetDefectAging>90 then 'Above 90 Days' end 'TargetDefectAgingCom'
 from (select a.defect_id,B.source_name'Source',(case when C.productline_name='-' then h.inloc_name else C.productline_name end)  'Product Line',
   case when (case when C.productline_name='-' then h.inloc_name else C.productline_name end) like '%parts' then 'Inward' else
 (case when C.productline_name='-' then h.inloc_name else C.productline_name end) end 'Tem_Product Line',
   A.model'Model',d.part_name'Part Category',
k.cat_name as 'Defect Category'
,l.cat_name as 'Defect Sub Category'
,e.ref_name 'Specification Reference',

case when f.resp_name in ('Design-Electrical','Design','ATIO','Design Electrical - PLC','NPPO','NPPO-Quality') then 'Design'
when f.resp_name in ('Manufacturing Engineering','ME','ME-Layout') then 'ME'
when f.resp_name in ('Global sourcing','SCM','Global support centre (GSC)','Materials') then 'SCM'
when f.resp_name in ('ELGI VALVES - ASSEMBLY','Assembly') then 'Assembly'
when f.resp_name in ('ENGINE','Supplier') then 'Supplier'
when f.resp_name in ('Airend','TOP BLOCK','PVD','ELGi Foundry','Motor plant') then 'In-House'
when f.resp_name in ('Inward quality','Line Quality') then 'Quality'
when f.resp_name in ('Stores- Material fed to assembly') then 'Stores'

else f.resp_name end 'Responsible Function',

 f.resp_name 'Resp Name'

,g.criticality_name as 'Criticality',a.sup_name as 'Supplier Name',h.inloc_name'Inward Location'
,i.producttype_name'Product Type',isnull(a.target_date,a.created_at+45) as 'Target Date'
,(case when a.occurence='S' then 'Once' when a.occurence='C' then 'Chronic' end)'Occurence'
,isnull(r.Occurence,s.Occurence) as 'Occurence Count'

,(j.emp_name+' '+j.emp_initial+' ['+convert(nvarchar,j.emp_code)+']') 'Created By'
,isnull((t.emp_name+' '+t.emp_initial+' ['+convert(nvarchar, f.resp_name)+']'),'Not Assigned') 'Accepted By'

,(case 
					when defect_status='P' then 'Pending' 
					when defect_status='AS' then 'Assigning Phase' 
					when defect_status='CAPA' then 'CAPA'
					when defect_status='SUB' then 'Sub Task'
					when defect_status='A' then 'Approval'
					when defect_status='C' then 'Closed'
					when defect_status='W' then 'Waiting'
					when defect_status='CL' then 'Clarification'
					when defect_status='PS' then 'Subsidary Confirmation'
					when defect_status='RE' then 'Reopen'
					when defect_status='R' then 'Rejected'
					when defect_status='CA' then 'Containment Action'
					when defect_status='ESUB' then 'Employee Feedback - Assigning Phase'


					end) 'Defect Status'
	,(case when defect_status='A' then 'Waiting Approval'
					when defect_status='C' then 'Closed'
					when defect_status in ('P','AS','CAPA','SUB','W','CL','PS','R','RE') then 'Pending' 
					
					 end) 'Temp Defect Status'

,datediff(DAY,isnull(a.created_at,getdate()),isnull(a.closed_date,getdate())) as 'DefectAging',a.reopen_status 'Defect Reopen Status'
,isnull(m.resp_name,'')'Subtask 1 - Responsible Function',isnull(m.aging,'')'Subtask 1 - Aging'
,isnull(n.resp_name,'')'Subtask 2 - Responsible Function',isnull(n.aging,'')'Subtask 2 - Aging'
,isnull(o.resp_name,'')'Subtask 3 - Responsible Function',isnull(o.aging,'')'Subtask 3 - Aging'
,isnull(p.resp_name,'')'Subtask 4 - Responsible Function',isnull(p.aging,'')'Subtask 4 - Aging'
,isnull(q.resp_name,'')'Subtask 5 - Responsible Function',isnull(q.aging,'')'Subtask 5 - Aging'
,'' as 'Effectiveness Monitoring - Comments [CLOSED]'
,'' as 'Effectiveness Monitoring - Comments [OPEN]',a.created_at'CreatedDate',convert(nvarchar,month(a.created_at))+'-'+convert(nvarchar,year(a.created_at))as 'MnthYear'
,datediff(DAY,isnull(a.target_date,a.created_at+45),isnull(a.closed_date,getdate())) as 'TargetDefectAging',isnull(a.closed_date,getdate())'ClosedDate'
,a.defect_cat,a.defect_subcat,a.product_line,a.part_cat,a.sup_name,a.resp_func
from dms_master a
	left Join dms_source_master  b on b.source_id=a.source_id
	left Join dms_productline_master c on c.productline_id=a.product_line
	left join dms_part_category_master  D ON D.part_id=A.part_cat
	left join dms_spec_ref_master   e on e.ref_id=a.spec_ref
	left join (select distinct resp_id,resp_name from dms_resp_function_master) f on f.resp_id=a.resp_func
	left join dms_criticality_master  g on g.criticality_id=a.criticality
	left join dms_inwardlocation_master h on h.inloc_id = a.inward_loc
	left join dms_producttype_master i on i.producttype_id=a.product_type 
	left join [ELGi_Intranet].dbo.Employee_Detail j on j.emp_code=a.created_by
	left join dms_defect_category_master k on k.cat_id=a.defect_cat
	left join [dms_defect_category_master] l on l.cat_id=a.defect_subcat
	left join [ELGi_Intranet].dbo.Employee_Detail t on t.emp_code=a.accepted_by

	left join (select ROW_NUMBER() OVER(partition by defect_id order by defect_id)'Id',defect_id,b.resp_name,isnull(a.target_date,a.created_at+45)target_date
												,datediff(DAY,a.created_at,isnull(a.action_closed_date,getdate())) as'Aging' 
											from dms_subtask_capa_assigning a
											left join (select distinct resp_id,resp_name from dms_resp_function_master) b on b.resp_id=a.resp_func
											)m on m.defect_id=a.defect_id and m.Id=1

	left join (select ROW_NUMBER() OVER(partition by defect_id order by defect_id)'Id',defect_id,b.resp_name,isnull(a.target_date,a.created_at+45)target_date 
											,datediff(DAY,a.created_at,isnull(a.action_closed_date,getdate())) as'Aging' 
											from dms_subtask_capa_assigning a
											left join (select distinct resp_id,resp_name from dms_resp_function_master) b on b.resp_id=a.resp_func
											)n on n.defect_id=a.defect_id and n.Id=2

	left join (select ROW_NUMBER() OVER(partition by defect_id order by defect_id)'Id',defect_id,b.resp_name,isnull(a.target_date,a.created_at+45)target_date
											,datediff(DAY,a.created_at,isnull(a.action_closed_date,getdate())) as'Aging'  
											from dms_subtask_capa_assigning a
											left join (select distinct resp_id,resp_name from dms_resp_function_master) b on b.resp_id=a.resp_func
											)o on o.defect_id=a.defect_id and o.Id=3

	left join (select ROW_NUMBER() OVER(partition by defect_id order by defect_id)'Id',defect_id,b.resp_name,isnull(a.target_date,a.created_at+45)target_date 
											,datediff(DAY,a.created_at,isnull(a.action_closed_date,getdate())) as'Aging' 
												from dms_subtask_capa_assigning a
											left join (select distinct resp_id,resp_name from dms_resp_function_master) b on b.resp_id=a.resp_func
											)p on p.defect_id=a.defect_id and p.Id=4

	left join (select ROW_NUMBER() OVER(partition by defect_id order by defect_id)'Id',defect_id,b.resp_name,target_date 
											,datediff(DAY,a.created_at,isnull(a.action_closed_date,getdate())) as'Aging'  
											from dms_subtask_capa_assigning a
											left join (select distinct resp_id,resp_name from dms_resp_function_master) b on b.resp_id=a.resp_func
											)q on q.defect_id=a.defect_id and q.Id=5

left join (select defect_id,resp_func,
											row_number() over (partition by a.sup_name,defect_cat,defect_subcat,part_cat order by	defect_id)'Occurence'
											from dms_master a) r on r.defect_id=a.defect_id 
											and r.resp_func in (select distinct resp_id from dms_resp_function_master	where resp_name='Supplier')
											
left join (select defect_id,resp_func,
											row_number() over (partition by a.product_line,defect_cat,defect_subcat,part_cat order by defect_id)'Occurence'
											from dms_master a) s on s.defect_id=a.defect_id 
											and s.resp_func not in (select distinct resp_id from dms_resp_function_master where resp_name='Supplier')



											


where isnull(a.notacom_status,'')<>'NA' and a.status=1
											)a)a
																						

left join (select distinct b.resp_name,rca_id,supplier_name,part_category,defect_category,defect_sub_category,a.resp_func from dms_rca_creation a
			inner join dms_resp_function_master b on b.resp_id=a.resp_func 
			) G1 on G1.supplier_name=a.[Supplier Name] 
					and G1.part_category=a.part_cat and G1.defect_category=a.defect_cat
					and G1.defect_sub_category=a.defect_subcat and g1.resp_name='Supplier' and a.[Resp Name]='supplier' and g1.resp_func=a.resp_func


--left join (select distinct b.resp_name,rca_id,supplier_name,part_category,defect_category,defect_sub_category,a.resp_func 
--			from dms_rca_creation a
--			inner join dms_resp_function_master b on b.resp_id=a.resp_func 
--		  ) G2 on G1.supplier_name=a.[Supplier Name] 
--					and G1.part_category=a.part_cat and G1.defect_category=a.defect_cat
--					and G1.defect_sub_category=a.defect_subcat and g1.resp_name='Supplier' and a.[Resp Name]='supplier' and g1.resp_func=a.resp_func


					order by defect_id
											
end
