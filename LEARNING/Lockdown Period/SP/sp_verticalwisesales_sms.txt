-- exec sp_verticalwisesales_sms
CREATE proc [dbo].[sp_verticalwisesales_sms]
as
begin
truncate table verticalwisesales_sms

insert into verticalwisesales_sms(Sdate,Vertical_Name,Shortname,
Dval,Mval,Yval,ordr,smstype)
select sdate,vertical_name,shortname,dval,mval,yval,
ordr,1 as smstype
from (
select *,
case 
when Vertical_Name ='Compressor-Channel                 ' then 1
when Vertical_Name ='Compressor-Direct                  ' then 2
when Vertical_Name ='Compressor-RLY' then 3
when Vertical_Name ='Compressor-C&M' then 4
when Vertical_Name ='Compressor-WW' then 5
when Vertical_Name ='After Sales                        ' then 6
--when Vertical_Name ='Collections' then 7
end as ordr
 From (
select convert(date,getdate()-1) as sdate,description as Vertical_Name,
case 
when description='After Sales                        ' then 'AS'
when description='Compressor-Channel                 ' then 'Channel'
when description='Compressor-Direct                  ' then 'Direct'
when description='Compressor-RLY' then 'Rly'
when description='Compressor-C&M' then 'CnM'
when description='Compressor-WW' then 'WW'
else description
end as Shortname,
sum(ondate)Dval,
sum(mtd)Mval,
sum(ytd)Yval 
from (
select 
case 
when dpstname ='CONSTRUCTION (C)' or dpstname ='MINING (M)'then'Compressor-C&M'
WHEN dpstname ='WW'  or dpstname ='WW – OE'then 'Compressor-WW'
ELSE Description END AS Description
, dpstname,sum(ondate_sale) as ONDATE,SUM(this_month_sale)as MTD ,SUM(this_year_sale)AS YTD
from (
--select Dpstcode ,dpstname ,description,ondate_sale,this_month_sale,this_year_sale    From  vw_sls_Global
--where global ='International'
--union all
select Dpstcode ,dpstname ,description,ondate_sale,this_month_sale,this_year_sale    From  vw_sls_Domestic_DPSTwise 
)a
group by dpstname,description 
)a
group by Description 
)a
)a

insert into verticalwisesales_sms(Sdate,Vertical_Name,Shortname,
Dval,Mval,Yval,ordr,smstype)
select convert(date,getdate()-1),'Total','TOT', sum(dval),sum(mval),sum(yval),10 as ordr,1 as smstype from verticalwisesales_sms

and areacode not in ('623','624','631','635')
group by areacode,areaname
having sum(isnull(last_year_sale,0)+isnull(this_year_sale,0))<>0
)a
group by ordr,smstype,substring(AreaName,1,3)
