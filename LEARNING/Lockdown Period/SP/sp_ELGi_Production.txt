----Method 2
insert into ELGi_production_Stage1(company,warehouse,item,Ordernum,date1,Actual_qty)
select * from (
select a.COMPANY,a.CWAR,a.MITM,a.PDNO,ADLD as Closed_Date,
sum(case when ADLD='9999-01-01 00:00:00.000' then 0 else (case when a.OSTA in (7,8,9) then qrdr else 0 end) end) as ClosedQty
from  TTISFC001ALL a 
where a.PDNO not like '%[a-z]%'
and convert(date,ADLD) <= getdate()-1
and convert(date,RDLD) <= getdate()-1
and month(convert(date,ADLD))=month(getdate()-1)
and year(convert(date,ADLD))=year(getdate()-1)
and month(convert(date,RDLD))=month(getdate()-1)
and year(convert(date,RDLD))=year(getdate()-1)

group by a.COMPANY,a.CWAR,a.MITM,a.PDNO,ADLD
)a
where ClosedQty<>0

--Advance Closed Qty (Closed Date < Planed Date , Here the Planned date between Current date and End date of a month)
insert into ELGi_production_Stage1(company,warehouse,item,Ordernum,date1,Actual_qty_Advance)
select * from (
select a.COMPANY,a.CWAR,a.MITM,a.PDNO,ADLD as Closed_Date,
sum(case when ADLD='9999-01-01 00:00:00.000' then 0 else (case when a.OSTA in (7,8,9) then qrdr else 0 end) end) as ClosedQty
from  TTISFC001ALL a 
where a.PDNO not like '%[a-z]%'
and convert(date,ADLD) <= getdate()-1
and convert(date,RDLD) > getdate()-1
and month(convert(date,ADLD))=month(getdate()-1)
and year(convert(date,ADLD))=year(getdate()-1)
and month(convert(date,RDLD))=month(getdate()-1)
and year(convert(date,RDLD))=year(getdate()-1)
group by a.COMPANY,a.CWAR,a.MITM,a.PDNO,ADLD
)a
where ClosedQty<>0



--Backlog Closed Qty (Closed Date < Planed Date , Here the Planned date belongs to any of the previous months)
insert into ELGi_production_Stage1(company,warehouse,item,Ordernum,date1,Actual_qty_Backlog)
select * from (
select a.COMPANY,a.CWAR,a.MITM,a.PDNO,ADLD as Closed_Date,
sum(case when ADLD='9999-01-01 00:00:00.000' then 0 else (case when a.OSTA in (7,8,9) then qrdr else 0 end) end) as ClosedQty
from  TTISFC001ALL a 
where a.PDNO not like '%[a-z]%'
and convert(date,ADLD) <= getdate()-1
and month(convert(date,ADLD))=month(getdate()-1)
and year(convert(date,ADLD))=year(getdate()-1)
--and convert(date,RDLD) < getdate()-1
and month(convert(date,RDLD))<month(getdate()-1)
and year(convert(date,RDLD))=year(getdate()-1)
group by a.COMPANY,a.CWAR,a.MITM,a.PDNO,ADLD
)a
where ClosedQty<>0

--Alter table ELGi_production_Stage2 add Actual_qty_Backlog float
--Alter table ELGi_production_Stage3 add Actual_qty_Backlog float







--WIP (Till Yesterday)

insert into ELGi_production_Stage1(company,warehouse,item,Ordernum,Date1,WIP_qty)
select * from (
select a.COMPANY,a.CWAR,a.MITM,a.PDNO,
b.date as WIP_Date,
sum(case when ADLD='9999-01-01 00:00:00.000' then 0 else (case when a.OSTA IN(4,6) then qrdr else 0 end) end) as WIP_qty
from  TTISFC001ALL a 
left outer join
(
select * from (
select PDNO,OSTA,convert(date,date) as date,row_number() over (partition by pdno order by date) dateRank 
from TTICST151ALL
where OSTA in (4,6)
--and PDNO='230009816' --for testing Purpose
)b1
where dateRank=1
)b
on a.PDNO=b.PDNO

where a.PDNO not like '%[a-z]%'
and b.date< getdate()-1
group by a.COMPANY,a.CWAR,a.MITM,a.PDNO,b.date
)a
where WIP_qty<>0


--Plan Pulling (greater than today)

insert into ELGi_production_Stage1(company,warehouse,item,Ordernum,Date1,PlanPullQty)
select * from (
select a.COMPANY,a.CWAR,a.MITM,a.PDNO,RDLD as Planned_Date,
sum(case when ADLD='9999-01-01 00:00:00.000' then 0 else (case when a.OSTA in (4,6) then qrdr else 0 end) end) as WIP_qty
from  TTISFC001ALL a 
where a.PDNO not like '%[a-z]%'
and convert(date,RDLD) > getdate()-1
group by a.COMPANY,a.CWAR,a.MITM,a.PDNO,RDLD
)a
where WIP_qty<>0

--Plan Pulling (Closed Qty greater than today is covered under Plan Pulling)
insert into ELGi_production_Stage1(company,warehouse,item,Ordernum,Date1,PlanPullQty)
select * from (
select a.COMPANY,a.CWAR,a.MITM,a.PDNO,RDLD as Planned_Date,
sum(case when ADLD='9999-01-01 00:00:00.000' then 0 else (case when a.OSTA in (8,9) then qrdr else 0 end) end) as ClosedQty
from  TTISFC001ALL a 
where a.PDNO not like '%[a-z]%'
and convert(date,RDLD) > getdate()-1
group by a.COMPANY,a.CWAR,a.MITM,a.PDNO,RDLD
)a
where ClosedQty<>0



--Backlog (Till yesterday)

insert into ELGi_production_Stage1(company,warehouse,item,Ordernum,date1,Backlog_Qty)
select * from (

from TTISFC001ALL a 
where a.PDNO not like '%[a-z]%'
and convert(date,RDLD) <getdate()-1
group by a.COMPANY,a.CWAR,a.MITM,a.PDNO,RDLD
)a
where Planned_qty>0


update ELGi_production_Stage1 set Released_qty=isnull(Actual_qty,0) + isnull(Actual_qty_Advance,0) +
isnull(Actual_qty_Backlog,0) +isnull(WIP_qty,0)
update ELGi_production_Stage1 set Planned_qty=isnull(Released_qty,0) +isnull(Backlog_Qty,0) -isnull(PlanPullQty,0)

group by COMPANY,CWAR,ITEM

and ELGi_production_Stage3.warehouse=a.CWAR
