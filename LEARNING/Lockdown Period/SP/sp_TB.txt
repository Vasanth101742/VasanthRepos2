CREATE proc [dbo].[sp_TB] as
Begin

Truncate table TB_details_Temp
Truncate table TB_details
Truncate table TB_Temp1
Truncate table TB_Temp2
Truncate table TB_Temp3

--sp_rename 'TB_details.Budget','YTD_Budget'
--sp_rename 'TB_details.BudPlan','YTD_BudPlan'

--sp_rename 'TB_details.Budget_BudRate','YTD_Budget_BudRate'
--sp_rename 'TB_details.BudPlan_PlanRate','YTD_BudPlan_PlanRate'

--Alter table TB_details add FTM_Budget float
--Alter table TB_details add FTM_BudPlan float
--Alter table TB_details add FTM_Budget_BudRate float
--Alter table TB_details add FTM_BudPlan_PlanRate float


--Opening Balance
insert into TB_details_Temp(FinPeriod,FinYear,FinMonth,COMPANY,CONO,LEAC,FOBH1,NOBH1)
select concat(convert(varchar,[YEAR]),'01') as Period,YEAR,'1' as FinMonth,COMPANY,CONO,LEAC,
sum(convert(float,FOBH$1)) FOBH1,sum(convert(float,NOBH$1)) NOBH1 from TTFGLD203400
--where LEAC = 'AA0103002'
group by COMPANY,CONO,YEAR,LEAC


--FTM Debits and Credits
Insert into TB_details_Temp(FinPeriod,FinYear,FinMonth,COMPANY,CONO,LEAC,Debit,Credit)
--select * into TB_details from (
select concat(convert(varchar,[YEAR]),RIGHT('00'+CAST(PRNO AS VARCHAR(2)),2)) as Period,
[YEAR] as FinYear,PRNO as FinMonth,
COMPANY,CONO,LEAC,
sum(convert(float,FDAH$1)) Debit,
sum(convert(float,FCAH$1)) Credit
from TTFGLD201400
where PTYP=1
--and LEAC = 'AA0103002'
--and YEAR=2019
--and PRNO<=11
group by COMPANY,CONO,[YEAR],PRNO,LEAC
--)a



update TB_details_Temp set LEAC_DESC=a.T$DESC,SUBL=a.T$SUBL
from (select distinct T$LEAC,T$DESC,T$SUBL from TTFGLD008ALL)a
where TB_details_Temp.LEAC=a.T$LEAC

delete from TB_details_Temp where SUBL<>0



insert into TB_details(FinPeriod,FinYear,FinMonth,COMPANY,CONO,LEAC,FOBH1,NOBH1,Debit,Credit,LEAC_DESC,Entity) 
select FinPeriod,FinYear,FinMonth,COMPANY,CONO,LEAC,FOBH1,NOBH1,Debit,Credit,LEAC_DESC,Entity from (
select FinPeriod,FinYear,FinMonth,COMPANY,CONO,LEAC,
sum(isnull(FOBH1,0)) FOBH1,sum(isnull(NOBH1,0)) NOBH1,sum(isnull(Debit,0)) Debit,sum(isnull(Credit,0)) Credit,LEAC_DESC,Entity
from(
select FinPeriod,FinYear,FinMonth,COMPANY,CONO,LEAC,FOBH1,NOBH1,Debit,Credit,LEAC_DESC,Entity
--,row_number() over(partition by cono,leac,finperiod order by cono,leac,finperiod ) Rnk 
from TB_details_Temp
--where FinPeriod='202001'
--and leac='AA0102002'
--and cono=400
)a
group by FinPeriod,FinYear,FinMonth,COMPANY,CONO,LEAC,LEAC_DESC,Entity
)b


update TB_details  set Entity=a.ENTY,Currency=a.CURR
from TZZGLD809400 a
where TB_details.CONO=a.COMP 


update TB_details set CONO=a.CMPN
from (select distinct CPNM,CMPN from TZZGLD806400)a
where TB_details.Entity=a.CPNM
and TB_details.CONO is null

update TB_details  set Entity=a.Entity,Currency=a.Currency
from (select distinct CONO,Entity,Currency from TB_details where CONO like '4%' and Entity is not null)a
where TB_details.CONO like '4%'




---YTD Value Calculation considering the opening Balance-------------

Declare @FinYear int
Declare @Count1 int
Declare @FinYearCnt int

set @FinYearCnt=(select count(distinct FinYear) from TB_details)

set @Count1=1
while (@Count1<=@FinYearCnt)
begin

Set @FinYear =(select FinYear from (
select FinYear,row_number() over (order by FinYear) as FinYearRnk from
(select distinct FinYear from TB_details)a)b
where FinYearRnk=@Count1)

--Alter table TB_Temp1 add Entity varchar(100)

insert into TB_Temp1(Entity,CONO,LEAC,Debit,Credit,FinYear,FinPeriod,FTM_Value,YTD_Value)
--select * into TB_Temp1 from ( 
select Entity,CONO,LEAC,Debit,Credit,FinYear,FinPeriod,
sum(FTM_Value) FTM_Value,
sum(FTM_Value) over (partition by Entity,cono,LEAC,FinYear order by Entity,CONO,LEAC,FinYear,FinPeriod) as YTD_Value
from (
select Entity,CONO,LEAC,FinYear,FinPeriod,
sum(Debit) Debit,
sum(Credit) Credit,
sum(isnull(Debit,0)-isnull(Credit,0)) FTM_Value
from (
--Opening balance
select Entity,CONO,LEAC,FOBH1 as Debit,0 as Credit,FinYear,FinPeriod
from TB_details
where LEAC  is not null 
and FOBH1 is not null
and FinYear=@FinYear
--and leac='AA0102001'
--order by CONO,LEAC,FinPeriod

union
--FTM Debits and Credits
select Entity,CONO,LEAC,Debit,Credit,FinYear,FinPeriod
from TB_details
where LEAC  is not null 
and isnull(Debit,0)+isnull(Credit,0)<>0
and FinYear=@FinYear
--and leac='AA0102001'
)a
group by Entity,CONO,LEAC,FinYear,FinPeriod
--order by CONO,LEAC,FinPeriod
)b
group by Entity,CONO,LEAC,Debit,Credit,FinYear,FinPeriod,FTM_Value
order by Entity,CONO,LEAC,FinPeriod
--)c

Set @Count1=@Count1+1

end


---End: YTD value Calculation----------------------------------------

update TB_details set FTM_Value=a.FTM_Value,YTD_Value=a.YTD_Value
from (select * from TB_Temp1)a
where TB_details.CONO=a.CONO
and TB_details.LEAC=a.LEAC
and TB_details.FinPeriod=a.FinPeriod


--Adjustment Entries from LN based on Entity ,Ledger and Ledger Description

Insert into TB_details(CalPeriod,CalYear,CalMonth,Entity,CONO,LEAC,LEAC_DESC,YTD_Value)
select concat(convert(varchar,[FYER]),RIGHT('00'+CAST(FMNT AS VARCHAR(2)),2)) as Period,
FYER,FMNT,ENTY,'Adj',LEDG,LEDE,sum(AMNT) AMNT
from TZZGLD804400
where type='Actuals'
and ENTY is not null
and AMNT<>0
group by FYER,FMNT,ENTY,LEDG,LEDE
order by Period
--and LEDG like '%Sales%' --for testing


--Non LN Entities

--Test case for more than one record

--select enty,FYER,LEDG,LEDE,FMNT,count(*) from TZZGLD803400

--order by enty,FYER,LEDG,LEDE,FMNT

--select * from TZZGLD803400 where
--enty='Michigan Air' and fyer='2020' and fmnt='11' and ledg='5410'

Insert into TB_details(CalPeriod,CalYear,CalMonth,Entity,CONO,LEAC,LEAC_DESC,YTD_Value)
select concat(convert(varchar,[FYER]),RIGHT('00'+CAST(FMNT AS VARCHAR(2)),2)) as Period,
FYER,FMNT,ENTY,'NonLN',LEDG,LEDE,sum(AMNT) AMNT
from TZZGLD803400
where 
ENTY is not null
and AMNT<>0
group by [FYER],FMNT,ENTY,LEDG,LEDE
order by Period




Insert into TB_details(Entity,PRD1,PRD2,PRD3,CONO,CalPeriod,CalYear,CalMonth,YTD_Budget)

concat(convert(varchar,FYER),RIGHT('00'+CAST(FMNT AS VARCHAR(2)),2)) as calPeriod,FYER,FMNT,sum(AMNT) AMNT
from TZZGLD805400
where [TYPE]='Budget'
group by ENTY,MR11,MR22,MR33,FYER,FMNT


Insert into TB_details(Entity,PRD1,PRD2,PRD3,CONO,CalPeriod,CalYear,CalMonth,YTD_BudPlan)
select ENTY,MR11,MR22,MR33,'Plan',
concat(convert(varchar,FYER),RIGHT('00'+CAST(FMNT AS VARCHAR(2)),2)) as CalPeriod,FYER,FMNT,sum(AMNT) AMNT 
from TZZGLD805400
where [TYPE]='Covid19 Plan'
group by ENTY,MR11,MR22,MR33,FYER,FMNT

update TB_details set CalMonth=FinMonth-9 where FinMonth>9 and CalMonth is null

update TB_details set CalYear=FinYear where FinMonth<=9 and CalYear is null

update TB_details set FinYear=CalYear where CalMonth>=4 and FinYear is null


update TB_details set FinMonth=CalMonth+9 where CalMonth<4 and FinMonth is null
update TB_details set FinMonth=CalMonth-3 where CalMonth>=4 and FinMonth is null

update TB_details set FinPeriod=concat(convert(varchar,FinYear),RIGHT('00'+CAST(FinMonth AS VARCHAR(2)),2)) where FinPeriod is null

--where mrcn='BCAS001'
)a

update TB_details  set BudRate=1 where BudRate=''
update TB_details  set PlanRate=1 where PlanRate=''

update TB_details  set FTM_Value_INR=isnull(FTM_Value,0)*CurRate_PL where Category='PL'
update TB_details  set FTM_Value_INR=isnull(FTM_Value,0) where category is null
update TB_details  set FTM_Value_INR=isnull(FTM_Value,0) where category=''


update TB_details  set YTD_Value_INR=isnull(YTD_Value,0)*CurRate_BS where Category='BS'

CurRate_BS_LYCM=a.CurRate_BS,
CurRate_PL_LYCM=a.CurRate_PL,
Debit_LYCM=a.Debit,

FTM_Value_LYCM=a.FTM_Value,
FTM_Value_LYCM_INR=a.FTM_Value_INR,
YTD_Value_LYCM=a.YTD_Value,
YTD_Value_LYCM_INR=a.YTD_Value_INR,
FTM_Value_LYCM_BudRate=a.FTM_Value_BudRate,
YTD_Value_LYCM_BudRate=a.YTD_Value_BudRate
from (select CONO,LEAC,LEAC_DESC,CalYear+1 as CalYear,CalMonth,CurRate_BS,CurRate_PL,
FTM_Value,FTM_Value_INR,
YTD_Value,YTD_Value_INR,Debit,Credit,

from TB_details)a

and TB_details.LEAC=a.LEAC
and TB_details.LEAC_DESC=a.LEAC_DESC
and TB_details.CalYear=a.CalYear
and TB_details.CalMonth=a.CalMonth
--and concat(TB_details.CONO,TB_details.LEAC)='500EC2101001'

update TB_details  set FTM_Value_PrevYrRate=isnull(FTM_Value,0) where category=''


update TB_details  set YTD_Value_PrevYrRate=isnull(YTD_Value,0)*isnull(CurRate_BS_LYCM,1) where Category='BS'
update TB_details  set YTD_Value_PrevYrRate=isnull(YTD_Value,0)*isnull(CurRate_PL_LYCM,1) where Category='PL'
update TB_details  set YTD_Value_PrevYrRate=isnull(YTD_Value,0) where category is null
update TB_details  set YTD_Value_PrevYrRate=isnull(YTD_Value,0) where category=''


update TB_details  set FTM_Value_LYCM_BudRate=isnull(FTM_Value_LYCM,0)*isnull(BudRate,1)
update TB_details  set YTD_Value_LYCM_BudRate=isnull(YTD_Value_LYCM,0)*isnull(BudRate,1)
