update Inventory set TRDT=b.PurchaseDt
from (
select * from (
select a.Company,a.Warehouse,a.item,a.[Received Quantity],
b.QSKT,a.[Total Amount  (IND)] GivenAmt,
b.Amnt3,a.[Date of Purchase] as PurchaseDt,b.TRDT,
case when a.[Received Quantity]-b.QSKT<>0 then b.Amnt3 + (a.[Received Quantity]-b.QSKT)*b.Amnt3
when a.[Received Quantity]-b.QSKT=0  and a.[Total Amount  (IND)]-b.Amnt3 between -0.1 and 0.1 then b.Amnt3 
end as DerivedAmt

from 
(select * from inv_465_manualdata) a
left outer join 
(select * from Inventory where company=465 and CWAR='MNT') b
on a.Company=b.COMPANY
and a.Warehouse=b.CWAR
and ltrim(rtrim(a.Item))=ltrim(rtrim(b.ITEM))
--where ltrim(rtrim(a.ITEM))='MNT-ELE-FOU-0006' --for testing purpose
)a
where GivenAmt -DerivedAmt between -0.1 and 0.1
)b
where Inventory.COMPANY=b.COMPANY
and Inventory.CWAR=b.Warehouse
and ltrim(rtrim(Inventory.ITEM))=ltrim(rtrim(b.item))
and Inventory.Amnt3=b.Amnt3

update Inventory_209_210WH_Issue_Manipulation set IssueDate=b.DateB
from (
select COMPANY,WarA,ITEM,max(DateB) DateB from (
select a.COMPANY,a.CWAR as WarA,a.ITEM,DateA,b.CWAR as WarB,DateB,datediff(dd,dateb,getdate()-1) Diff from 
(select distinct COMPANY,CWAR,item,IssueDate as DateA from Inventory
where ltrim(rtrim(ITEM))in (select distinct ltrim(rtrim(ITEM)) from Inventory2 where CWAR in ('209','210'))
and company =401 
and CWAR in ('209','210')
--AND LTRIM(RTRIM(ITEM))='B005106400028'
)a
left outer join
(
select * from
(select distinct COMPANY,CWAR,item,IssueDate as DateB,
row_number() over (partition by COMPANY,ITEM order by IssueDate desc) as Issuedateorder
from (select distinct COMPANY, T$CWAR CWAR,T$ITEM  item, convert(date,T$CTDT) IssueDate from TWHINA114ALL 
where company in (select distinct company from ELGI_Companies_Master where CompanyGroup='ELGI')
and T$KOOR in (1,2,3,32,42,46)
and company =401 
--AND LTRIM(RTRIM(T$ITEM))='B005106400028'
)a1
)a2
where Issuedateorder=1
)b
on a.COMPANY=b.COMPANY
and ltrim(rtrim(a.ITEM))=ltrim(rtrim(b.ITEM))
where datediff(dd,dateb,getdate()-1)<=180
)a
group by COMPANY,WarA,ITEM

and Inventory_209_210WH_Issue_Manipulation.CWAR=b.WarA
and Inventory_209_210WH_Issue_Manipulation.ITEM=b.ITEM
