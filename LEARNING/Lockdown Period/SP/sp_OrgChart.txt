CREATE PROCEDURE [dbo].[sp_OrgChart]

AS

Begin

DECLARE @emp_code numeric(10),
@Name char(50),
@lvl integer,
@level integer,
@target numeric(10),
@strq varchar(500),
@rnum int

SET NOCOUNT ON

CREATE TABLE #GROUP (emp_code numeric(10), empname Char(50), glevel int)
CREATE TABLE #FTABLE (emp_code numeric(10), empname Char(50), glevel int,ordid int)
Create table #Stack (emp_code numeric(10), empname char(50), glevel int)

INSERT #GROUP 
   SELECT emp_code,empname, 1  FROM emp_appr WHERE isemp_code=0 and emp_code =10084 ORDER BY emp_code DESC
WHILE EXISTS(SELECT * FROM #GROUP)
BEGIN
   SELECT @lvl = 0
   SELECT @emp_Code= emp_code, @Name=empname FROM #Group 
   DELETE FROM #Group WHERE emp_code=@emp_Code	

   INSERT INTO #STACK VALUES(@emp_Code,@Name,1)
   SELECT @LEVEL = 1
			select @rnum=1
   WHILE @LEVEL > 0
   BEGIN
     IF EXISTS ( SELECT * FROM #STACK WHERE GLEVEL=@LEVEL )
     BEGIN
       SELECT @emp_Code=emp_code, @Name=empname FROM #STACK WHERE Glevel = @Level

						select @rnum=@rnum+1
       INSERT INTO #FTABLE VALUES ( @emp_Code,@Name,@Level,@rnum)
       DELETE FROM #STACK WHERE Glevel = @Level AND emp_code = @emp_Code

       INSERT #STACK
        SELECT emp_code, empname, @Level + 1
        FROM emp_appr WHERE employee_status = 'Live' and  isemp_code = @emp_Code ORDER BY emp_code DESC
       IF @@ROWCOUNT > 0 SELECT @Level = @Level + 1
     END

    ELSE SELECT @Level = @Level - 1
   END

   --


truncate table travel_emp

insert into travel_emp(emp_code,emp_name,glevel,ordid) SELECT  *FROM #FTABLE 


update travel_emp set spaceid=6 where glevel=2
update travel_emp set spaceid=10 where glevel=3
update travel_emp set spaceid=14 where glevel=4
update travel_emp set spaceid=18 where glevel=5
update travel_emp set spaceid=22 where glevel=6
update travel_emp set spaceid=26 where glevel=7
update travel_emp set spaceid=30 where glevel=8

update Travel_emp set reportid = isemp_code   from  emp_appr   where Travel_emp.emp_code = emp_appr.emp_code 



end


end
