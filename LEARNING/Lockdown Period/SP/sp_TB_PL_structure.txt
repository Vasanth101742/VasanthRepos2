declare @FinPeriod varchar(20)
declare @Entity  varchar(100)
declare @Count1 int
declare @Count2 int
declare @TotalPeriodCounts int
declare @TotalEntityCounts int

set @TotalPeriodCounts=(select count(distinct finperiod) from TB_details)

set @Count1=1
while (@Count1<=@TotalPeriodCounts)
begin

set @FinPeriod= (select FinPeriod from (
select finperiod,row_number() over (order by finperiod) as rownum from 
(select distinct finperiod from TB_details)a)b
where rownum=@Count1)

set @TotalEntityCounts=(select count(distinct Entity) from TB_details where finperiod=@FinPeriod)

set @Count2=1
while(@Count2<=@TotalEntityCounts)
begin



set @Entity= (select Entity from (
select Entity,FinPeriod,row_number() over (order by Entity) as rownum from 
(select distinct Entity,FinPeriod from TB_details where finperiod=@FinPeriod)a)b
where rownum=@Count2)

insert into TB_PL_structure(Entity,BSPL,MR1,MR1_ID,DataType,Calculation,Finperiod)
select @Entity,BSPL,MR1,MR1_ID,DataType,Calculation,@FinPeriod from TB_PL_structure where finperiod is null

end
