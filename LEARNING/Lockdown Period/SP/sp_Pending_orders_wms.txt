CREATE proc [dbo].[sp_Pending_orders_wms] 
as
Begin



truncate table Cancell_orders_wms
insert into Cancell_orders_wms
select distinct concat(ORNO,convert(decimal(19,0),PONO)) as Cancell_orders From [ELGi_wms].dbo.TTDSLS401490  where CLYN =1


Truncate table [ELGi_wms].dbo.Pending_orders
Truncate table  Pending_orders_wms
insert into Pending_orders_wms


--alter table Pending_orders_wms add OSET int
--alter table Pending_orders_wms ADD wh_orderdate datetime

select *,0 as OSET,'' wh_orderdate,null as Next_Due_Date1,null as Next_Due_Date2,null as Reason1,null as Reason2,null as supplier from (
select Company,CustCode,ItemCode,OrderDate,OrderNum,PositionNum,ord_qty,ShipQty,DelQty,PendQty,TaxCode,DeliveryDate,WareHouse,Price,DiscPerc1,RATS$1,OAMT,Value,diffdays,                   
null as Item_desc,null as Custname,null as Dpstname,null as areacode,null as areaname,
null as Dpst,null as ordertype,null as Buyer ,null as origin,
--case when h.Origin=3 then 'EDI' else 'Manual' End as orjn,
null as orjn ,null as Indent,
0 AS STOCK_440_act,0 AS STOCK_440_onhand,0 AS STOCK_401_act,0 AS STOCK_401_onhand,0 AS STOCK_416_act,0 AS STOCK_416_onhand,
0 AS STOCK_450_act,0 AS STOCK_450_onhand,0 AS STOCK_490_act,0 AS STOCK_490_onhand,0 as STOCK_DEL_act,0 as STOCK_DEL_onhand,
0 AS STOCK_CAL_act,0 AS STOCK_CAL_onhand,
null as Baan_orderno,null as Baan_positionno,0 as Cancel_Qty,null as Zone,null as OrderMonthNo,null as OrderMonth,null as OrderMonthYear,
NULL AS Origin_desc,NULL AS Region,NULL AS Sale_Type, NULL AS Tax_Desc ,NULL AS HSN_CODE ,NULL AS GSTN_NO ,NULL AS PRICE_ORIGIN ,NULL AS SP_SEQ_NO,
NULL AS Ref,NULL AS Old_orderno,NULL AS Old_positionno,null as State,
NULL as Sales_Inv_Address,NULL as Cust_Inv_Address,
NULL as Sales_GSTN_NO,NULL as Cust_GSTN_NO,
NULL as SALES_SEQ,NULL as Cust_SEQ,
NULL as SALES_state,NULL as Cust_state,
NULL as plan_del_Date,
DiscPerc2,'' as disc_origin,'' as Blocked ,null as currecny,null as plan_recp_date,
0 as MaxQty,0 as Monthwise_Ordqty,
'' as ExportOrder,null as invoice,null as inv_date, null as dest_type, null as issue_item,
'      ' as itemtype,'     ' as itemclass,0 AS ISSS , '  ' as estimated_date,0 as lead_days,0 AS NATIONAL_HOLIDAYS,
'TYP'as typ,'     ' as shpment_date,0 as Delay_days,'' as trantype,'   'as Mitachi,'   ' as Hanva,
'                  ' as ShipmentNo,'         'packingslip_comp_Date ,0 AS STOCK_490_blocked


From   
	
(
select a.Company,a.CustCode,a.ItemCode,a.OrderDate,a.OrderNum,a.PositionNum,
avg(a.ord_qty) as ord_qty,
--TranType,--b.ShipNo,--c.INVNUM,
sum(ShipQty) ShipQty,
sum(DelQty) DelQty,
avg(a.ord_qty)-sum(isnull(DelQty,0)) as PendQty,
TaxCode,case when DeliveryDate='1970-01-01 00:00:00.000' then null else DeliveryDate end as DeliveryDate ,WareHouse,Price,DiscPerc1,DiscPerc2,RATS$1,OAMT,case when RATS$1<1 then 1*OAMT else RATS$1*OAMT end  as Value,
DATEDIFF(dd,OrderDate,getdate())as diffdays 
 from 
(
--Detailed Data
select a.Company,CustCode,ItemCode,a.OrderDate,a.OrderNum,a.PositionNum,a.ord_qty,TaxCode,DeliveryDate,WareHouse,Price,DiscPerc1,DiscPerc2,
Value,
DATEDIFF(dd,OrderDate,getdate())as diffdays,RATS$1,OAMT, 
--c.TranType,b.ShipNo,--c.INVNUM,
ISNULL(b.ShipQty,0)ShipQty,
ISNULL(c.DelQty,0)DelQty From 
(
--Fetching OrderQty
select Company,odat as OrderDate,ofbp as CustCode,ltrim(rtrim(item)) as ItemCode,orno as OrderNum ,pono as PositionNum,qoor as ord_qty, invn as invnum,
CVAT as TaxCode,DLDT as DeliveryDate, CWAR as WareHouse,pric as Price,DISC$1 as DiscPerc1,DISC$2 as DiscPerc2,case when RATS$1 <1 then 1*OAMT else RATS$1*OAMT end  as Value,
DATEDIFF(dd,odat,getdate())as diffdays,RATS$1,OAMT 
From [ELGi_wms].dbo.TTDSLS401490 
where sqnb=0 
--and clyn<>1 
and CONCAT(ORNO,PONO) not in (select * From Cancell_orders_wms)
and QOOR >0

)a

left outer join
(
--Fetching Packing Qty below comment data old table
select COMPANY,orno as OrderNum , pono as PosNum,sum(qpli)as ShipQty  from  [ELGi_wms].dbo.TWHINH480490
where qpli>0 and COMPANY in (490) group by COMPANY,orno ,pono

)b
on a.Company=b.COMPANY
and a.OrderNum=b.OrderNum
and a.PositionNum=b.PosNum

left outer join 
(
--Fetching Deliverd Qty
select COMPANY,ORNO,PONO,sum(QIDL) as DelQty from [ELGi_wms].dbo.TTDSLS406490 
where QIDL>0 

group by COMPANY,ORNO,PONO
)c
on a.Company=c.COMPANY
and a.OrderNum=c.ORNO
and a.PositionNum=c.PONO


)a

group by a.Company,a.CustCode,a.ItemCode,a.OrderDate,a.OrderNum,a.PositionNum,TaxCode,DeliveryDate,WareHouse,Price,DiscPerc1,DiscPerc2,RATS$1,OAMT

)Details
)a

update pending_orders_wms set T$OSET=a.oset,wh_orderdate=a.ODAT  from(
select a.ORNO ,b.PONO,b.OSET,a.ODAT  From (select convert(decimal(19,0),ORNO)ORNO,convert(decimal(19,0),OSET)OSET,ODAT from [ELGi_wms].dbo.TWHINH200490 where OORG =1)a
left join (select convert(decimal(19,0),ORNO)ORNO,convert(decimal(19,0),PONO)PONO,convert(decimal(19,0),OSET)OSET from [ELGi_wms].dbo.TWHINH220490  where OORG =1)b
on a.ORNO =b.ORNO 
and a.OSET =b.OSET)a
where pending_orders_wms.OrderNum =a.orno
and pending_orders_wms.PositionNum =a.PONO


------------ foc orders removed as req to karun kannan
delete from Pending_orders_wms  where CONCAT(company,ordertype) in
(select distinct CONCAT(company,saleordertype)as type  From saletype_master where Sales_Category =1)
---------------------

update Pending_orders_wms set ShipQty =(ShipQty-DelQty)

update Pending_orders_wms set plan_del_date=a.t_ddta,ISSS=A.InvIssStat  ,plan_recp_date=a.planrecdate
from Sales_Lines a 
where 
a.company=490 and 
Pending_orders_wms.Company =a.Company
and Pending_orders_wms.OrderNum =a.OrderNum
and Pending_orders_wms.PositionNum =a.PositionNum 


update Pending_orders_wms set 
Dpst=h.Dpst,
ordertype=H.ordertype,
origin=h.origin,
orjn=case when h.Origin=3 then 'EDI' else 'Manual' End ,
Indent=case when h.t_corn=' ' then h.RefA else h.t_corn end
from 
(select * From Sales_Header where dpst not like '%[a-z]%' and Company =490 ) h
where Pending_orders_wms.OrderNum =h.OrderNum 
and Pending_orders_wms.Company =h.Company 


-------Buyer AND SUPPLIER Updation

	update Pending_orders_wms  set Buyer= a.BUYER,supplier=a.SUPPLIER  from (
	SELECT  A.ITEM ,B.t_nama  AS SUPPLIER ,C.NAMA AS BUYER fROM
	(SELECT DISTINCT LTRIM(RTRIM(ITEM))AS ITEM,OTBP,BUYR   fROM [ELGi_LN].DBO.TTDIPU001ALL WHERE COMPANY =490)A
LEFT JOIN [ELGi_LN].DBO.TTCCOM100400 B
ON A.OTBP =B.t_bpid 
LEFT JOIN (select  EMNO, NAMA,COMPANY  From [ELGi_LN].DBO.TTCCOM001ALL where company=400 and nama is not null and NAMA <>'')C
ON A.BUYR =C.EMNO)a
where LTRIM(RTRIM(Pending_orders_wms.ItemCode)) =LTRIM(RTRIM(a.ITEM)) 
AND Pending_orders_wms.Company =490

------------------ buyer updation end

from (select * From tsacom999all where type=2)b
where ltrim(rtrim(OrderNum)) =ltrim(rtrim(b.NORN ))
and ltrim(rtrim(PositionNum)) =ltrim(rtrim(b.NPOS))


update Pending_orders_wms set Cancel_Qty=a.qty 
from (select company,OrderNum,PositionNum,t_clyn,sum(t_qoor)as qty 
From Sales_Lines 
where t_clyn=1 and t_qoor>0  and company=490

)a
where a.Company =Pending_orders_wms.Company 
and a.OrderNum =Pending_orders_wms.OrderNum 
and a.PositionNum =Pending_orders_wms.PositionNum

where Pending_orders_wms.custcode=a.custcode

from TTCIBD001490 a
where Pending_orders_wms.company =490
and ltrim(rtrim(Pending_orders_wms.ItemCode))=ltrim(rtrim(a.item))

from TTCIBD001400 a
where Pending_orders_wms.company like '4%' and item_desc is null
and ltrim(rtrim(Pending_orders_wms.ItemCode))=ltrim(rtrim(a.t_item))

update Pending_orders_wms set  Origin_desc= (case when origin=3 then 'Dealer Portal' when origin<>3 then 'Manual'
when convert(date,OrderDate)='2017-07-03' then 'Converted' end)
 
update Pending_orders_wms set Region=(case when substring(OrderNum,1,2)='29' then 'Exports' when substring(OrderNum,1,2)<>'29' then 'Domestic' end)


update Pending_orders_wms set sale_type='NON FOC'
update Pending_orders_wms set sale_type='FOC'
WHERE CONCAT(COMPANY,ordertype) IN (SELECT CONCAT(COMPANY,SALEORDERTYPE) fROM saletype_master WHERE Sales_Category IN (1) )

Item,t_corn,t_hcod_l,t_rnsb_l  from Sales_Lines where company=490 )b
where Pending_orders_wms.Company =b.Company
and Pending_orders_wms.OrderNum=b.OrderNum 
and Pending_orders_wms.PositionNum =b.PositionNum 
and ltrim(rtrim(Pending_orders_wms.ItemCode)) =ltrim(rtrim(b.Item)) 


update Pending_orders_wms set PRICE_ORIGIN=
case when  b.t_porg =4 then 'Manual' ELSE 'System' end
 from (select distinct company,OrderNum,PositionNum,ltrim(rtrim(ItemCode))as Item,t_porg from Sales_Lines where company=490 )b
where Pending_orders_wms.Company =b.Company
and Pending_orders_wms.OrderNum=b.OrderNum 
and Pending_orders_wms.PositionNum =b.PositionNum 
and ltrim(rtrim(Pending_orders_wms.ItemCode)) =ltrim(rtrim(b.Item)) 

update Pending_orders_wms set GSTN_NO=b.t$FOVN  from (
select distinct A.company,A.OrderNum,A.PositionNum,ltrim(rtrim(A.ItemCode))as Item,A.t_rnsb_l,B.T$FOVN from Sales_Lines A

and a.CustCode =b.T$BPID 
and a.company=490
)b
where Pending_orders_wms.Company =b.Company
and Pending_orders_wms.OrderNum=b.OrderNum 
and Pending_orders_wms.PositionNum =b.PositionNum 
and ltrim(rtrim(Pending_orders_wms.ItemCode)) =ltrim(rtrim(b.Item))

 from (SELECT DISTINCT T$FOVN,T$SEQN$L,T$BPID,T$CADR$L FROM  TTCTAX400400 where T$CATG$L =9)b
	where Pending_orders_wms.CustCode =b.T$BPID
	and Pending_orders_wms.cust_Inv_Address=b.T$CADR$L

 from (SELECT DISTINCT T$FOVN,T$SEQN$L,T$BPID,T$CADR$L FROM  TTCTAX400400 where T$CATG$L =9)b
where Pending_orders_wms.CustCode =b.T$BPID
	and Pending_orders_wms.Sales_Inv_Address =b.T$CADR$L

	from (select distinct cadr,cste  From TTCCOM130400)b
	where Pending_orders_wms.Sales_Inv_Address =b.CADR
