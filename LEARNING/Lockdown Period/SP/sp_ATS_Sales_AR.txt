left join (select distinct CADR,NAMA,NAMB,NAMC  from ttccom130400 )b
on ltrim(rtrim(a.t_itad))=ltrim(rtrim(b.CADR)) 
)b
where ATS_Sales_AR.company =b.Company 
and ATS_Sales_AR.OrderNo =b.OrderNum 


update ATS_Sales_AR set FRT=a.Sales_amount+t_amth_2
from FRT_Sales a
where ATS_Sales_AR.company=a.Company
and ATS_Sales_AR.trantype=a.trantype
and ATS_Sales_AR.Invoice=a.Invoice


update ATS_Sales_AR set BDI=a.Sales_amount
from LN_SALES_BDI a
where ATS_Sales_AR.company=a.Company
and ATS_Sales_AR.trantype=a.trantype
and ATS_Sales_AR.Invoice=a.Invoice


update ATS_Sales_AR set Auction=a.Sales_amount
from Auction_sales a
where ATS_Sales_AR.company=a.Company
and ATS_Sales_AR.trantype=a.trantype
and ATS_Sales_AR.Invoice=a.Invoice


update ATS_Sales_AR set InvAmt =ACR.InvAmt ,PaidAmt =ACR.PaidAmt,Balance=ACR.BalanceAmt
from (
select * from ACR_Debtors_Details 
where company in (select company from ELGI_Companies_Master where CompanyGroup='ATS')
)ACR
where ATS_Sales_AR.company=ACR.Company
and ATS_Sales_AR.trantype=ACR.Trantyp
and ATS_Sales_AR.Invoice=ACR.InvNo
