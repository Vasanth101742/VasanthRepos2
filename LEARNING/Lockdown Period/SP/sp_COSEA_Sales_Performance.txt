Begin

Truncate table COSEA_Sales_Performance_Tmp
Truncate table COSEA_Sales_Performance


---------------------------------------------Budget--------------------------------------------------
Insert into COSEA_Sales_Performance_Tmp(
--Global
organ,
--region,user_names,Division,description
Dpstcode,
--dpstname
AreaCode,
--AreaName
--ordr
budget_year,last_month_budget,ytd_budget,
this_month_budget,next_month_budget,nextof_next_month_budget,Currency)

select organ,Dpst,Area, 
SUM(CASE WHEN date BETWEEN 
('04/01/' + convert(varchar,(case when month(getdate()-1) between 1 and 3 then year(getdate()-1) - 1 else year(getdate()-1) end ))) 
AND 
('03/31/' + convert(varchar,(case when month(getdate()-1) between 1 and 3 then year(getdate()-1) else year(getdate()-1)+1 end )))
THEN (Budget_USD) ELSE 0 END) AS budget_year,  

SUM(CASE WHEN date BETWEEN 
('04/01/' + convert(varchar,(case when month(getdate()-1) between 1 and 4 then year(getdate()-1) - 1 else year(getdate()-1) end ))) 
AND 
convert(varchar,dateadd(dd, - 1, CONVERT(varchar, datepart(mm, dateadd(dd, - 1, getdate()))) + '/1/' + datename(yy, dateadd(dd, - 1, getdate()))),101) 
THEN (Budget_USD) ELSE 0 END) AS last_month_budget,  

SUM(CASE WHEN date BETWEEN 
'04/01/' + convert(varchar,(case when month(getdate()-1) between 1 and 3 then year(getdate()-1) - 1 else year(getdate()-1) end )) 
AND 
dateadd(dd, - 1, getdate()) 
THEN (Budget_USD) ELSE 0 END) AS ytd_budget, 

SUM(CASE WHEN date BETWEEN 
CONVERT(varchar, datepart(mm, dateadd(dd, - 1, getdate()))) + '/1/' + datename(yy, dateadd(dd, - 1,getdate())) 
AND 
convert(varchar,dateadd(dd, - 1, getdate()),101) 
THEN (Budget_USD) ELSE 0 END) AS this_month_budget, 

SUM(CASE WHEN month(date)= datepart(mm, dateadd(mm, 1, dateadd(dd, - 1, getdate()))) AND 
year(date) = datepart(yy, dateadd(mm, 1, dateadd(dd, - 1, getdate()))) 
THEN (Budget_USD) ELSE 0 END) AS next_month_budget,     

SUM(CASE WHEN month(date) = datepart(mm, dateadd(mm, 2, dateadd(dd, - 1,getdate()))) AND 
year(date) = datepart(yy, dateadd(mm, 2, dateadd(dd, - 1, getdate()))) 
THEN (Budget_USD) ELSE 0 END) AS nextof_next_month_budget,

'USD' as Currency

FROM Sales_Budget a
where organ='COSEA'
--and Dpst='90257' and Area='685'
group by Organ,Dpst,Area


----Australia Budget from User

Insert into COSEA_Sales_Performance_Tmp(organ,AreaCode,CustCode,Vertical1,
budget_year,last_month_budget,ytd_budget,this_month_budget,next_month_budget,nextof_next_month_budget,Currency)
SELECT	Organ,Areacode,CustCode,Vertical,
SUM(budget_year) AS budget_year, 
SUM(last_month_budget) AS last_month_budget, 
SUM(ytd_budget) AS ytd_budget,     
SUM(this_month_budget) AS this_month_budget, 
SUM(next_month_budget) AS next_month_budget, 
SUM(nextof_next_month_budget) AS nextof_next_month_budget,
'AUD' as Currency
FROM (

SELECT	Organ,Areacode,CustCode,Vertical,

SUM(CASE WHEN BudDate BETWEEN 
('04/01/' + convert(varchar,(case when month(getdate()-1) between 1 and 3 then year(getdate()-1) - 1 else year(getdate()-1) end ))) 
AND 
('03/31/' + convert(varchar,(case when month(getdate()-1) between 1 and 3 then year(getdate()-1) else year(getdate()-1)+1 end )))
THEN (Budget_AUD) ELSE 0 END) AS budget_year,  

SUM(CASE WHEN BudDate BETWEEN 
('04/01/' + convert(varchar,(case when month(getdate()-1) between 1 and 4 then year(getdate()-1) - 1 else year(getdate()-1) end ))) 
AND 
convert(varchar,dateadd(dd, - 1, CONVERT(varchar, datepart(mm, dateadd(dd, - 1, getdate()))) + '/1/' + datename(yy, dateadd(dd, - 1, getdate()))),101) 
THEN (Budget_AUD) ELSE 0 END) AS last_month_budget,  

SUM(CASE WHEN BudDate BETWEEN 
'04/01/' + convert(varchar,(case when month(getdate()-1) between 1 and 3 then year(getdate()-1) - 1 else year(getdate()-1) end )) 
AND 
dateadd(dd, - 1, getdate()) 
THEN (Budget_AUD) ELSE 0 END) AS ytd_budget, 

SUM(CASE WHEN BudDate BETWEEN 
CONVERT(varchar, datepart(mm, dateadd(dd, - 1, getdate()))) + '/1/' + datename(yy, dateadd(dd, - 1,getdate())) 
AND 
convert(varchar,dateadd(dd, - 1, getdate()),101) 
THEN (Budget_AUD) ELSE 0 END) AS this_month_budget, 

SUM(CASE WHEN month(BudDate)= datepart(mm, dateadd(mm, 1, dateadd(dd, - 1, getdate()))) AND 
year(BudDate) = datepart(yy, dateadd(mm, 1, dateadd(dd, - 1, getdate()))) 
THEN (Budget_AUD) ELSE 0 END) AS next_month_budget,     

SUM(CASE WHEN month(BudDate) = datepart(mm, dateadd(mm, 2, dateadd(dd, - 1,getdate()))) AND 
year(BudDate) = datepart(yy, dateadd(mm, 2, dateadd(dd, - 1, getdate()))) 
THEN (Budget_AUD) ELSE 0 END) AS nextof_next_month_budget

FROM	SalesBudgetMaster_Australia a
group by Organ,Areacode,CustCode,Vertical
) AS budget    

GROUP BY Organ,Areacode,CustCode,Vertical
HAVING      (SUM(budget_year) > 0) or (SUM(last_month_budget) > 0)

SUM(this_month_budget) AS this_month_budget, 
SUM(next_month_budget) AS next_month_budget, 
SUM(nextof_next_month_budget) AS nextof_next_month_budget,
'USD' as Currency

SELECT	Organ,Areacode,CustCode,CustName,Vertical,
SUM(CASE WHEN BudDate BETWEEN 
('04/01/' + convert(varchar,(case when month(getdate()-1) between 1 and 3 then year(getdate()-1) - 1 else year(getdate()-1) end ))) 
AND 
('03/31/' + convert(varchar,(case when month(getdate()-1) between 1 and 3 then year(getdate()-1) else year(getdate()-1)+1 end )))
THEN (Budget_USD) ELSE 0 END) AS budget_year,  

SUM(CASE WHEN BudDate BETWEEN 
('04/01/' + convert(varchar,(case when month(getdate()-1) between 1 and 4 then year(getdate()-1) - 1 else year(getdate()-1) end ))) 
AND 
convert(varchar,dateadd(dd, - 1, CONVERT(varchar, datepart(mm, dateadd(dd, - 1, getdate()))) + '/1/' + datename(yy, dateadd(dd, - 1, getdate()))),101) 
THEN (Budget_USD) ELSE 0 END) AS last_month_budget,  

SUM(CASE WHEN BudDate BETWEEN 
'04/01/' + convert(varchar,(case when month(getdate()-1) between 1 and 3 then year(getdate()-1) - 1 else year(getdate()-1) end )) 
AND 
dateadd(dd, - 1, getdate()) 
THEN (Budget_USD) ELSE 0 END) AS ytd_budget, 

SUM(CASE WHEN BudDate BETWEEN 
CONVERT(varchar, datepart(mm, dateadd(dd, - 1, getdate()))) + '/1/' + datename(yy, dateadd(dd, - 1,getdate())) 
AND 
convert(varchar,dateadd(dd, - 1, getdate()),101)

year(BudDate) = datepart(yy, dateadd(mm, 2, dateadd(dd, - 1, getdate()))) 
THEN (Budget_USD) ELSE 0 END) AS nextof_next_month_budget

) AS budget    

GROUP BY Organ,Areacode,CustCode,CustName,Vertical
HAVING      (SUM(budget_year) > 0) or (SUM(last_month_budget) > 0)





----------------------------------------End: Budget------------------------------------------------

--AND 
--dateadd(dd,-1,dateadd(mm, - 2, CONVERT(varchar, datepart(mm,dateadd(dd, - 1, getdate()))) +'/1/' + datename(yy, dateadd(dd, - 1, getdate()))))

OrderDate  <

THEN (Foreign_currency_amount/1000) ELSE 0 END) AS OpenOrderValue_tillLM,

OrderDate between
CONVERT(varchar, datepart(mm, dateadd(dd, - 1, getdate()))) + '/1/' + datename(yy, dateadd(dd, - 1, getdate())) 
AND getdate() THEN (Foreign_currency_amount/1000) ELSE 0 END) AS OpenOrderValue_MTD,
