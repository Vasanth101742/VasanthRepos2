select dpst,date,convert(numeric(16,0),value1)/1000000,area,a,b from
(select dpst,concat(@FromYear,'-04-01') as date,sum(apr_value) as value1 ,area,0 as a,0 as b  from intra_sales_budget_details
where fin_year=concat(@FromYear,'-',@ToYear) group by dpst,area
union all
select dpst,concat(@FromYear,'-05-01'),sum(may_value),area,0 as a,0 as b  from intra_sales_budget_details 
where fin_year=concat(@FromYear,'-',@ToYear) group by dpst,area
union all
select dpst,concat(@FromYear,'-06-01'),sum(jun_value),area,0 as a,0 as b  from intra_sales_budget_details 
where fin_year=concat(@FromYear,'-',@ToYear) group by dpst,area
union all

where fin_year=concat(@FromYear,'-',@ToYear) group by dpst,area
union all
select dpst,concat(@FromYear,'-09-01'),sum(sep_value),area,0 as a,0 as b  from intra_sales_budget_details 
where fin_year=concat(@FromYear,'-',@ToYear) group by dpst,area
union all
select dpst,concat(@FromYear,'-10-01'),sum(oct_value),area,0 as a,0 as b  from intra_sales_budget_details 
where fin_year=concat(@FromYear,'-',@ToYear) group by dpst,area

where fin_year=concat(@FromYear,'-',@ToYear) group by dpst,area
union all

where fin_year=concat(@FromYear,'-',@ToYear) group by dpst,area
union all
select dpst,concat(@ToYear,'-01-01'),sum(jan_value),area,0 as a,0 as b  from intra_sales_budget_details 
where fin_year=concat(@FromYear,'-',@ToYear) group by dpst,area
union all
select dpst,concat(@ToYear,'-02-01'),sum(feb_value),area,0 as a,0 as b  from intra_sales_budget_details 
where fin_year=concat(@FromYear,'-',@ToYear) group by dpst,area
union all
select dpst,concat(@ToYear,'-03-01'),sum(mar_value),area,0 as a,0 as b  from intra_sales_budget_details 
where fin_year=concat(@FromYear,'-',@ToYear) group by dpst,area
)a

insert into TempBudget(Dpst,Date,Target,Area,SaleType,HoTarget)
select dpst,date,convert(numeric(16,0),value1)/1000000,AREACODE,a,b from
(select dpst,concat(@FromYear,'-04-01') as date,sum(AprilValue)*avg(INRConversionRate) as value1 ,AREACODE,0 as a,0 as b  from Australia_Budget_Master
where FinYear=concat(@FromYear,'-',@ToYear) group by dpst,AREACODE
union all
select dpst,concat(@FromYear,'-05-01'),sum(MayValue)*avg(INRConversionRate),AREACODE,0 as a,0 as b  from Australia_Budget_Master 
where FinYear=concat(@FromYear,'-',@ToYear) group by dpst,AREACODE
union all
select dpst,concat(@FromYear,'-06-01'),sum(JuneValue)*avg(INRConversionRate),AREACODE,0 as a,0 as b  from Australia_Budget_Master 
where FinYear=concat(@FromYear,'-',@ToYear) group by dpst,AREACODE
union all
select dpst,concat(@FromYear,'-07-01'),sum(JulyValue)*avg(INRConversionRate),AREACODE,0 as a,0 as b  from Australia_Budget_Master 
where FinYear=concat(@FromYear,'-',@ToYear) group by dpst,AREACODE
union all
select dpst,concat(@FromYear,'-08-01'),sum(AugustValue)*avg(INRConversionRate),AREACODE,0 as a,0 as b  from Australia_Budget_Master 
where FinYear=concat(@FromYear,'-',@ToYear) group by dpst,AREACODE
union all
select dpst,concat(@FromYear,'-09-01'),sum(SeptValue)*avg(INRConversionRate),AREACODE,0 as a,0 as b  from Australia_Budget_Master 
where FinYear=concat(@FromYear,'-',@ToYear) group by dpst,AREACODE
union all
select dpst,concat(@FromYear,'-10-01'),sum(OctValue)*avg(INRConversionRate),AREACODE,0 as a,0 as b  from Australia_Budget_Master 
where FinYear=concat(@FromYear,'-',@ToYear) group by dpst,AREACODE
union all
select dpst,concat(@FromYear,'-11-01'),sum(NovValue)*avg(INRConversionRate),AREACODE,0 as a,0 as b  from Australia_Budget_Master 
where FinYear=concat(@FromYear,'-',@ToYear) group by dpst,AREACODE
union all
select dpst,concat(@FromYear,'-12-01'),sum(DecValue)*avg(INRConversionRate),AREACODE,0 as a,0 as b  from Australia_Budget_Master 
where FinYear=concat(@FromYear,'-',@ToYear) group by dpst,AREACODE
union all
select dpst,concat(@ToYear,'-01-01'),sum(JanValue)*avg(INRConversionRate),AREACODE,0 as a,0 as b  from Australia_Budget_Master 
where FinYear=concat(@FromYear,'-',@ToYear) group by dpst,AREACODE
union all
select dpst,concat(@ToYear,'-02-01'),sum(FebValue)*avg(INRConversionRate),AREACODE,0 as a,0 as b  from Australia_Budget_Master 
where FinYear=concat(@FromYear,'-',@ToYear) group by dpst,AREACODE
union all
select dpst,concat(@ToYear,'-03-01'),sum(MarValue)*avg(INRConversionRate),AREACODE,0 as a,0 as b  from Australia_Budget_Master 
where FinYear=concat(@FromYear,'-',@ToYear) group by dpst,AREACODE

select *,'' from TempBudget where date between concat(@FromYear,'-04-01') and concat(@ToYear,'-03-01') 


update Sales_Budget set organ=a.organ from 
(SELECT DISTINCT DPST,AREA,ORGAN FROM Sales_Budget where organ !=' ') a 
where ltrim(rtrim(Sales_Budget.area))=ltrim(rtrim(a.area)) and ltrim(rtrim(Sales_Budget.dpst))=ltrim(rtrim(a.dpst))  
and date between concat(@FromYear,'-04-01') and concat(@ToYear,'-03-01')



--select * from Sales_Budget where date between concat(@FromYear,'-04-01') and concat(@ToYear,'-03-01') and  organ =''
update Sales_Budget set organ='India' where date between concat(@FromYear,'-04-01') and concat(@ToYear,'-03-01')
and organ ='' and area='001'


update Sales_Budget set organ = ltrim(rtrim(a.organ)) from vw_SALESMAPPING_MASTER a where Sales_Budget.area=a.areacode and Sales_Budget.dpst=a.DPSTCode



update Sales_Budget set organ=a.organ from 
(select distinct areacode,organ  from vw_SALESMAPPING_MASTER where areacode in (
select distinct area area  from Sales_Budget where date between concat(@FromYear,'-04-01') and concat(@ToYear,'-03-01')  and organ = '' ))a 
where Sales_Budget.organ='' and a.areacode=Sales_Budget.area
