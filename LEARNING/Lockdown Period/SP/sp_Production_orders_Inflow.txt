Insert into Production_orders_Inflow_Temp(Company,CWAR,ItemCode,PDNO,Planned_qty_till_CurrDay)
select * from (
select --convert(date,RDLD) as date1,
company,
CWAR as cwar,
MITM as Item,
PDNO as pdno,
sum(case when osta in (1,2,3) then QRDR else 0 end ) as Planned_qty_till_CurrDay
--sum(case when RDLD='9999-01-01 00:00:00.000' then 0 else 
--(case when osta in (1,2,3) and EFDT <=convert(date,getdate()-1) then qrdr else 0 end) end) as Planned_qty_till_CurrDay,
--sum(case when RDLD='9999-01-01 00:00:00.000' then 0 else 
--(case when osta in (1,2,3) and EFDT <=convert(date,getdate()-2) then qrdr else 0 end) end) as Planned_qty_till_PrevDay
from TTISFC001ALL
--where ltrim(rtrim(MITM))='S012454'
group by --convert(date,RDLD),
company,CWAR,MITM,PDNO
)a
where Planned_qty_till_CurrDay <>0

Insert into Production_orders_Inflow_Temp2(Company,CWAR,ItemCode,PDNO,Planned_qty_till_PrevDay,EntryDate)
select * from (
select --convert(date,RDLD) as date1,
company,
CWAR as cwar,
MITM as Item,
PDNO as pdno,
sum(case when osta in (1,2,3) then QRDR else 0 end ) as Planned_qty,

from TTISFC001ALL
--where ltrim(rtrim(MITM))='S012454'
group by --convert(date,RDLD),
company,CWAR,MITM,PDNO
)a
where Planned_qty<>0


Insert into Production_orders_Inflow_Temp(Company,CWAR,ItemCode,
PDNO,Released_qty_CurrDay
)
select * from (
select 
--convert(date,ReleaseDt) as date1,
company,
CWAR as cwar,
a.MITM as Item,
PDNO as pdno,
sum(case when ADLD='9999-01-01 00:00:00.000' then 0 else (case when osta=6 then qrdr else 0 end) end) as Released_qty_CurrDay
from (
select a.*,b.date as ReleaseDt from  TTISFC001ALL a 
left outer join
(
select * from (
select company,PDNO,OSTA,convert(date,date) as date,row_number() over (partition by company,pdno order by date) dateRank 
from TTICST151ALL
where OSTA =6
--and PDNO='230009816' --for testing Purpose
)b1
where dateRank=1
)b
on a.company=b.company
and a.PDNO=b.PDNO
--where a.PDNO='280021638'  --for testing Purpose
)a
where convert(date,ReleaseDt)=convert(date,getdate()-1)
group by --convert(date,ReleaseDt),
company,CWAR,MITM,PDNO
)b
where Released_qty_CurrDay<>0




insert into Production_orders_Inflow(Company,CWAR,ItemCode,ItemDescr,Description,PDNO,
Planned_qty_till_PrevDay,Planned_qty_till_CurrDay,Released_qty_CurrDay)
select Company,CWAR,ItemCode,ItemDescr,Description,PDNO,
Planned_qty_till_PrevDay,Planned_qty_till_CurrDay,Released_qty_CurrDay
from (
select Company,CWAR,
ItemCode,ItemDescr,Description,PDNO,
sum(Planned_qty_till_CurrDay) Planned_qty_till_CurrDay,
sum(Planned_qty_till_PrevDay) Planned_qty_till_PrevDay,
sum(Released_qty_CurrDay) Released_qty_CurrDay
from (select Company,CWAR,ItemCode,ItemDescr,Description,PDNO,
Planned_qty_till_CurrDay,Planned_qty_till_PrevDay,Released_qty_CurrDay
from Production_orders_Inflow_Temp
union
select Company,CWAR,ItemCode,ItemDescr,Description,PDNO,
Planned_qty_till_CurrDay,Planned_qty_till_PrevDay,Released_qty_CurrDay
from Production_orders_Inflow_Temp2
where EntryDate=convert(date,getdate()-2)
)a
group by Company,CWAR,ItemCode,ItemDescr,Description,PDNO
)a
--where isnull(Planned_qty_till_CurrDay,0) <>0
--and isnull(Planned_qty_till_PrevDay,0) <>0 
--and isnull(Released_qty_CurrDay,0)<>0


update Production_orders_Inflow set InflowOrders=isnull(Planned_qty_till_CurrDay,0)-isnull(Planned_qty_till_PrevDay,0)+isnull(Released_qty_CurrDay,0)
update Production_orders_Inflow set InflowOrders=0 where InflowOrders<0


update Production_orders_Inflow set Description=b.Description
from (
select distinct a.company,Description,prd_from,prd_to from ELGi_Production_Master a
left outer join ELGi_AssemblyLine_Master b
on b.code = a.assemblycode
)b
where case when Production_orders_Inflow.pdno like '%[a-z]%' then '0' else Production_orders_Inflow.pdno end 
between case when b.prd_from like '%[a-z]%' then '0' else b.prd_from end 
and case when b.prd_to like '%[a-z]%' then '0' else b.prd_to end  
and Production_orders_Inflow.company = b.company
