CREATE proc sp_mc_Bud_act

truncate table mc_sales 
insert into mc_sales 
select *,0 as sales_bud,0 as sales_bud_Qty,0 as mc_budget --into mc_sales 
From(
select dpst,areacode,year(Invoice_Date) as year,month(invoice_date) as month,ltrim(rtrim(Item)) as Item,sum(Sales_amount) as Sales,sum(DeliveryQty)DeliveryQty,0 as sales_Mc
From elgi_sales_details 
where company <500 and convert(date,Invoice_Date)>='2019-04-01' --and (ltrim(rtrim(Item)) is not null and ltrim(rtrim(Item))<>'') and dpst not like '%[a-z]%'
and sales_category not in (1) and
ORGAN IN ('INDIA','COSEA','SAAME','ID','Subsidiary')
--and year(Invoice_Date)=@year and month(invoice_date) in (@month)
group by year(Invoice_Date),month(invoice_date),ltrim(rtrim(Item)),dpst,areacode)a

-- select areacode,Item,dpst,company from elgi_sales_details where item='          S02074' and dpst='90010' and company =401
update mc_sales set sales_Mc=(mc_sales.DeliveryQty*a.sales_Mc)  
from(select t$year,t$mnth,ltrim(rtrim(t$sele))t$sele,sum(convert(float,t$impo)) sales_Mc  From TZZCPR104ALL group by t$sele,t$year,t$mnth)a
where mc_sales.item=ltrim(rtrim(t$sele))
and mc_sales.year=a.T$YEAR 
and mc_sales.month=a.T$MNTH 
-----------------------------End Sales

select ltrim(rtrim(item_code))item,dpst,area as areacode,year(date)as year,month(date) as month,sum(target) as sales_bud,sum(qty)sales_bud_Qty 
From Itemwise_Budget where date>='2019-04-01'
and dpst<>'70080'
and len(ltrim(rtrim(item_code)))>2 and target >0
group by year(date),month(date),dpst,area,ltrim(rtrim(item_code))
)a
