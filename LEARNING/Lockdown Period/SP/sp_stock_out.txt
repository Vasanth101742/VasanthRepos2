select a.id,a.company,a.Division,a.Department,b.war,c.item,d.qskt 
from Lean_Stock_Check_Div_Dept_Master a
left outer join Lean_Stock_Check_WH_Master b
on a.id=b.id
left outer join (select * from Lean_Stock_Out)c
on a.company=c.company
and a.Division=c.Department
and a.Department=c.Supermarket
left outer join (select company,cwar,ltrim(rtrim(item)) item,sum(qskt) qskt from inventorydetails
group by company,cwar,ltrim(rtrim(item)))d
on c.company=d.company
and b.war=d.cwar
and ltrim(rtrim(c.item))=ltrim(rtrim(d.item))
where d.qskt is not null
--and ltrim(rtrim(c.item))='970301310'
)a

and ltrim(rtrim(Lean_Stock_Out.DEPARTMENT))=ltrim(rtrim(a.Division))
and ltrim(rtrim(Lean_Stock_Out.item))=ltrim(rtrim(a.item))
