SELECT * FROM 
(
SELECT	vw_mon_bud_perf_domestic.budget_year,  
	vw_mon_bud_perf_domestic.last_month_budget,  
	vw_mon_bud_perf_domestic.ytd_budget, 
	vw_mon_bud_perf_domestic.this_month_budget,  
	vw_mon_bud_perf_domestic.next_month_budget, 
	vw_mon_bud_perf_domestic.nextof_next_month_budget,  
	isnull(vw_mon_bud_perf_domestic.description,vw_mon_sls_perf_domestic.description) as description,  
	vw_mon_sls_perf_domestic.last_year_sale, 
	vw_mon_sls_perf_domestic.last_year_ytd,  
	vw_mon_sls_perf_domestic.this_year_sale,  
	vw_mon_sls_perf_domestic.last_month_sale, 
	vw_mon_sls_perf_domestic.thr_month_sale,
	vw_mon_sls_perf_domestic.this_month_sale,
	isnull(vw_mon_bud_perf_domestic.dpstname,  vw_mon_sls_perf_domestic.dpstname) as dpstname,
	n1_month_sale,
	n2_month_sale,
	n3_month_sale

FROM	vw_mon_bud_perf_domestic_dpstwise vw_mon_bud_perf_domestic 
	full JOIN
        vw_mon_sls_perf_domestic_dpstwise vw_mon_sls_perf_domestic 
	ON  vw_mon_bud_perf_domestic.description =  vw_mon_sls_perf_domestic.description 
	and vw_mon_bud_perf_domestic.dpstname =  vw_mon_sls_perf_domestic.dpstname
)A
Where Description IN (@Description)
