select  
	isnull(a.organ,b.organ) as organ,
	 case when coalesce(a.region,b.region) ='Neighbouring' then 'S. Asia                       ' 
		     else coalesce(a.region,b.region) end as region ,

	case when coalesce(a.Zone,b.Zone) ='S. Asia                       '  then 'Neighbouring'
	     else coalesce(a.Zone,b.Zone) end as Zone,
	isnull(a.division,b.division) as division,
	isnull(a.user_names,b.user_names) as user_names, 
	isnull(last_year_sale,0) as last_year_sale,
	isnull(last_year_ytd,0) as last_year_ytd,
	isnull(this_year_sale,0) as this_year_sale,
	isnull(last_month_sale,0) as last_month_sale,
	isnull(this_month_sale,0) as this_month_sale,
	isnull(budget_year,0) as budget_year,
	isnull(last_month_budget,0) as last_month_budget,
	isnull(ytd_budget,0) as ytd_budget ,
	isnull(this_month_budget,0) as this_month_budget,
	isnull(next_month_budget,0) as next_month_budget,
	isnull(nextof_next_month_budget,0) as nextof_next_month_budget
 from (
	SELECT BDM_Inter.organ, BDM_Inter.region,BDM_Inter.areaname as Division, AreaTmp.Zone, BDM_Inter.user_names, 
	       SUM(CASE WHEN date BETWEEN @lyfromdate AND @lytodate THEN (slsdespdata.assesvalue) / 1000000 ELSE 0 END) AS last_year_sale, 
	       SUM(CASE WHEN date BETWEEN @lyfromdate AND dateadd(yy, - 1, dateadd(dd, - 1, getdate())) THEN (slsdespdata.assesvalue) / 1000000 ELSE 0 END) AS          

               last_year_ytd,
	       SUM(CASE WHEN date BETWEEN @fromdate AND dateadd(dd, - 1, getdate()) THEN (slsdespdata.assesvalue) / 1000000 ELSE 0 END) AS this_year_sale, 
	       SUM(CASE WHEN date BETWEEN @fromdate AND dateadd(dd, - 1, CONVERT(varchar, datepart(mm, dateadd(dd, - 1, getdate()))) + '/1/' + datename(yy, 		

               dateadd(dd, - 1, getdate()))) THEN (slsdespdata.assesvalue) / 1000000 ELSE 0 END) AS last_month_sale, 
	       SUM(CASE WHEN date BETWEEN CONVERT(varchar, datepart(mm, dateadd(dd, - 1, getdate()))) + '/1/' + datename(yy, dateadd(dd, - 1, getdate())) AND 		

               dateadd(dd, - 1, getdate()) 
	       THEN (slsdespdata.assesvalue) / 1000000 ELSE 0 END) AS this_month_sale

	FROM dbo.SlsDespData 
             INNER JOIN
             dbo.AreaTmp ON dbo.AreaTmp.AreaCode = dbo.SlsDespData.Area
             INNER JOIN	                      
             dbo.TsalesMapping AS BDM_Inter 
	     ON (dbo.AreaTmp.region = BDM_Inter.region and AreaTmp.areaname=BDM_inter.areaname)  AND dbo.SlsDespData.Dpst = BDM_Inter.dpst 

	WHERE  (SlsDespData.Date BETWEEN @lyfromdate AND GETDATE()) AND 
	       (SlsDespData.SaleType IN (0, 2, 5, 6)) AND 
	       (dbo.SlsDespData.Area IS NOT NULL) AND 
	       (LTRIM(RTRIM(dbo.AreaTmp.Global)) = 'International') and 
               BDM_Inter.organ = 'ISAAME' and
	       BDM_Inter.region in (@region)  and
	       AreaTmp.Zone in (@zone)   and
               BDM_Inter.areaname in (@division)  and
               BDM_Inter.user_names in (@user_names)  
	       

        GROUP BY BDM_Inter.organ,BDM_Inter.region,AreaTmp.Zone,BDM_Inter.areaname, BDM_Inter.user_names
     )a

full outer join

(
	SELECT  organ,region,Division,Zone, user_names, 
		isnull(SUM(budget_year),0) AS budget_year, 
		isnull(SUM(last_month_budget),0) AS last_month_budget, 
		isnull(SUM(ytd_budget),0) AS ytd_budget, 
		isnull(SUM(this_month_budget),0) AS this_month_budget, 
		isnull(SUM(next_month_budget),0) AS next_month_budget, 
		isnull(SUM(nextof_next_month_budget),0) AS nextof_next_month_budget
	FROM(
		SELECT  BDM_inter.organ,BDm_inter.region as region,AreaTmp.Zone, BDM_Inter.user_names, 
			SUM(CASE WHEN date BETWEEN @fromdate AND @todate THEN (target) ELSE 0 END) AS budget_year, 
			SUM(CASE WHEN date BETWEEN @fromdate AND dateadd(dd, - 1, CONVERT(varchar, datepart(mm, dateadd(dd, - 1, getdate()))) + '/1/' + datename(yy, 	

			dateadd(dd, - 1, getdate()))) THEN (target) ELSE 0 END) AS last_month_budget, SUM(CASE WHEN date BETWEEN  @fromdate AND dateadd(dd, - 1, 	

			getdate()) THEN (target) ELSE 0 END) AS ytd_budget, 
			SUM(CASE WHEN date BETWEEN        CONVERT(varchar, datepart(mm, dateadd(dd, - 1, getdate()))) + '/1/' + datename(yy, dateadd(dd, - 1, getdate	

			())) AND dateadd(dd, - 1, getdate())THEN (target) ELSE 0 END) AS this_month_budget, SUM(CASE WHEN month(date) = datepart(mm, dateadd(mm, 1, 	

		dateadd(dd, - 1, getdate())))        	                AND year(date) = datepart(yy, dateadd(mm, 1, dateadd(dd, - 1, getdate()))) THEN (target) ELSE 0 

END) AS next_month_budget, SUM(CASE WHEN month			(date) 			= datepart(mm, dateadd(mm, 2, dateadd(dd, - 1, getdate()))) AND year(date) = 

datepart(yy, dateadd(mm, 2, dateadd(dd, - 1, getdate()))) THEN 			(target) ELSE 0 END) AS 			nextof_next_month_budget, 

BDm_inter.areaname as Division
	       FROM     dbo.TSalesBudget AS dpstwise_target
                        INNER JOIN  
                        dbo.AreaTmp ON dbo.AreaTmp.AreaCode = dpstwise_target.Area 
			INNER JOIN  
                        dbo.TSalesMapping AS BDM_Inter ON  (dbo.AreaTmp.region = BDM_Inter.region AND areatmp.areaname=BDm_inter.areaname  
                                              		     and dpstwise_target.Dpst = BDM_Inter.dpst) 
	       WHERE     BDM_Inter.organ = 'ISAAME' and
	   		 BDM_Inter.region in (@region)  and
			 AreaTmp.Zone in (@zone)   and
			 BDM_Inter.areaname in (@division)  and
			 BDM_Inter.user_names in (@user_names)  
			
	       GROUP BY BDM_Inter.organ,BDM_Inter.region,AreaTmp.Zone,BDM_inter.areaname,BDM_Inter.user_names
            ) AS budget
      
       GROUP BY organ,region,Zone,Division, user_names
       HAVING      (SUM(budget_year) > 0)
)b
on a.region = b.region and 
   a.organ = b.organ and 
   a.user_names = b.user_names and 
   a.division = b.division
