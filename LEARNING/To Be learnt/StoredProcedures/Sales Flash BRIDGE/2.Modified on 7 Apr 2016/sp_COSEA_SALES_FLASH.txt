Alter procedure sp_COSEA_SALES_FLASH(@ytdyear varchar(10))
as


declare @ytddate varchar(200)
declare @monthnum int 
--declare @ytdyear varchar(5)='2015'

begin

set @ytddate = '04/01/'+@ytdyear
set @monthnum=(select case when monthno between 1 and 3 then monthno + 9 else monthno - 3 end as monthnum from (select month(getdate()-3) as monthno) a)
--set @monthnum=(select case when @monthnum-1 =0 then 12 else @monthnum-1 end)  --recently added

insert into ELGI_SALES_FLASH

select  'COSEA' as Organ, 
user_names,
Division,
round(sum(month_budget)/1000000,2) month_budget, 
round(sum(month_actuals)/1000000,2) month_actuals,
round(sum(last_month_actuals)/1000000,2) last_month_actuals,
round(sum(ytd_budget)/1000000,2) ytd_budget,
round(sum(ytd_actuals)/1000000,2) ytd_actuals,
round(sum(ly_ytd_actuals)/1000000,2) ly_ytd_actuals,
round(sum(ly_month_actuals)/1000000,2) ly_month_actuals

from (

select isnull(c.vertical_order,b.vertical_order) as vertical_order,
ltrim(rtrim(isnull(c.divn_order,b.divn_order))) as divn_order,
isnull(c.user_names,b.user_names) as user_names,
ltrim(rtrim(isnull(c.division,b.division))) as division,
isnull(c.dpstcode, b.dpstcode) as dpstcode, 
isnull(c.dpstname, b.dpstname) as dpstname, 
isnull(b.month_budget,0) as month_budget,
isnull(b.ytd_budget,0) as ytd_budget,
isnull(b.month_actuals,0) as month_actuals,
isnull(b.last_month_actuals,0) as last_month_actuals,
isnull(b.ytd_actuals,0) as ytd_actuals,
isnull(ly_ytd_actuals,0) as ly_ytd_actuals,
isnull(ly_month_actuals,0) as ly_month_actuals 

from(
	select 0 as vertical_order,0 as divn_order,
	--isnull(a.user_names,b.user_names) as user_names,
	--isnull(a.division,b.division) as division,
	--isnull(a.dpstcode, b.dpstcode) as dpstcode, 
	--isnull(a.dpstname, b.dpstname) as dpstname, 
	
	coalesce(a.user_names,b.user_names,c.user_names) as user_names,
	coalesce(a.division,b.division,c.division) as division,
	coalesce(a.dpstcode, b.dpstcode,c.dpstcode) as dpstcode, 
	coalesce(a.dpstname, b.dpstname,c.dpstname) as dpstname, 
	isnull(month_budget,0)*1000000 as month_budget,
	isnull(ytd_budget,0)*1000000 as ytd_budget,
	isnull(month_actuals,0) as month_actuals,
	isnull(last_month_actuals,0) as last_month_actuals,
	isnull(ytd_actuals,0) as ytd_actuals

		from(
		----DPST BREAKUP FOR BUDGET
		select 	user_names,division,dpstcode,dpstname,
			sum(case when monthnum = @monthnum then budget_month else 0 end) as month_budget,
			sum(case when monthnum between 1 and  @monthnum then budget_month else 0 end) as ytd_budget
		from(
			----MONTH BREAKUP FOR BUDGET
			SELECT 	user_names,Division,dpstcode,dpstname,budget_month,
				case when monthnum < 4 then monthnum + 9 else monthnum -3 end as monthnum,
				monthnames
			FROM(
				SELECT 	BDM_Inter.vertical1 as user_names,month(date) as monthnum,datename(mm,date) monthnames,
					SUM(target) AS budget_month, 
	                                dbo.dpstmaster_new.Division,dpstcode,dpstname

				FROM    TSalesbudget AS dpstwise_target 
					INNER JOIN dbo.dpstmaster_new 
						ON dbo.dpstmaster_new.DpstCode = dpstwise_target.Dpst 
					INNER JOIN dbo.AreaTmp 
						ON dbo.AreaTmp.AreaCode = dpstwise_target.Area 
					INNER JOIN TSalesMapping AS BDM_Inter 
						ON dbo.AreaTmp.areaname = BDM_Inter.areaname 
						AND dbo.dpstmaster_new.DpstCode = BDM_Inter.dpst

				WHERE	convert(Datetime,date) >= @ytddate 
					and convert(Datetime,date) < dateadd(yy,1,@ytddate) and convert(Datetime,date) <= dateadd(dd,-1,getdate()) 
					and BDM_Inter.organ='COSEA' 
					and dpstwise_target.organ='COSEA'

				GROUP BY dbo.dpstmaster_new.Division, BDM_Inter.vertical1,dpstcode,dpstname,month(date),datename(mm,date)
			     ) AS budget

			GROUP BY Division, user_names,dpstcode,dpstname,budget_month,
			case when monthnum < 4 then monthnum + 9 else monthnum -3 end,monthnames
			----END OF MONTH BREAKUP FOR BUDGET
		    ) a
		where monthnum <= @monthnum
		group by user_names,division,dpstcode,dpstname
		----END OF DPST BREAKUP FOR BUDGET
	   )a

	full join

	(

	----DPST BREAKUP FOR CURRENT YEAR AND MONTH ACTUALS

	 select user_names,division,dpstcode,dpstname,
		sum(case when monthnum = @monthnum then budget_month else 0 end) as month_actuals,
		--Old Method
		--sum(case when monthnum = @monthnum -1 then budget_month else 0 end) as last_month_actuals,
		sum(case when monthnum between 1 and  @monthnum then budget_month else 0 end) as ytd_actuals
	 from(
		----MONTH BREAKUP FOR CURRENT YEAR AND MONTH ACTUALS
		SELECT 	user_names,Division,dpstcode,dpstname,budget_month,
			case when monthnum < 4 then monthnum + 9 else monthnum -3 end as monthnum,
			monthnames
		FROM(
			SELECT 	BDM_Inter.vertical1 as user_names,month(date) as monthnum,datename(mm,date) monthnames,
				SUM(actuals) AS budget_month, 
				dbo.dpstmaster_new.Division,dpstcode,dpstname
                       	FROM (
				select 	dpst,convert(Datetime,date) date, sum(assesvalue) as actuals,area
				from 	slsdespdata (nolock) 
				where 	
					--OLD Method
					convert(Datetime,date) >= @ytddate 

					--New Method 
					--convert(Datetime,date) >=case when month(getdate()-1)=4 then dateadd(mm,-1,convert(datetime,@ytddate)) end
					
					and saletype in (0,2,5,6)
				
				group by dpst,convert(Datetime,date) ,area
			    ) AS dpstwise_actuals 
			INNER JOIN dbo.dpstmaster_new 
				ON dbo.dpstmaster_new.DpstCode = dpstwise_actuals.Dpst 
			INNER JOIN dbo.AreaTmp 
				ON dbo.AreaTmp.AreaCode = dpstwise_actuals.Area 
			INNER JOIN TSalesMapping AS BDM_Inter 
				ON dbo.AreaTmp.areaname = BDM_Inter.areaname 
				AND dbo.dpstmaster_new.DpstCode = BDM_Inter.dpst

			WHERE 	(dbo.dpstmaster_new.Description = 'exports') 
				AND (dbo.dpstmaster_new.Division <> 'AED') 
				
				--OLD Method
				and convert(Datetime,date) >= @ytddate 
				--New Method
				--and convert(Datetime,date) >=case when month(getdate()-1)=4 then dateadd(mm,-1,convert(datetime,@ytddate)) end
				
				and convert(Datetime,date) < dateadd(yy,1,@ytddate) 
				and BDM_Inter .organ='COSEA' 
                       
			GROUP BY dbo.dpstmaster_new.Division, BDM_Inter.vertical1,dpstcode,dpstname,month(date),datename(mm,date)
		     ) AS budget
		GROUP BY Division, user_names,dpstcode,dpstname,budget_month,
		case when monthnum < 4 then monthnum + 9 else monthnum -3 end,monthnames
		----MONTH BREAKUP FOR CURRENT YEAR AND MONTH ACTUALS
	      ) a
	where	
		--Old Method
		monthnum <= @monthnum

		--New Method
		--monthnum in (case when @monthnum -1=0 then 12 else @monthnum end)

	group by user_names,division,dpstcode,dpstname

	)b
	----END OF DPST BREAKUP FOR CURRENT YEAR AND MONTH ACTUALS

	on a.user_names = b.user_names 
	and a.division = b.division 
	and a.dpstcode = b.dpstcode 

	

	full join

	(

	----DPST BREAKUP FOR LAST MONTH ACTUALS

	 select user_names,division,dpstcode,dpstname,
	 	sum(case when monthnum = (case when @monthnum -1=0 then 11 else @monthnum-1 end) then budget_month else 0 end) as last_month_actuals
		
	 from(
		----MONTH BREAKUP FOR LAST MONTH ACTUALS
		SELECT 	user_names,Division,dpstcode,dpstname,budget_month,
			case when monthnum < 4 then monthnum + 9 else monthnum -3 end as monthnum,
			monthnames
		FROM(

			SELECT 	BDM_Inter.vertical1 as user_names,
				month(date) as monthnum,
				datename(mm,date) monthnames,
				SUM(actuals) AS budget_month, 
				dbo.dpstmaster_new.Division,
				dpstcode,dpstname
                       	FROM (

				select 	dpst,convert(Datetime,date) date, sum(assesvalue) as actuals,area
				from 	slsdespdata (nolock) 
				where 	
				--	convert(Datetime,date) >=dateadd(mm,-1,convert(datetime,@ytddate))
					--case when month(getdate()-1)=4 then dateadd(mm,-1,convert(datetime,@ytddate)) end and 
					saletype in (0,2,5,6)
				
				group by dpst,convert(Datetime,date) ,area
				--order by convert(Datetime,date),dpst,area

			    ) AS dpstwise_actuals 
			INNER JOIN dbo.dpstmaster_new 
				ON dbo.dpstmaster_new.DpstCode = dpstwise_actuals.Dpst 
			INNER JOIN dbo.AreaTmp 
				ON dbo.AreaTmp.AreaCode = dpstwise_actuals.Area 
			INNER JOIN TSalesMapping AS BDM_Inter 
				ON dbo.AreaTmp.areaname = BDM_Inter.areaname 
				AND dbo.dpstmaster_new.DpstCode = BDM_Inter.dpst

			WHERE 	(dbo.dpstmaster_new.Description = 'exports') 
				AND (dbo.dpstmaster_new.Division <> 'AED') 
				and convert(Date,date) >=
				convert(varchar,year(dateadd(mm,-2,getdate()-1)))+'-'+ RIGHT('00'+CAST(month(dateadd(mm,-2,getdate()-1)) AS VARCHAR(2)),2)+'-01'
				and convert(Date,date) < 
				convert(varchar,year(dateadd(mm,-1,getdate()-1)))+'-'+ RIGHT('00'+CAST(month(dateadd(mm,-1,getdate()-1)) AS VARCHAR(2)),2)+'-01'
				and BDM_Inter .organ='COSEA' 
                       
			GROUP BY dbo.dpstmaster_new.Division, BDM_Inter.vertical1,dpstcode,dpstname,month(date),datename(mm,date)
		     ) AS budget
		GROUP BY Division, user_names,dpstcode,dpstname,budget_month,
		case when monthnum < 4 then monthnum + 9 else monthnum -3 end,monthnames
		----END OF MONTH BREAKUP FOR LAST MONTH ACTUALS
	      ) a
	where	monthnum in (case when @monthnum -1=0 then 11 else @monthnum-1 end)
		group by user_names,division,dpstcode,dpstname

	)c
	----END OF DPST BREAKUP FOR LAST MONTH ACTUALS

	on a.user_names = c.user_names 
	and a.division = c.division 
	and a.dpstcode = c.dpstcode 

	

) b


full join


(

----DPST BREAKUP FOR LAST YEAR AND MONTH ACTUALS
 select 0 as vertical_order,0 as divn_order,user_names,division,dpstcode,dpstname,
 sum(case when monthnum = @monthnum then budget_month else 0 end) as ly_month_actuals,
 sum(case when monthnum between 1 and  @monthnum then budget_month else 0 end) as ly_ytd_actuals

 from(

	----MONTH BREAKUP FOR LAST YEAR AND MONTH ACTUALS
	SELECT 	user_names,Division,dpstcode,dpstname,budget_month,
		case when monthnum < 4 then monthnum + 9 else monthnum -3 end as monthnum,
		monthnames
	FROM(
		SELECT 	BDM_Inter.vertical1 as user_names,month(date) as monthnum,
			datename(mm,date) monthnames,SUM(actuals) AS budget_month, 
			dbo.dpstmaster_new.Division,dpstcode,dpstname
		FROM (
			select 	dpst,convert(Datetime,date) date, sum(assesvalue) as actuals,area
			from 	slsdespdata (nolock) 
			where 	convert(Datetime,date) >= dateadd(yy,-1,@ytddate) 
				and saletype in (0,2,5,6)
			group by dpst,convert(Datetime,date) ,area
		     )AS dpstwise_actuals 
			INNER JOIN dbo.dpstmaster_new 
				ON dbo.dpstmaster_new.DpstCode = dpstwise_actuals.Dpst 
			INNER JOIN dbo.AreaTmp 
				ON dbo.AreaTmp.AreaCode = dpstwise_actuals.Area 
			INNER JOIN TSalesMapping AS BDM_Inter 
				ON dbo.AreaTmp.areaname = BDM_Inter.areaname 
				AND dbo.dpstmaster_new.DpstCode = BDM_Inter.dpst

		WHERE 	(dbo.dpstmaster_new.Description = 'exports') 
			AND (dbo.dpstmaster_new.Division <> 'AED') 
			and BDM_Inter .organ='COSEA' 
			
			--OLD METHOD
			and convert(Datetime,date) <=  dateadd(dd,-1,convert(datetime,convert(varchar,datepart(mm,dateadd(mm,1,dateadd(yy,-1,getdate()-1 )))) + '/01/' + datename(yy,dateadd(mm,1,dateadd(yy,-1,getdate()-1)))))  
			and convert(Datetime,date) < @ytddate 
			and convert(Datetime,date) >= dateadd(yy,-1,@ytddate) 

			--New Method
			--and convert(Datetime,date)>=	convert(datetime,convert(varchar,datepart(mm,dateadd(yy,-1,dateadd(dd,-1,getdate())))) + '/1/' + datename(yy,dateadd(yy,-1,dateadd(dd,-1,getdate())))) 
			--and convert(Datetime,date)<=	dateadd(yy,-1,dateadd(dd,-1,getdate())) 

		GROUP BY dbo.dpstmaster_new.Division, BDM_Inter.vertical1,dpstcode,dpstname,month(date),datename(mm,date)
	    ) AS budget

	GROUP BY Division,user_names,dpstcode,dpstname,budget_month,
		case when monthnum < 4 then monthnum + 9 else monthnum -3 end,monthnames

	----END OF MONTH BREAKUP FOR LAST YEAR AND MONTH ACTUALS
      ) a
where monthnum <= @monthnum
group by user_names,division,dpstcode,dpstname

----END OF DPST BREAKUP FOR LAST YEAR AND MONTH ACTUALS
)c
on c.user_names = b.user_names 
and c.division = b.division 
and c.dpstcode = b.dpstcode
)complete
group by user_names,Division
order by user_names,Division

end
