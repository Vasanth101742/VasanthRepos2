declare @ytddate varchar(20)='04/01/2014'

SELECT 	BDM_Inter.vertical1 as user_names,dbo.dpstmaster_new.Division,dpstcode,dpstname,
month(date) as monthnum,datename(mm,date) monthnames,year(date) cal_year,SUM(target) AS budget
					

				FROM    TSalesbudget AS dpstwise_target 
					INNER JOIN dbo.dpstmaster_new 
						ON dbo.dpstmaster_new.DpstCode = dpstwise_target.Dpst 
					INNER JOIN dbo.AreaTmp 
						ON dbo.AreaTmp.AreaCode = dpstwise_target.Area 
					INNER JOIN TSalesMapping AS BDM_Inter 
						ON dbo.AreaTmp.areaname = BDM_Inter.areaname 
						AND dbo.dpstmaster_new.DpstCode = BDM_Inter.dpst

				WHERE	convert(Datetime,date) >= @ytddate 
					and convert(Datetime,date) < dateadd(yy,1,@ytddate) 
					and convert(Datetime,date) <= dateadd(dd,-1,getdate()) 
					and BDM_Inter.organ='COSEA' 
					and dpstwise_target.organ='COSEA'

				GROUP BY dbo.dpstmaster_new.Division, BDM_Inter.vertical1,dpstcode,dpstname,month(date),datename(mm,date),year(date)
				order by year(date),month(date)

declare @ytddate varchar(20)='04/01/2015'

				SELECT 	BDM_Inter.vertical1 as user_names,dbo.dpstmaster_new.Division,dpstcode,dpstname,
				month(date) as monthnum,datename(mm,date) monthnames,year(date) as cal_year,SUM(actuals) AS Actuals
				
			FROM (
				select 	dpst,convert(Datetime,date) date, sum(assesvalue) as actuals,area
				from 	slsdespdata (nolock) 
				where 	saletype in (0,2,5,6)
					and convert(Datetime,date) >= @ytddate 
					and convert(Datetime,date) < dateadd(yy,1,@ytddate) 
				group by dpst,convert(Datetime,date) ,area
			     ) AS dpstwise_actuals 
			INNER JOIN dbo.dpstmaster_new 
				ON dbo.dpstmaster_new.DpstCode = dpstwise_actuals.Dpst 
			INNER JOIN dbo.AreaTmp 
				ON dbo.AreaTmp.AreaCode = dpstwise_actuals.Area 
			INNER JOIN TSalesMapping AS BDM_Inter 
				ON dbo.AreaTmp.areaname = BDM_Inter.areaname 
				AND dbo.dpstmaster_new.DpstCode = BDM_Inter.dpst

			WHERE 	(dbo.dpstmaster_new.Description = 'exports') 
				AND (dbo.dpstmaster_new.Division <> 'AED') 
				and convert(Datetime,date) >= @ytddate 
				and convert(Datetime,date) < dateadd(yy,1,@ytddate) 
				and BDM_Inter .organ='COSEA' 
                       
			GROUP BY dbo.dpstmaster_new.Division, BDM_Inter.vertical1,dpstcode,dpstname,month(date),datename(mm,date),year(date)
			order by year(date),month(date)

declare @ytddate varchar(20)='04/01/2015'
declare @monthnum int =(select case when monthno between 1 and 3 then monthno + 9 else monthno - 3 end as monthnum from (select month(getdate()-3) as monthno) a)
--select convert(varchar,case when @monthnum=1 then year(dateadd(yy,-1,convert(date,@ytddate))) else year(convert(date,@ytddate)) end)+'-'




SELECT 	BDM_Inter.vertical1 as user_names,month(date) as monthnum,	datename(mm,date) monthnames,SUM(actuals) AS budget_month, 
				dbo.dpstmaster_new.Division,dpstcode,dpstname
FROM (
				select 	dpst,convert(Datetime,date) date, sum(assesvalue) as actuals,area
				from 	slsdespdata (nolock) 
				where 	
					convert(Datetime,date) >=dateadd(mm,-1,convert(datetime,@ytddate))  -- take data from last year last month.
					and saletype in (0,2,5,6)
 			group by dpst,convert(Datetime,date) ,area
	    ) AS dpstwise_actuals 
			INNER JOIN dbo.dpstmaster_new 
				ON dbo.dpstmaster_new.DpstCode = dpstwise_actuals.Dpst 
			INNER JOIN dbo.AreaTmp 
				ON dbo.AreaTmp.AreaCode = dpstwise_actuals.Area 
			INNER JOIN TSalesMapping AS BDM_Inter 
				ON dbo.AreaTmp.areaname = BDM_Inter.areaname 
				AND dbo.dpstmaster_new.DpstCode = BDM_Inter.dpst

			WHERE 	(dbo.dpstmaster_new.Description = 'exports') 
				AND (dbo.dpstmaster_new.Division <> 'AED') 
				and BDM_Inter .organ='COSEA' 
				and convert(Date,date) >=
				convert(varchar,year(dateadd(mm,-1,getdate()-1)))+'-'+ RIGHT('00'+CAST(month(dateadd(mm,-1,getdate()-1)) AS VARCHAR(2)),2)+'-01'
				and convert(Date,date) < 
				convert(varchar,year(getdate()-1))+'-'+ RIGHT('00'+CAST(month(getdate()-1) AS VARCHAR(2)),2)+'-01'
				
                       
			GROUP BY dbo.dpstmaster_new.Division, BDM_Inter.vertical1,dpstcode,dpstname,month(date),datename(mm,date)
