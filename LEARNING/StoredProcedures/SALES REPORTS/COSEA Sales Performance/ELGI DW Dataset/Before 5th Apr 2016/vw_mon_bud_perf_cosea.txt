SELECT 
user_names, 
Division, 
SUM(budget_year) AS budget_year, 
SUM(last_month_budget) AS last_month_budget, 
SUM(ytd_budget) AS ytd_budget,
SUM(this_month_budget) AS this_month_budget, 
SUM(next_month_budget) AS next_month_budget, 
SUM(nextof_next_month_budget)AS nextof_next_month_budget   

FROM (  

SELECT 
vertical1 as user_names , 
SUM(CASE WHEN date BETWEEN 
'04/01/'+convert(varchar,year(getdate())) 
AND 
'03/31/'+convert(varchar,year(getdate())+1) 
THEN (target) ELSE 0 END) AS budget_year,

SUM(CASE WHEN date BETWEEN 
'04/01/'+convert(varchar,year(getdate())) 
AND 
dateadd(dd, - 1, CONVERT(varchar, datepart(mm, dateadd(dd, - 1, getdate())))+ '/1/' + datename(yy, dateadd(dd, - 1, getdate()))) 
THEN (target) ELSE 0 END) AS last_month_budget, 

SUM(CASE WHEN date BETWEEN 
'04/01/'+convert(varchar,year(getdate())) 
AND 
dateadd(dd, - 1, getdate()) 
THEN (target) ELSE 0 END) AS ytd_budget, 

SUM(CASE WHEN date BETWEEN 
CONVERT(varchar, datepart(mm, dateadd(dd, - 1, getdate()))) + '/1/' + datename(yy, dateadd(dd, - 1, getdate())) 
AND 
dateadd(dd, - 1, getdate()) 
THEN (target) ELSE 0 END) AS this_month_budget, 
SUM(CASE WHEN month(date) = datepart(mm, dateadd(mm, 1, dateadd(dd, - 1, getdate()))) 
AND 
year(date) = datepart(yy, dateadd(mm, 1, dateadd(dd, - 1, getdate()))) 
THEN (target) ELSE 0 END) AS next_month_budget, 

SUM(CASE WHEN 
month(date) = datepart(mm, dateadd(mm, 2, dateadd(dd, - 1, getdate()))) 
AND 
year(date) = datepart(yy, dateadd(mm, 2,dateadd(dd, - 1, getdate()))) 
THEN (target) ELSE 0 END) AS nextof_next_month_budget, 
vertical2 as Division  

FROM dbo.tar1 AS dpstwise_target   
INNER JOIN dbo.AreaTmp 
ON dbo.AreaTmp.AreaCode = dpstwise_target.Area   
INNER JOIN reg_vert_map AS BDM_Inter 
ON AreaTmp.areaname = BDM_Inter.areaname AND dpstwise_target.Dpst = BDM_Inter.dpst  

WHERE       vertical1 <> ''  
GROUP BY vertical1,vertical2) AS budget  
GROUP BY Division, user_names  

HAVING (SUM(budget_year) > 0) or (SUM(last_month_budget) > 0)
