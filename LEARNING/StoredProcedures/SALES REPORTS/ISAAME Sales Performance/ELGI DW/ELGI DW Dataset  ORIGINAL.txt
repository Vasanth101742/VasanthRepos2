SELECT  vw_mon_bud_perf_region.budget_year,
	vw_mon_bud_perf_region.last_month_budget, 
	vw_mon_bud_perf_region.ytd_budget, 
	vw_mon_bud_perf_region.this_month_budget, 
	vw_mon_bud_perf_region.next_month_budget, 
	vw_mon_bud_perf_region.nextof_next_month_budget, 
	vw_mon_sls_perf_region.last_year_sale,
	vw_mon_sls_perf_region.last_year_ytd, 
	vw_mon_sls_perf_region.thr_month_sale,
	vw_mon_sls_perf_region.this_month_sale, 
	vw_mon_sls_perf_region.this_year_sale, 
	vw_mon_sls_perf_region.last_month_sale,
	vw_mon_bud_perf_region.global ,
	n1_month_sale,
	n2_month_sale,
	n3_month_sale

FROM	dbo.vw_mon_bud_perf_region_new  as vw_mon_bud_perf_region 
	INNER JOIN dbo.vw_mon_sls_perf_region_new as vw_mon_sls_perf_region  
	ON vw_mon_bud_perf_region.global = vw_mon_sls_perf_region.global

where 	vw_mon_bud_perf_region.global = 'Domestic'


union all

select 	sum(budget_year) as budget_year,
	sum(last_month_budget) as last_month_budget,
	isnull(sum(ytd_budget),0) as ytd_budget,
	sum(this_month_budget) as this_month_budget, 
	sum(next_month_budget) as next_month_budget,
	sum(nextof_next_month_budget) as nextof_next_month_budget,
	sum(last_year_sale) as last_year_sale,
	sum(last_year_ytd) as last_year_ytd,
	sum(thr_month_sale) as thr_month_sale,
	sum(this_month_sale) as this_month_sale,
	sum(this_year_sale) as this_year_sale,
	sum(last_month_sale) as last_month_sale,
	'International' as user_names,
	sum(n1_month_sale) as n1_month_sale,
	sum(n2_month_sale) as n2_month_sale,
	sum(n3_month_sale) as n3_month_sale

from(

	SELECT	vw_mon_bud_perf_int_bdm.budget_year,   
		vw_mon_bud_perf_int_bdm.last_month_budget, 
		vw_mon_bud_perf_int_bdm.ytd_budget,   
		vw_mon_bud_perf_int_bdm.this_month_budget, 
		vw_mon_bud_perf_int_bdm.next_month_budget,   
		vw_mon_bud_perf_int_bdm.nextof_next_month_budget,
		vw_mon_sls_perf_int_bdm.last_year_sale, 
		case when  vw_mon_sls_perf_int_bdm.last_year_ytd = 0 then  0.01 
		     else  vw_mon_sls_perf_int_bdm.last_year_ytd end as last_year_ytd,
		vw_mon_sls_perf_int_bdm.this_year_sale,   
		vw_mon_sls_perf_int_bdm.last_month_sale, 
		vw_mon_sls_perf_int_bdm.this_month_sale,
		vw_mon_sls_perf_int_bdm.thr_month_sale,
		case when vw_mon_bud_perf_int_bdm.user_names is null then vw_mon_sls_perf_int_bdm.user_names 
		     else vw_mon_bud_perf_int_bdm.user_names end as user_names,
		n1_month_sale,
		n2_month_sale,
		n3_month_sale,
		case when vw_mon_bud_perf_int_bdm.division is null then vw_mon_sls_perf_int_bdm.division 
		else vw_mon_bud_perf_int_bdm.division end as division

	FROM	(select * from vw_mon_bud_perf_int_org_new  where organ = 'ISAAME')    vw_mon_bud_perf_int_bdm 
		full JOIN
		(select * from vw_mon_sls_perf_int_org_new  where organ = 'ISAAME')    vw_mon_sls_perf_int_bdm 
		ON   vw_mon_bud_perf_int_bdm.division =   vw_mon_sls_perf_int_bdm.division 
		and  vw_mon_bud_perf_int_bdm.region=vw_mon_sls_perf_int_bdm.region
		and  vw_mon_bud_perf_int_bdm.user_names =   vw_mon_sls_perf_int_bdm.user_names
     ) BDM
